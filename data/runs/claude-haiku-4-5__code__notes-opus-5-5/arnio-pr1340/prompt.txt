# arnio.integrations.sklearn.ArnioCleaner.get_feature_names_out
    def get_feature_names_out(self, input_features=None):
        """Enable compatibility with ColumnTransformer workflows.

        Parameters
        ----------
        input_features : array-like of str or None
            If None, the fitted output feature names are returned.
            If an array-like, it must be a non-string sequence of feature
            names that match the fitted input feature names.

        Returns
        -------
        feature_names_out : ndarray
            The output feature names.
        """
        check_is_fitted(self, "feature_names_out_")

        if input_features is None:
            return self.feature_names_out_

        if isinstance(input_features, (str, bytes)):
            raise TypeError("input_features must be a sequence of strings, not str")

        if not hasattr(input_features, "__len__") or not hasattr(
            input_features, "__iter__"
        ):
            raise TypeError("input_features must be a sequence of strings")

        if not all(isinstance(f, str) for f in input_features):
            raise TypeError("All input_features must be strings")

        if len(input_features) != self.n_features_in_:
            raise ValueError(f"input_features should have length {self.n_features_in_}")
        if list(input_features) != list(self.feature_names_in_):
            raise ValueError("input_features must match columns seen during fit.")

        return self.feature_names_out_

def read_csv(
    path: str | os.PathLike[str] | io.TextIOBase,
    *,
    delimiter: str | None = None,
    has_header: bool = True,
    usecols: list[str] | None = None,
    nrows: int | None = None,
    skiprows: int | None = None,
    encoding: str = "utf-8",
    trim_headers: bool = True,
    decimal_separator: str = ".",
    thousands_separator: str | None = None,
    null_values: list[str] | None = None,
    dtype: dict[str, str] | None = None,
    mode: str = "strict",
    encoding_errors: str = "strict",
    on_bad_lines: str = "error",
) -> ArFrame:
    """Read a CSV file into an ArFrame via C++ backend.

    Parameters
    ----------
    path : str or file-like object
        Filesystem path or text file-like object containing CSV data.
        Any file extension is accepted, including compressed ``.csv.gz`` files.
        For ``.tsv`` files, the delimiter
        is automatically set to ``'\t'`` when ``delimiter`` is omitted.
    delimiter : str or None, default None
        Field delimiter character.  When ``None`` (the default) the
        delimiter is inferred from the file extension: ``'\t'`` for
        ``.tsv`` files and ``','`` for everything else.  Passing an
        explicit value always takes precedence — for example,
        ``delimiter=','`` reads a comma-delimited ``.tsv`` file without
        any auto-detection.
    has_header : bool, default True
        Whether the file has a header row.
    usecols : list[str], optional
        Columns to read. If None, reads all columns.
    nrows : int, optional
        Number of rows to read. If None, reads all rows.
    skiprows : int, optional
        Number of lines to skip before reading the header. Useful for
        CSV files with metadata preambles before the actual data.
        If None, no lines are skipped.
    encoding : str, default "utf-8"
        File encoding.
    trim_headers : bool, default True
        Strip leading/trailing whitespace from column names.  Regardless
        of this setting, headers that differ only by leading or trailing
        whitespace are always rejected with a :exc:`CsvReadError` because
        they would produce ambiguous column access.
    decimal_separator : str, default "."
        Single non-alphanumeric character used as the decimal separator
        during numeric parsing. Use "," to opt in to European-style decimals
        such as ``"12,45"``. Values containing the CSV delimiter must still
        be quoted.
    thousands_separator : str, optional
        Single non-alphanumeric character used as a thousands separator
        during numeric parsing.



        Values containing delimiter characters must still be quoted
        properly in the CSV input. For example, when using a comma
        delimiter, the value "1,234" must be quoted, while unquoted
        1,234 is interpreted as two separate fields.

    dtype : dict[str, str], optional
        Explicit column dtype mapping. Specified columns skip automatic
        type inference and use the requested dtype directly.

        Supported dtypes:
        - "string"
        - "int64"
        - "float64"
        - "bool"

    mode : {"strict", "permissive"}, default "strict"
        Controls malformed row handling.

        - strict: raises CsvReadError on inconsistent row widths.
        - permissive: fills missing trailing fields with nulls.
        - both modes reject extra fields because they would otherwise be
          silently dropped.

    on_bad_lines : {"error", "warn", "skip"}, default "error"
        Action to take on rows classified as bad by ``mode``.

        - error: raise CsvReadError on the first bad row.
        - warn: drop the row and emit a UserWarning.
        - skip: drop the row silently.

        In permissive mode, narrow rows are still padded silently and do
        not reach this dispatch; only wide rows do. Dropped rows count
        toward ``nrows``.

    Returns
    -------
    ArFrame
        Data frame containing the CSV data.

    Raises
    ------
    ValueError
        If thousands_separator is invalid.

    TypeError
        If delimiter is not a string or None, or thousands_separator is
        not a string or None.

    CsvReadError
        If CSV input contains NUL bytes and appears binary or corrupted.

    Examples
    --------
    >>> import arnio as ar

    Read a basic CSV file:

    >>> df = ar.read_csv("data.csv")              # comma delimiter

    Read a CSV with specific columns and row limit:

    >>> df = ar.read_csv("large_data.csv", usecols=["id", "name"], nrows=1000)

    Other important behaviors:

    >>> df = ar.read_csv("data.tsv")              # tab auto-detected
    >>> df = ar.read_csv("data.tsv", delimiter=",")  # explicit comma honoured
    >>> df = ar.read_csv("data.dat")              # non-standard extension accepted
    """
    native_path, should_cleanup, is_materialized_text = _materialize_csv_input(path)

    try:
        # Explicitly validate the decompressed temp file (or local path) rather than the compressed bytes
        _validate_csv_path(native_path, encoding)

        path_lower = native_path.lower()

        # Resolve the sentinel: auto-detect tab for .tsv only when the caller
        # truly omitted delimiter (None).  An explicit delimiter="," is always
        # honoured, even for .tsv paths.
        if delimiter is None:
            delimiter = "\t" if path_lower.endswith(".tsv") else ","

        decimal_separator = _validate_decimal_separator(decimal_separator)
        _validate_thousands_separator(thousands_separator, decimal_separator)
        delimiter = _validate_delimiter(delimiter)
        mode = _validate_parser_mode(mode)
        encoding_errors = _validate_encoding_errors(encoding_errors)
        on_bad_lines = _validate_on_bad_lines(on_bad_lines)
        config = _CsvConfig()
        config.delimiter = delimiter
        config.has_header = _validate_bool_option(has_header, "has_header")
        config.encoding = encoding
        config.trim_headers = _validate_bool_option(trim_headers, "trim_headers")
        config.decimal_separator = decimal_separator
        config.thousands_separator = thousands_separator
        config.mode = mode
        config.encoding_errors = encoding_errors
        if null_values is not None:
            config.null_values = _validate_null_values(null_values)
        if dtype is not None:
            config.dtype = _validate_dtype_mapping(dtype)

        if usecols is not None:
            config.usecols = _validate_usecols(usecols)

        if nrows is not None:
            config.nrows = _validate_nrows(nrows)

        if skiprows is not None:
            config.skip_rows = _validate_skip_rows(skiprows)

        reader = _CsvReader(config)
    except Exception:
        if should_cleanup and os.path.exists(native_path):
            os.unlink(native_path)
        raise

    try:
        effective_encoding = "utf-8" if is_materialized_text else encoding
        with _utf8_csv_path(
            native_path,
            effective_encoding,
            encoding_errors=encoding_errors,
            delimiter=delimiter,
        ) as native_csv_path:
            try:
                cpp_frame, bad_rows = reader.read(native_csv_path, on_bad_lines)
            except CsvReadError:
                raise
            except (ValueError, TypeError):
                raise
            except RuntimeError as e:
                raise _enrich_csv_runtime_error(
                    e, native_path, encoding, delimiter
                ) from None

        if on_bad_lines == "warn" and bad_rows:
            _warn_bad_rows(bad_rows)

        frame = ArFrame(cpp_frame)

        # Case 2: Delimiter mismatch — check only when usecols was not restricted
        # (usecols can legitimately produce 1 column) and has_header is True
        # so we can peek at the data line.
        if usecols is None and has_header:
            _warn_delimiter_mismatch(native_path, delimiter, frame.shape[1])

        return frame

    except (ValueError, TypeError):
        raise
    except CsvReadError:
        raise
    except Exception as e:
        raise CsvReadError(str(e)) from None

    finally:
        if should_cleanup and os.path.exists(native_path):
            os.unlink(native_path)

def read_csv_chunked(
    path: str | os.PathLike[str] | io.TextIOBase,
    *,
    chunksize: int = 10_000,
    dtype: dict[str, str] | None = None,
    delimiter: str | None = None,
    has_header: bool = True,
    usecols: list[str] | None = None,
    nrows: int | None = None,
    skip_rows: int = 0,
    skiprows: int | None = None,
    encoding: str = "utf-8",
    trim_headers: bool = True,
    decimal_separator: str = ".",
    thousands_separator: str | None = None,
    null_values: list[str] | None = None,
    mode: str = "strict",
    on_bad_lines: str = "error",
) -> Iterator[ArFrame]:
    """Read a CSV file in chunks, yielding ArFrame objects.

    Column types are inferred from the first chunk and applied consistently
    to all subsequent chunks. Memory use is bounded by the chunk size.

    Parameters
    ----------
    path : str or file-like object
        Path to the CSV file. Supports .csv, .txt, .tsv, and compressed .csv.gz extensions.
        Text file-like objects are copied to a temporary file in bounded
        chunks before native parsing.  For ``.tsv`` paths the delimiter is
        automatically set to ``'\\t'`` when ``delimiter`` is omitted.
    chunksize : int, default 10_000
        Maximum number of data rows per yielded chunk.
    delimiter : str or None, default None
        Field delimiter character.  When ``None`` (the default) the
        delimiter is inferred from the file extension: ``'\\t'`` for
        ``.tsv`` files and ``','`` for everything else.  Passing an
        explicit value always takes precedence — for example,
        ``delimiter=','`` reads a comma-delimited ``.tsv`` file without
        any auto-detection.
    has_header : bool, default True
        Whether the file has a header row.
    usecols : list[str], optional
        Columns to read. If None, reads all columns.
    nrows : int, optional
        Maximum total number of data rows to read across all chunks.
    skip_rows : int, default 0
        Number of data rows to skip after the header row.
    skiprows : int, optional
        Alias for ``skip_rows`` for API consistency with ``read_csv``.
        Note: in chunked mode both ``skip_rows`` and ``skiprows`` skip
        data rows *after* the header, not lines before it.
        If both are supplied they must agree; conflicting values raise
        ``ValueError``.
    encoding : str, default "utf-8"
        File encoding.
    trim_headers : bool, default True
        Strip leading/trailing whitespace from column names.  Regardless
        of this setting, headers that differ only by leading or trailing
        whitespace are always rejected with a :exc:`CsvReadError` because
        they would produce ambiguous column access.
    decimal_separator : str, default "."
        Single non-alphanumeric character used as the decimal separator
        during numeric parsing.
    thousands_separator : str, optional
        Single non-alphanumeric character used as a thousands separator
        during numeric parsing.
    null_values : list[str], optional
        Strings treated as null values.

    mode : {"strict", "permissive"}, default "strict"
        Controls malformed row handling.
        Both modes reject extra fields; permissive mode only allows missing
        trailing fields, which are filled with nulls.
    on_bad_lines : {"error", "warn", "skip"}, default "error"
        Action to take on rows classified as bad by ``mode``.

        - error: raise CsvReadError on the first bad row.
        - warn: drop the row and emit a UserWarning.
        - skip: drop the row silently.

        In permissive mode, narrow rows are still padded silently and do
        not reach this dispatch; only wide rows do. Dropped rows count
        toward ``nrows``.

    Yields
    ------
    ArFrame
        Successive chunks of the CSV data.

    Examples
    --------
    >>> for chunk in ar.read_csv_chunked("huge.csv", chunksize=100_000):
    ...     clean = ar.pipeline(chunk, [("drop_nulls",)])
    ...     df = ar.to_pandas(clean)
    ...     process(df)

    Read a TSV file — tab delimiter is inferred automatically:

    >>> for chunk in ar.read_csv_chunked("data.tsv", chunksize=10_000):
    ...     process(chunk)

    Override auto-detection (e.g. a comma-delimited file with a .tsv extension):

    >>> for chunk in ar.read_csv_chunked("data.tsv", delimiter=",", chunksize=10_000):
    ...     process(chunk)
    """
    is_path_input = isinstance(path, (str, os.PathLike))
    native_path, should_cleanup, is_materialized_text = _materialize_csv_input(
        path, caller="read_csv_chunked"
    )
    try:
        path_lower = native_path.lower()
        if is_path_input:
            # We check the original path extension if it was passed as a path
            if isinstance(path, str):
                orig_path_lower = path.lower()
            elif isinstance(path, os.PathLike):
                orig_path_lower = os.fspath(path).lower()
            else:
                orig_path_lower = ""

            if not (
                orig_path_lower.endswith(".csv")
                or orig_path_lower.endswith(".txt")
                or orig_path_lower.endswith(".tsv")
                or orig_path_lower.endswith(".gz")
            ):
                raise ValueError(
                    f"Unsupported file format: {path}. "
                    "Only .csv, .txt, .tsv, and compressed .csv.gz are supported."
                )

        # Explicitly validate the decompressed temp file (or local path) rather than the compressed bytes
        _validate_csv_path(native_path, encoding, reject_utf8_nul_bytes=False)

        # Resolve the sentinel: auto-detect tab for .tsv only when the caller
        # truly omitted delimiter (None).  An explicit delimiter="," is always
        # honoured, even for .tsv paths.  File-like objects are materialised
        # to a temporary .csv path, so auto-detection safely falls back to ","
        # for those inputs — consistent with read_csv behaviour.
        if delimiter is None:
            delimiter = "\t" if path_lower.endswith(".tsv") else ","

        decimal_separator = _validate_decimal_separator(decimal_separator)
        _validate_thousands_separator(thousands_separator, decimal_separator)
        delimiter = _validate_delimiter(delimiter)
        mode = _validate_parser_mode(mode)
        chunksize = _validate_chunksize(chunksize)

        # Resolve skiprows / skip_rows alias.
        # Both skip data rows after the header in chunked mode.
        # skip_rows is kept for backward compatibility; skiprows matches
        # the read_csv parameter name. Both may be passed as long as they
        # agree; conflicting values raise ValueError.
        if skiprows is not None:
            if isinstance(skiprows, bool) or not isinstance(skiprows, int):
                raise TypeError("skiprows must be an integer")
            if skiprows < 0:
                raise ValueError("skiprows must be non-negative")
            if skip_rows != 0 and skip_rows != skiprows:
                raise ValueError(
                    f"Conflicting values: skiprows={skiprows!r} and "
                    f"skip_rows={skip_rows!r}. Pass only one of them."
                )
            skip_rows = skiprows

        skip_rows = _validate_skip_rows(skip_rows)
        on_bad_lines = _validate_on_bad_lines(on_bad_lines)

        config = _CsvConfig()
        config.delimiter = delimiter
        config.has_header = _validate_bool_option(has_header, "has_header")
        config.encoding = encoding
        config.trim_headers = _validate_bool_option(trim_headers, "trim_headers")
        config.decimal_separator = decimal_separator
        config.thousands_separator = thousands_separator
        config.mode = mode
        config.skip_rows = skip_rows

        if null_values is not None:
            config.null_values = _validate_null_values(null_values)

        if dtype is not None:
            config.dtype = _validate_dtype_mapping(dtype)

        if usecols is not None:
            config.usecols = _validate_usecols(usecols)

        if nrows is not None:
            config.nrows = _validate_nrows(nrows)

        reader = _CsvChunkReader(config)
    except Exception:
        if should_cleanup and os.path.exists(native_path):
            os.unlink(native_path)
        raise
    try:
        effective_encoding = "utf-8" if is_materialized_text else encoding
        with _utf8_csv_path(
            native_path, effective_encoding, delimiter=delimiter
        ) as native_csv_path:
            reader.open(native_csv_path)
            yielded_nonempty_chunk = False
            try:
                while True:
                    chunk = reader.next_chunk(chunksize, on_bad_lines)
                    if chunk is None:
                        break
                    cpp_frame, bad_rows = chunk

                    if on_bad_lines == "warn" and bad_rows:
                        _warn_bad_rows(bad_rows)
                    frame = ArFrame(cpp_frame)

                    if frame.shape[0] == 0 and bad_rows:
                        if yielded_nonempty_chunk:
                            continue

                    yielded_nonempty_chunk = (
                        yielded_nonempty_chunk or frame.shape[0] > 0
                    )

                    yield frame
            finally:
                reader.close()
                if should_cleanup and os.path.exists(native_path):
                    try:
                        os.unlink(native_path)
                    except OSError:
                        pass
    except ValueError:
        raise
    except CsvReadError:
        raise
    except RuntimeError as e:
        raise CsvReadError(str(e)) from None

### arnio.integrations.sklearn.ArnioCleaner.get_feature_names_out

# Notes: `ArnioCleaner.get_feature_names_out(self, input_features=None)`

## 1. Purpose
- Returns the fitted output column names, for sklearn `ColumnTransformer` compatibility.
- When `input_features` is given, it checks that argument against the columns seen at fit time.
- It never computes names from `input_features`. It always returns the stored `self.feature_names_out_`.

## 2. Inputs
- `input_features` accepts:
  - `None` (the default), or
  - a non-string sized iterable of `str`: a list, tuple, `pd.Index` of strings, or numpy str array. `np.str_` is a subclass of `str`, so it passes.
- Validation runs only when `input_features is not None`, in this order:
  1. Is it `str` or `bytes`?
  2. Does it have both `__len__` and `__iter__`?
  3. Is every element an instance of `str`?
  4. Is `len == self.n_features_in_`?
  5. Is `list(input_features) == list(self.feature_names_in_)`? This is exact and order-sensitive.
- Fitted state it relies on:
  - `feature_names_in_` is `np.array(X.columns, dtype=object)`.
  - `n_features_in_` is `X.shape[1]`.
  - Both are set by `fit`.

## 3. Raises (in check order)
1. **Not fitted.** `check_is_fitted(self, "feature_names_out_")` raises `sklearn.exceptions.NotFittedError`, which subclasses both `ValueError` and `AttributeError`.
   - The message text comes from sklearn: "This ArnioCleaner instance is not fitted yet. Call 'fit' ...". Match only the substring `"not fitted"`.
   - Trigger: `ar_sk.ArnioCleaner().get_feature_names_out()`. It fires even when `input_features=None`.
2. **String or bytes.** `TypeError("input_features must be a sequence of strings, not str")`.
   - Triggers: `est.get_feature_names_out("a")` and `est.get_feature_names_out(b"a")`.
3. **No `__len__` or no `__iter__`.** `TypeError("input_features must be a sequence of strings")`.
   - Triggers: `est.get_feature_names_out(5)`, or a generator `(c for c in ["a","b"])`, which has `__iter__` but no `__len__`.
4. **Non-string element.** `TypeError("All input_features must be strings")`.
   - Trigger: `est.get_feature_names_out([1, 2])`.
   - This check comes before the length check, so `[1]` on a 2-column fit gives this TypeError, not a ValueError.
5. **Wrong length.** `ValueError(f"input_features should have length {self.n_features_in_}")`.
   - Trigger: fit on columns `a,b`, then `est.get_feature_names_out(["a"])` gives `"input_features should have length 2"`.
6. **Wrong names or order.** `ValueError("input_features must match columns seen during fit.")`.
   - Triggers: `["b","a"]` or `["a","c"]` on a fit with `a,b`.
- Use `re.escape` in `match=`, because messages contain `.`.

## 4. Returns and guarantees
- Returns the attribute object itself: `est.get_feature_names_out() is est.feature_names_out_` is True.
  - This holds both with `None` and with valid `input_features`.
  - It is not a copy, and it is never the passed `input_features`.
- The result is a `numpy.ndarray` with `dtype == object`, holding Python `str` column names.
- The value depends on what was run last:
  - **After `fit`** with `allow_schema_changes=False`, or with an empty `steps`: a copy of `feature_names_in_`, i.e. the input column order. It is a different object from `feature_names_in_` but equal in content.
  - **After `fit`** with `allow_schema_changes=True` and non-empty `steps`: the columns of the pipeline run on a probe copy of X.
  - **After `transform`**: overwritten with `np.array(X_out.columns, dtype=object)`, the columns of the transform output.

## 5. Side effects
- None. It reads attributes only and does not mutate the estimator or `input_features`.

## 6. Branches and edge cases
- **`input_features=None`**: returns early with no validation, e.g. `['a','b']` after `fit(pd.DataFrame({"a":[1],"b":[2]}))`.
- **Valid `input_features`** (`["a","b"]`, `("a","b")`, `pd.Index(["a","b"])`, `np.array(["a","b"])`): all return the same object as with `None`.
- **Schema-changing fit**: `steps=[("rename_columns", {"mapping": {"a": "x"}})]` with `allow_schema_changes=True`.
  - The output is expected to be `['x','b']`, while `input_features` must still be `["a","b"]`.
  - The step-argument format depends on `arnio.pipeline`, which is not shown. Verify it, or prefer the non-schema cases.
  - Without `allow_schema_changes=True`, `fit` raises `ValueError("Schema-changing step 'rename_columns' not allowed. Use allow_schema_changes=True.")`.
- **Non-str column names** (e.g. int columns `0,1`): passing `[0,1]` hits the "All input_features must be strings" TypeError. Avoid this case otherwise, since `from_pandas` behaviour with it is unknown.

## 7. Rules from the docstring
- "If None, the fitted output feature names are returned."
- Otherwise the argument "must be a non-string sequence of feature names that match the fitted input feature names".

## 8. Testing hints
- Setup:
  ```python
  import pandas as pd, numpy as np, pytest, re
  from arnio.integrations.sklearn import ArnioCleaner
  df = pd.DataFrame({"a": [1, 2], "b": ["x", "y"]})
  est = ArnioCleaner().fit(df)   # steps=[] -> no pipeline run in fit
  ```
- Compare arrays with `list(out) == ["a","b"]` or `np.array_equal`. Never use bare `assert out == [...]`.
- Also assert `out.dtype == object` and `out is est.feature_names_out_`.
- Pin the check order:
  - `[1]` on a 2-column fit gives TypeError (non-string), not ValueError (length).
  - `["a"]` gives the length error.
  - `["b","a"]` gives the mismatch error.
- A safe step that keeps the schema: `steps=[("strip_whitespace",)]`. After `transform`, the names should still be `["a","b"]`.
- Requires scikit-learn to be installed. Use `pytest.importorskip("sklearn")`.

### arnio.io.read_csv

# Notes: `arnio.io.read_csv`

## 1. Purpose
- Reads CSV text (local path, http(s) URL, `.gz` file, or text file-like object) into an `ArFrame`.
- Parsing is done by the C++ `_CsvReader`. The Python layer validates options, materialises inputs to a UTF-8 temp file, and wraps errors and warnings.

## 2. Inputs
All parameters after `path` are keyword-only.

- **`path`**
  - `str` / `os.PathLike`:
    - For `str` only, the URL scheme is checked (lower-cased).
    - Cloud schemes `s3, gs, gcs, az, abfs, abfss` → ValueError.
    - `http`/`https` are fetched via urllib (network; avoid in tests).
  - Paths ending `.gz` (case-insensitive) are gunzipped to a temp file with suffix **`.csv`**.
  - Any object with a callable `.read()` (e.g. `io.StringIO`) is copied to a temp `.csv`. Each `read()` must return `str`.
  - Any file extension is accepted (`.dat`, `.txt`, and so on).
- **`delimiter=None`**
  - `None` resolves to `"\t"` if the *materialised* path lower-cased ends with `.tsv`, otherwise `","`.
  - So `x.TSV` → tab. But `x.tsv.gz` → `","` and StringIO → `","`, because their temp files end `.csv`.
  - An explicit value always wins.
  - Validated rules:
    - Must be a `str` of length 1.
    - Not `\n` or `\r`.
    - Not `"`.
    - Not a control char (code point ≤0x1F except tab 0x09, or 0x7F).
- **`has_header=True`, `trim_headers=True`**: must be real `bool` (an `int` such as `1` is rejected).
- **`encoding="utf-8"`**
  - Must be `str`.
  - `"utf-8"`, `"UTF8"`, `"utf_8"` are treated as UTF-8 (lower-cased, `_`→`-`).
  - Any other value transcodes the file through a temp UTF-8 file.
  - For file-like/URL inputs, transcoding is skipped (they are already text).
- **`decimal_separator="."`**: one char, not alphanumeric, not `"`, `\n`, `\r`, `+`, `-`.
- **`thousands_separator=None`**: same rules as `decimal_separator`, and must differ from `decimal_separator`.
  - Not checked against `delimiter`.
- **`mode`**: `"strict"` | `"permissive"`.
- **`encoding_errors`**: `"strict"` | `"replace"` | `"ignore"`.
- **`on_bad_lines`**: `"error"` | `"warn"` | `"skip"`.
- **`null_values`**: `list` of `str` only (a tuple is rejected).
- **`dtype`**: `dict[str, str]`; values must be in `{"string", "int64", "float64", "bool"}`.
- **`usecols`**: non-string Sequence of `str`, non-empty, no duplicates.
- **`nrows`**: int ≥ 0 (`bool` rejected; `0` allowed).
- **`skiprows`**: int ≥ 0; skips lines *before* the header.

## 3. Raises
`CsvReadError` is in `arnio.exceptions`. Checks run in the order listed.

1. **Input type / scheme** (before any cleanup block):
   - `read_csv(123)` → `TypeError("read_csv expected a filesystem path, a URL, or a text file-like object")`.
   - A file-like whose `read()` returns bytes (e.g. `io.BytesIO(b"a\n1\n")`) → `TypeError("read_csv file-like objects must return text, not bytes")`.
   - `read_csv("s3://b/k.csv")` → `ValueError("Cloud scheme 's3' is not yet supported by arnio. Install the optional extra when available: pip install \"arnio[s3]\"")`.
     - The message literally contains `pip install "arnio[s3]"`.
     - `gs`/`gcs` → `arnio[gcs]`; `az`/`abfs`/`abfss` → `arnio[azure]`.
2. **Encoding type**: `encoding=5` → `TypeError("encoding must be a string, got 'int'")`.
3. **NUL bytes** (only when encoding is UTF-8): a file containing `b"\0"` → `CsvReadError("CSV input contains NUL bytes and appears to be binary or corrupted")`.
4. **Zero-byte file** → `CsvReadError(f"CSV file is empty: {native_path!r}")`.
   - Happens BEFORE option validation, so `read_csv(empty, delimiter=";;")` raises the empty-file error.
   - An empty `StringIO("")` raises the same error, but with the temp path in the message; use `match="CSV file is empty"`.
5. **Option validation**, in this order:
   - `decimal_separator`:
     - non-str → `TypeError("decimal_separator must be a string")`
     - `",,"` → `ValueError("decimal_separator must be a single character")`
     - `"a"` → `"decimal_separator must be a single non-alphanumeric character"`
     - `"+"` → `"Invalid decimal_separator: '+' and '-' are not allowed"`
   - `thousands_separator`: same messages with `thousands_separator` in place of `decimal_separator`. The TypeError reads `"thousands_separator must be a string or None"`.
     - `thousands_separator="."` (with default decimal) → `ValueError("thousands_separator must differ from decimal_separator")`.
   - `delimiter`:
     - `1` → `TypeError("delimiter must be a string")`
     - `";;"` → `ValueError("delimiter must be a single character; delimiter must be exactly one character")`
     - `"\n"` → `"delimiter must not be a newline character"`
     - `'"'` → `"delimiter must not be the CSV quote character"`
     - `"\x01"` → `"delimiter must not be a control character"`
   - `mode`:
     - non-str → `TypeError("mode must be a string")`
     - `"x"` → `ValueError("mode must be either 'strict' or 'permissive'")`
   - `encoding_errors`:
     - non-str → `TypeError("encoding_errors must be a string")`
     - bad value → `ValueError("encoding_errors must be one of 'strict', 'replace', or 'ignore'")`
   - `on_bad_lines`:
     - non-str → `TypeError("on_bad_lines must be a string")`
     - bad value → `ValueError("on_bad_lines must be either 'error', 'warn', 'skip'")`
   - `has_header=1` → `TypeError("has_header must be True or False, got int: 1")`. `trim_headers` uses the same template.
   - `null_values`:
     - `"NA"` → `TypeError("null_values must be a list of strings, not a bare string")`
     - `("NA",)` → `"null_values must be a list of strings"`
     - `[1]` → `"null_values must contain only strings"`
   - `dtype`:
     - `[]` → `TypeError("dtype must be a dictionary mapping column names to dtype strings")`
     - `{1: "int64"}` → `"dtype column names must be strings"`
     - `{"a": 1}` → `"dtype values must be strings"`
     - `{"a": "int32"}` → `ValueError("Unsupported dtype 'int32'. Expected one of: ['bool', 'float64', 'int64', 'string']")`
   - `usecols`:
     - `"a"` → `TypeError("usecols must be a sequence of column names, not a string")`
     - `{"a"}` → `"usecols must be a sequence of strings"`
     - `[]` → `ValueError("usecols must not be empty")`
     - `[1]` → `TypeError("usecols must contain only strings")`
     - `["a", "a"]` → `ValueError("usecols must not contain duplicate column names")`
   - `nrows`:
     - `True` / `1.0` → `TypeError("nrows must be an integer")`
     - `-1` → `ValueError("nrows must be non-negative")`
   - `skiprows` (the message says **skip_rows**):
     - `"1"` → `TypeError("skip_rows must be an integer")`
     - `-1` → `ValueError("skip_rows must be non-negative")`
6. **Non-UTF-8 encoding paths**:
   - `encoding="bogus"` on a real file → `ValueError("Unknown encoding: bogus")` (no quotes).
   - Undecodable bytes with e.g. `encoding="ascii"` and file bytes `b"a\n\xe9\n"` → `CsvReadError("Could not decode '<path>' using encoding 'ascii': invalid byte(s) at position 2 (byte value: 0xe9). Try a different encoding or use encoding_errors='replace'.")`.
   - Missing file with a non-UTF-8 encoding → `CsvReadError("Could not access CSV file '<path>': No such file or directory")`.
7. **Native parse errors**: a `RuntimeError` from C++ becomes `CsvReadError`.
   - Row-width messages are enriched to `"Malformed CSV: row N has too many|few fields (A found, E expected). To skip bad rows silently use on_bad_lines='skip', or on_bad_lines='warn' to collect them. For rows with missing trailing fields, try mode='permissive'. (<orig>)"`.
   - Whether C++ raises `RuntimeError` or `CsvReadError` directly is not visible. Only assert `pytest.raises(CsvReadError)` for `"a,b\n1,2,3\n"`.
   - A missing UTF-8 file is passed to C++. The result is most likely `CsvReadError` (exact type and message depend on C++).

## 4. Returns and guarantees
- Returns a new `ArFrame(cpp_frame)` with no attrs.
- Column names, order, inference, and null handling all come from C++ and are not visible here.
  - Per the docs, `has_header=False` gives `col_0`, `col_1`, …
  - `dtype` forces the given column types; `frame.dtypes` uses strings like `"int64"` and `"string"`.
  - With `trim_headers=True`, a header `" a "` becomes `"a"`.
  - Headers differing only by surrounding whitespace raise `CsvReadError` (doc claim, C++).

## 5. Side effects
- The input file is never modified.
- Temp files (gz, file-like, URL, transcoding) are deleted in `finally`, including on errors.
- May emit `UserWarning`s (see §6).

## 6. Branches and edge cases
- **`on_bad_lines="warn"` with bad rows**: one `UserWarning` with text `f"{n} malformed CSV row(s):\n  CSV row {r} has {a} fields; expected {e}"`.
  - At most 10 lines are listed, then `"  (+k more)"`.
  - Row numbering comes from C++; use `match="malformed CSV row"`.
- **`on_bad_lines="skip"`**: no warning; bad rows are dropped. In permissive mode, narrow rows are padded rather than dropped.
- **Delimiter-mismatch warning**: fires only if `usecols is None`, `has_header` is True, and the result has exactly 1 column.
  - A candidate in `{",", ";", "\t", "|"}` minus the delimiter must appear unquoted in both the header and the first data record.
  - Example: file `"a;b\n1;2\n"`, default delimiter → UserWarning: `"Parsed as a single column — possible delimiter mismatch. The file appears to contain ';' characters but delimiter=',' was used. Try ar.read_csv(path, delimiter=';') or ar.sniff_delimiter(path) to auto-detect."`
  - No warning for:
    - header-only files;
    - `usecols=[...]`;
    - `has_header=False`;
    - `"a;b\n1\n"` (no `;` in the data record);
    - `'"a;b"\n"1;2"\n'` (quoted).
- **`.gz` input**: `gzip.open` failures (e.g. a non-gzip file named `x.csv.gz` → `gzip.BadGzipFile`; a missing `.gz` → `FileNotFoundError`) propagate unwrapped.

## 7. Rules from comments and docstrings
- An explicit delimiter is always honoured, even for `.tsv` files.
- Validation runs on the decompressed file, not the gzip bytes.
- Dropped bad rows count toward `nrows`.
- Both modes reject extra fields.

## 8. Testing hints
- **Building inputs**: write files with `tmp_path / "x.csv"` and `.write_text(..., newline="")` or `.write_bytes(...)`. `io.StringIO("a,b\n1,2\n")` also works.
- **Observing results**: use `frame.shape`, `frame.columns`, `frame.dtypes`, `ar.to_pandas(frame)`.
  - Prefer string data or `dtype={...}` when asserting values; inferred dtypes are C++-defined.
- **Gzip and TSV**:
  - gz round-trip: `gzip.open(p, "wt").write("a,b\n1,2\n")`, then `shape == (1, 2)`.
  - `.tsv` auto-tab: `"a\tb\n1\t2\n"` → 2 columns. With `delimiter=","` → 1 column.
- **Warnings**:
  - Use `pytest.warns(UserWarning, match="possible delimiter mismatch")`.
  - Avoid tab as the mismatch candidate (its repr formatting is messy).
  - Avoid files containing two candidate characters (set iteration order decides which is named).
- **Error messages**: use `re.escape` with `match=` for messages containing `(`, `[`, `+`.
- **URLs**: don't test http(s); it needs the network.

### arnio.io.read_csv_chunked

# Notes: `arnio.read_csv_chunked`

## 1. Purpose
- Reads a CSV in chunks and yields one `ArFrame` per chunk, with at most `chunksize` data rows each.
- **It is a generator function.** Calling it runs no code at all. All validation, file access and errors happen on the first `next()` or `list()`.

## 2. Inputs
- **`path`**
  - Accepted: `str`, `os.PathLike`, or a text file-like object (anything with a callable `.read()` that returns `str`).
  - A string with scheme `http`/`https` is fetched to a temporary UTF-8 file.
  - A path ending in `.gz` (case-insensitive) is gunzipped to a temporary **`.csv`** file. So `x.tsv.gz` does **not** get automatic tab detection.
  - File-like objects are copied to a temporary `.csv` file. For these, `encoding` is ignored and the data is treated as UTF-8.
- **Extension check** (only for `str`/PathLike input, checked on the original path, case-insensitive): it must end in `.csv`, `.txt`, `.tsv` or `.gz`.
- **`chunksize=10_000`**: must be an `int` and not a `bool`; must be `> 0`.
- **`delimiter=None`**: `None` becomes `"\t"` if the materialized path ends with `.tsv`, otherwise `","`.
  - It must then be a single character.
  - It must not be `\n`, `\r` or `"`.
  - It must not be a control character, except tab, which is allowed.
- **`has_header=True`, `trim_headers=True`**: must be real `bool` values. `1` is rejected.
- **`usecols`**: a non-empty Sequence of unique `str` values. A tuple is accepted. A bare `str` is rejected.
- **`nrows`**: an `int`, not a `bool`, `>= 0`. There is **no** short-circuit for `nrows=0`; the value is passed to C++, whose behaviour is unknown.
- **`skip_rows=0` / `skiprows=None`**: the two are aliases. Both skip *data rows after the header*.
  - If `skiprows` is given it replaces `skip_rows`.
  - They conflict only when `skip_rows != 0` and `skip_rows != skiprows`.
- **`encoding="utf-8"`**:
  - Must be a `str`.
  - `"utf-8"`, `"UTF8"` and `"utf_8"` all count as UTF-8.
  - Any other encoding is transcoded to a temporary file using `errors="strict"`.
  - There is **no `encoding_errors` parameter**; passing one gives Python's normal `TypeError`.
- **`decimal_separator="."`, `thousands_separator=None`**: see the Raises table.
- **`null_values`**: must be a `list` of `str` (a tuple is rejected).
- **`dtype`**: a `dict` mapping `str` to one of `"string"`, `"int64"`, `"float64"`, `"bool"`.
- **`mode`**: `"strict"` or `"permissive"`.
- **`on_bad_lines`**: `"error"`, `"warn"` or `"skip"`.

## 3. Raises (in check order; every one raises on first `next()`)
1. **Input-type and URL problems**, raised while the input is materialized:
   - `TypeError("read_csv_chunked expected a filesystem path or text file-like object")`, e.g. for `path=123`.
   - `ValueError("Cloud scheme 's3' is not yet supported by arnio. Install the optional extra when available: pip install \"arnio[s3]\"")` for `"s3://b/x.csv"`. The same pattern applies to `gs`, `gcs`, `az`, `abfs` and `abfss`.
   - `TypeError("read_csv file-like objects must return text, not bytes")` for `io.BytesIO(b"a\n1\n")`.
   - A missing `.gz` file gives gzip's own `FileNotFoundError`.
2. **Bad extension**: `ValueError(f"Unsupported file format: {path}. Only .csv, .txt, .tsv, and compressed .csv.gz are supported.")`.
   - Raised even if the file does not exist, e.g. `"data.dat"`.
3. **Encoding not a string**: `TypeError(f"encoding must be a string, got {type(encoding).__name__!r}")`, e.g. `"... got 'int'"`.
4. **Empty file** (0 bytes): `CsvReadError(f"CSV file is empty: {path!r}")`.
   - An empty `StringIO("")` also raises this, with the temporary file's path in the message.
   - A missing file is *not* reported here.
5. **`decimal_separator`**:
   - `TypeError("decimal_separator must be a string")`.
   - `ValueError("decimal_separator must be a single character")`.
   - `ValueError("decimal_separator must be a single non-alphanumeric character")` for `"a"`, `"1"`, `'"'`, `"\n"`.
   - `ValueError("Invalid decimal_separator: '+' and '-' are not allowed")`.
6. **`thousands_separator`**:
   - `TypeError("thousands_separator must be a string or None")`.
   - `ValueError` with the same wording as above, with the prefix `thousands_separator` instead of `decimal_separator`.
   - `ValueError("thousands_separator must differ from decimal_separator")` for `thousands_separator="."`.
7. **`delimiter`**:
   - `TypeError("delimiter must be a string")`.
   - `ValueError("delimiter must be a single character; delimiter must be exactly one character")`.
   - `"delimiter must not be a newline character"`.
   - `"delimiter must not be the CSV quote character"`.
   - `"delimiter must not be a control character"`, e.g. `"\x01"`.
8. **`mode`**: `TypeError("mode must be a string")`; `ValueError("mode must be either 'strict' or 'permissive'")`.
9. **`chunksize`**: `TypeError("chunksize must be an integer")` (also for `True` or `2.0`); `ValueError("chunksize must be a positive integer")` for `0` or `-1`.
10. **`skiprows`**:
    - `TypeError("skiprows must be an integer")`.
    - `ValueError("skiprows must be non-negative")`.
    - `ValueError("Conflicting values: skiprows=3 and skip_rows=2. Pass only one of them.")` for `skip_rows=2, skiprows=3`.
11. **`skip_rows`**: `TypeError("skip_rows must be an integer")`; `ValueError("skip_rows must be non-negative")`.
12. **`on_bad_lines`**: `TypeError("on_bad_lines must be a string")`; `ValueError("on_bad_lines must be either 'error', 'warn', 'skip'")`.
13. **`has_header` / `trim_headers`**: `TypeError("has_header must be True or False, got int: 1")`. The template is `f"{name} must be True or False, got {type(value).__name__}: {value!r}"`.
14. **`null_values`**:
    - `TypeError("null_values must be a list of strings, not a bare string")`.
    - `TypeError("null_values must be a list of strings")` (for a tuple).
    - `TypeError("null_values must contain only strings")`.
15. **`dtype`**:
    - `TypeError("dtype must be a dictionary mapping column names to dtype strings")`.
    - `TypeError("dtype column names must be strings")`.
    - `TypeError("dtype values must be strings")`.
    - `ValueError("Unsupported dtype 'int32'. Expected one of: ['bool', 'float64', 'int64', 'string']")`.
16. **`usecols`**:
    - `TypeError("usecols must be a sequence of column names, not a string")`.
    - `TypeError("usecols must be a sequence of strings")`.
    - `ValueError("usecols must not be empty")`.
    - `TypeError("usecols must contain only strings")`.
    - `ValueError("usecols must not contain duplicate column names")`.
17. **`nrows`**: `TypeError("nrows must be an integer")`; `ValueError("nrows must be non-negative")`.
18. **Transcoding** (non-UTF-8 encodings only):
    - `ValueError(f"Unknown encoding: {encoding}")`, with **no** quotes around the name, e.g. `encoding="nope"`.
    - `CsvReadError("Could not decode '<path>' using encoding 'ascii': invalid byte(s) at position N (byte value: 0x..). Try a different encoding or use encoding_errors='replace'.")`.
    - A missing file with a non-UTF-8 encoding gives `CsvReadError("Could not access CSV file '<path>': <strerror>")`.
19. **Native parse errors**: any `RuntimeError` from C++ becomes `CsvReadError(str(e))`, with the message unchanged.
    - Unlike `read_csv`, there is **no** "Malformed CSV: ..." enrichment.
    - With the default UTF-8 encoding, a missing file is handled by C++ (exact exception and message not visible).

## 4. Returns and guarantees
- Yields `ArFrame` objects.
  - Each chunk has at most `chunksize` rows.
  - The total row count respects `nrows`.
  - Dropped bad rows count toward `nrows` (per the docstring).
- Column types are inferred from the first chunk and applied to all later chunks (per the docstring; done in C++).
- Concatenating the chunks should equal the whole file. For example, a 5-row file with `chunksize=2` gives row counts `[2, 2, 1]`.
- **Not done here (unlike `read_csv`):**
  - No NUL-byte rejection.
  - No delimiter-mismatch warning.

## 5. Side effects
- Temporary files are created for URL, gzip, file-like and non-UTF-8 inputs.
  - They are deleted when the generator finishes, or on validation error.
- `on_bad_lines="warn"` emits a `UserWarning` for each chunk that has bad rows:
  - Format: `"{n} malformed CSV row(s):\n  CSV row R has A fields; expected E"`.
  - At most 10 rows are listed, followed by `"  (+k more)"` if there are more.

## 6. Branches and edge cases
- If a chunk has 0 rows **and** `bad_rows` is non-empty, it is skipped only if a non-empty chunk was already yielded. Otherwise it is yielded.
- An explicit `delimiter=","` on a `.tsv` path is honoured.
- A `.tsv` file with `delimiter=None` uses tab.
- `skip_rows=2, skiprows=2` is accepted.
- `skip_rows=0, skiprows=3` gives an effective skip of 3.
- `pathlib.Path` input works, and the extension check applies to it.
- A `StringIO` input skips the extension check entirely.

## 7. Rules from comments and docstrings
- Explicit `delimiter` always wins over `.tsv` auto-detection.
- `skip_rows` and `skiprows` both skip data rows after the header.
- Validation runs on the decompressed temporary file, not on the `.gz` bytes.

## 8. Testing hints
- Build inputs with `p = tmp_path / "d.csv"; p.write_text("a,b\n1,x\n2,y\n3,z\n")`.
- Collect results with `chunks = list(ar.read_csv_chunked(p, chunksize=2))`.
- Check `[c.shape for c in chunks] == [(2, 2), (1, 2)]` and `ar.to_pandas(c)["a"].tolist()`.
- **Pitfall:** errors only appear when the generator is consumed. Write `with pytest.raises(ValueError, match="..."): next(ar.read_csv_chunked(p, chunksize=0))`. Calling the function without iterating never raises.
- `match=` is a regex. Use `re.escape` for messages containing `(`, `[`, `.` or `+`.
  - Avoid matching full `repr(path)` strings: on Windows the backslashes are doubled.
- To check ordering, combine two bad arguments: `decimal_separator="ab", chunksize=0` raises the decimal error. A file `x.dat` with `chunksize=0` raises the format error.
- **Avoid asserting:**
  - Results for header-only files.
  - `nrows=0`.
  - Results with `has_header=False` (the column names are decided in C++).
  - Exact native error texts.
- To test TSV: write tab-separated data to `d.tsv` and assert 2 columns.
- To test gzip: write the file with `gzip.open(tmp_path/"d.csv.gz", "wt")`.
