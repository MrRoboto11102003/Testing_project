# arnio.schema.Schema.bootstrap_from_report
    @classmethod
    def bootstrap_from_report(cls, report: Any) -> Schema:
        """Create a Schema from a DataQualityReport.

        Args:
            report: A DataQualityReport produced by arnio.profile().

        Returns:
            Schema: A new Schema inferred from the report's column profiles.

        Raises:
            TypeError: If the input is not a DataQualityReport.
            ValueError: If the report has no columns, or any column profile
                is missing a dtype.

        Example:
            >>> report = ar.profile(frame)
            >>> schema = ar.Schema.bootstrap_from_report(report)
            >>> result = schema.validate(frame)
        """
        from .quality import DataQualityReport

        if not isinstance(report, DataQualityReport):
            raise TypeError(f"Expected DataQualityReport, got {type(report).__name__}")
        if not report.columns:
            raise ValueError(
                "Cannot bootstrap schema from an empty report (no columns)."
            )

        fields = {}
        for col_name, profile in report.columns.items():
            dtype_val = getattr(profile, "dtype", None)
            null_count = getattr(profile, "null_count", 0)

            if not dtype_val:
                raise ValueError(
                    f"Column profile for {col_name!r} is missing 'dtype' key."
                )

            arnio_dtype = _DTYPE_MAP.get(str(dtype_val).lower(), "string")
            fields[col_name] = Field(dtype=arnio_dtype, nullable=null_count > 0)

        return cls(fields=fields)

# arnio.schema.Schema.from_json
    @classmethod
    def from_json(cls, value: str) -> Schema:
        """Deserialize a schema from a JSON string produced by ``to_json()``."""
        try:
            payload = json.loads(value)
        except json.JSONDecodeError as exc:
            raise ValueError(f"Invalid schema JSON: {exc.msg}") from exc

        if not isinstance(payload, dict):
            raise TypeError(
                "Schema JSON must decode to an object with 'fields', 'strict', and optional 'unique'."
            )

        unknown = set(payload.keys()) - _ALLOWED_SCHEMA_KEYS
        if unknown:
            raise ValueError(
                f"Schema JSON contains unknown key(s): {sorted(unknown)}. "
                f"Allowed keys: {sorted(_ALLOWED_SCHEMA_KEYS)}"
            )

        fields_payload = payload.get("fields")
        if not isinstance(fields_payload, dict):
            raise TypeError(
                "Schema JSON 'fields' must be an object mapping names to field definitions."
            )

        fields = {
            name: _field_from_json_dict(name, field_payload)
            for name, field_payload in fields_payload.items()
        }

        strict = payload.get("strict", False)
        if not isinstance(strict, bool):
            raise TypeError("Schema JSON 'strict' must be a boolean.")

        unique = payload.get("unique")
        if unique is not None and not isinstance(unique, list):
            raise TypeError("Schema JSON 'unique' must be a list of strings or null.")

        # rules_omitted marker is accepted and silently ignored — rules
        # cannot be round-tripped through JSON and must be re-attached
        # manually by the caller if needed.
        return cls(fields=fields, strict=strict, unique=unique)

# arnio.schema.Schema.to_json
    def to_json(self) -> str:
        """Serialize the schema to a stable JSON string.

        Cross-field ``rules`` cannot be serialized. When rules are present
        they are silently omitted and a ``UserWarning`` is emitted. The
        returned payload includes ``"rules_omitted": true`` so downstream
        consumers can detect the gap. Call ``from_json()`` to restore the
        field-only schema; rules must be re-attached manually afterward.
        """
        rules_omitted = bool(self.rules)
        if rules_omitted:
            warnings.warn(
                "Schema.to_json(): cross-field rules cannot be serialized "
                "and have been omitted. The JSON payload includes "
                '"rules_omitted": true. Restore rules manually after '
                "from_json().",
                UserWarning,
                stacklevel=2,
            )

        payload = {
            "fields": {
                name: _field_to_json_dict(field_def)
                for name, field_def in sorted(self.fields.items())
            },
            "strict": self.strict,
            "unique": list(self.unique) if self.unique is not None else None,
        }
        if rules_omitted:
            payload["rules_omitted"] = True
        return json.dumps(payload, sort_keys=True)

### arnio.schema.Schema.bootstrap_from_report

# Notes: `arnio.schema.Schema.bootstrap_from_report(report)`

## 1. Purpose
- Classmethod that builds a `Schema` from a `DataQualityReport` (normally from `ar.profile(frame)`).
- It creates one `Field(dtype=..., nullable=...)` per profiled column. Nothing else is inferred.

## 2. Inputs
- `report`: must be an instance of `arnio.quality.DataQualityReport`. The class is imported lazily inside the method.
- Only two parts of the report are read:
  - `report.columns`: a dict mapping column name to profile. It is iterated with `.items()`, in insertion order.
  - For each profile:
    - `getattr(profile, "dtype", None)`
    - `getattr(profile, "null_count", 0)`. A missing attribute counts as 0 nulls.
- Every other report or profile attribute is ignored: `null_ratio`, `empty_string_count`, `min`/`max`, `semantic_type`, `unique_ratio`, and so on.
- dtype mapping uses `_DTYPE_MAP.get(str(dtype).lower(), "string")`. Matching is case-insensitive.

| Profile dtype (lowercased) | Field dtype |
|---|---|
| `int`, `int64`, `int32` | `int64` |
| `float`, `float64`, `float32` | `float64` |
| `bool` | `bool` |
| `object`, `string`, `str` | `string` |
| `null` | `"null"` → **Field raises** (see Raises) |
| anything else (`datetime`, `date`, `int8`, `category`, …) | `string` (fallback) |

- Note: `"datetime"` maps to `"string"`, not to `"datetime"`.
- `"INT64"` maps to `int64`, and `"Float32"` maps to `float64`.
- nullable rule: `nullable = null_count > 0`. So 0 nulls gives `False`, and 1 or more gives `True`.

## 3. Raises (checks happen in this order)
1. **Wrong report type**
   - `TypeError(f"Expected DataQualityReport, got {type(report).__name__}")`
   - Example: `ar.Schema.bootstrap_from_report({})` gives `"Expected DataQualityReport, got dict"`.
   - Example: `None` gives `"... got NoneType"`.
   - A duck-typed object with `.columns` is still rejected.
2. **Empty report**
   - Condition: `report.columns` is falsy (`{}`).
   - `ValueError("Cannot bootstrap schema from an empty report (no columns).")`
3. **Missing dtype** (checked per column, in dict order, during the loop)
   - Condition: a profile's `dtype` is falsy (`None`, `""`) or the attribute is missing.
   - `ValueError(f"Column profile for {col_name!r} is missing 'dtype' key.")`
   - Example: column `"a"` with `dtype=""` gives `"Column profile for 'a' is missing 'dtype' key."`
   - No partial schema is returned.
4. **Errors raised by `Field.__post_init__`**
   - dtype `"null"` is mapped to `"null"`, which is not a valid Field dtype:
     - `ValueError("dtype must be one of int64, float64, string, bool, datetime, or None, got 'null'")`
   - `null_count` is a numpy integer (e.g. `np.int64(2)`): `np.int64(2) > 0` gives a `np.bool_`, which is not a Python `bool`.
     - `TypeError("nullable must be a bool")`
     - Use plain Python ints when you build profiles by hand.
   - `null_count=None`: `None > 0` raises a Python `TypeError` (comparison not supported).
5. **Errors raised by `Schema.__post_init__`**
   - A non-str column key (e.g. `columns={1: prof}`):
     - `TypeError("Schema field names must be strings, got 'int': 1")`

## 4. Returns and guarantees
- Returns a new `cls(fields=fields)` object. `type(result) is ar.Schema` (or the subclass it was called on).
- Schema-level settings: `strict=False`, `unique=None`, `rules=None`.
- `list(result.fields)` equals `list(report.columns)`: same names, same order.
- Each Field has only `dtype` and `nullable` set. Every other attribute keeps its default:
  - `min/max/pattern/semantic/allowed/min_length/max_length/format/required_if` are `None`
  - `unique=False`, `case_sensitive=True`, `severity="error"`
- Field equality works: `result.fields["a"] == ar.Schema({...}).fields["a"]` for `Field(dtype="int64", nullable=False)`. Field is a frozen dataclass.

## 5. Side effects
- None. The report is not mutated and no global state changes.

## 6. Branches and edge cases
- Non-report input gives `TypeError` (check 1).
- `columns == {}` gives `ValueError` (check 2).
- A profile with `dtype=None` or `""` gives `ValueError` (check 3). The first failing column in dict order is the one named in the message.
- Unknown dtype string such as `"datetime64[ns]"` or `"weird"` gives a Field with dtype `"string"`.
- `null_count=0` gives `nullable=False`. `null_count=3` gives `nullable=True`.
- A profile object without a `null_count` attribute (e.g. `types.SimpleNamespace(dtype="int64")`) gives `nullable=False`.
- The input type is checked strictly, so `columns` itself must sit inside a real DataQualityReport.

## 7. Rules from the docstring
- Raises `TypeError` for non-report input.
- Raises `ValueError` for an empty report or a missing dtype.
- Intended flow: `ar.profile(frame)` → `bootstrap_from_report` → `schema.validate(frame)`.

## 8. Testing hints
- Build reports by hand for exact control:
  ```python
  from arnio.quality import ColumnProfile, DataQualityReport
  def prof(name, dtype, nulls=0):
      return ColumnProfile(name=name, dtype=dtype, semantic_type="x", row_count=3,
                           null_count=nulls, null_ratio=nulls/3, unique_count=3, unique_ratio=1.0)
  rep = DataQualityReport(row_count=3, column_count=2, memory_usage=0, duplicate_rows=0,
                          duplicate_ratio=0.0, columns={"a": prof("a","int32"), "b": prof("b","WEIRD",1)})
  s = ar.Schema.bootstrap_from_report(rep)
  # s.fields["a"] == Field(dtype="int64", nullable=False); s.fields["b"] == Field(dtype="string", nullable=True)
  ```
  - I assume both classes are plain dataclasses with no extra validation, based on the field lists in the index. I cannot see `quality.py`, so this is unverified.
  - `Field` can be imported with `from arnio.schema import Field`.
- Test every row of the mapping table. In particular, check that `"float32"` maps to `float64`, `"str"` and `"object"` map to `string`, and `"null"` raises.
- Use `pytest.raises(..., match=re.escape(msg))` with the exact messages listed in Raises.
- Real-profile path: `ar.profile(ar.from_pandas(df))`.
  - The exact dtype strings come from the C++ core / `quality.py`, which I cannot see.
  - Assert relative to the report rather than hard-coding: `s.fields[c].dtype == expected_map(rep.columns[c].dtype)` and `s.fields[c].nullable == (rep.columns[c].null_count > 0)`.
  - Do not assume what dtype an all-null column gets. If it is `"null"`, the call raises.
- Pitfalls:
  - Do not assume that `schema.validate(frame).passed` is True after bootstrapping. Validation treats empty or whitespace strings as null, but `null_count` may not count them, so a column could get `nullable=False` and still fail validation.
  - Dtype mismatches caused by the `"string"` fallback can also make validation fail.
  - Do not expect `unique`, `min`, or `max` to be inferred.

### arnio.schema.Schema.from_json

# Notes: `arnio.schema.Schema.from_json(value)` (classmethod)

## 1. Purpose
- Turns a JSON string, normally one produced by `Schema.to_json()`, back into a `Schema` that contains only fields.
- Cross-field `rules` are never restored. The result always has `rules=None`.

## 2. Inputs
- `value` is a JSON text (`str`, `bytes` or `bytearray`) and is passed straight to `json.loads`.
- **Top-level object.** Allowed keys are exactly `fields`, `strict`, `unique` and `rules_omitted`.
  - `fields` is required and must be a JSON object of `{column_name: field_object}`. `{}` is allowed.
  - `strict` defaults to `False` when the key is missing. An explicit `null` does **not** fall back to the default and raises.
  - `unique` may be missing, `null` or a list.
  - `rules_omitted` is accepted with any value and ignored.
- **Each field object.** Allowed keys:
  - `dtype`, `nullable`, `min`, `max`, `pattern`, `semantic`, `allowed`, `case_sensitive`, `unique`
  - `min_length`, `max_length`, `format`
  - `datetime_min`, `datetime_max`, `date_min`, `date_max` (no leading underscore)
  - `required_if`, `severity`
- **Field defaults and conversions.**
  - A missing key gives the default: `None`, or `nullable=True`, `case_sensitive=True`, `unique=False`, `severity="error"`.
  - An explicit JSON `null` for `nullable`, `unique`, `case_sensitive` or `severity` passes `None` and raises in `Field`.
  - `allowed` list → `set`.
  - `required_if` list → `tuple`.
  - `datetime_min`/`datetime_max` → `pd.Timestamp` via `pd.to_datetime`.
  - `date_min`/`date_max` → `datetime.date`.

## 3. Raises (in check order)
1. `value` is not str/bytes, e.g. `from_json(None)` → stdlib `TypeError` ("the JSON object must be str, bytes or bytearray, not NoneType"). This is not wrapped.
2. Bad JSON → `ValueError(f"Invalid schema JSON: {exc.msg}")`. `exc.msg` has no position info.
   - `from_json("not json")` → `"Invalid schema JSON: Expecting value"`.
3. Not a JSON object, e.g. `"[]"`, `"null"`, `"1"` → `TypeError("Schema JSON must decode to an object with 'fields', 'strict', and optional 'unique'.")`.
4. Unknown top-level key → `ValueError(f"Schema JSON contains unknown key(s): {sorted(unknown)}. Allowed keys: ['fields', 'rules_omitted', 'strict', 'unique']")`.
   - `'{"fields":{},"x":1}'` → `"... unknown key(s): ['x']. ..."`.
5. `fields` missing or not an object, e.g. `'{}'` or `'{"fields":[]}'` → `TypeError("Schema JSON 'fields' must be an object mapping names to field definitions.")`.
6. Errors from each field, in JSON key order. These come **before** the `strict`/`unique` checks.
   - Field value not an object, `'{"fields":{"a":1}}'` → `TypeError("Schema JSON field for column 'a' must be an object, got int.")`.
   - Unknown field key → `ValueError(f"Schema JSON field 'a' contains unknown key(s): ['_date_min']. Allowed keys: [...sorted 18 keys...]")`.
   - `"allowed": "x"` → `TypeError("Schema JSON field 'a' 'allowed' must be a list or null.")`.
   - `"required_if": ["b"]` or a non-list → `TypeError("Schema JSON field 'a' 'required_if' must be a 2-item list or null.")`.
   - `"datetime_min": "nope"` → `ValueError("DateTime datetime_min must be a parseable datetime scalar")`. The same template applies to `datetime_max`.
   - `"date_min": 5` → `TypeError("Date date_min must be a date string, datetime.date, or pd.Timestamp, got int")`.
   - `"date_min": "nope"` → `ValueError("Date date_min must be a parseable date (YYYY-MM-DD), got 'nope'")`.
   - Bound parsing happens before `Field.__post_init__`, which then checks:
     - `"dtype":"foo"` → `ValueError("dtype must be one of int64, float64, string, bool, datetime, or None, got 'foo'")`.
     - `"dtype":1` → `TypeError("dtype must be a string or None, got int")`.
     - `"nullable":null` → `TypeError("nullable must be a bool")`.
     - `"unique":null` → `TypeError("unique must be a bool")`.
     - `"case_sensitive":null` → `TypeError("case_sensitive must be a bool")`.
     - `"required_if":[1,"x"]` → `TypeError("required_if column name must be a string")`.
     - `"required_if":["b",[1]]` → `TypeError("required_if expected value must be a scalar")`.
     - `"dtype":"int64","min":5,"max":1` → `ValueError("min must be less than or equal to max")`.
     - `"dtype":"int64","min":"x"` → `TypeError("min must be numeric or None")`.
     - The min/max checks run **only** for int64/float64. For example, `"dtype":"string","min":"x"` is accepted.
     - `"pattern":"("` → `ValueError("pattern is not a valid regular expression: ...")`.
     - `"pattern":"(a+)+"` → `ValueError("Unsafe regex pattern rejected: nested quantifiers can cause catastrophic backtracking during schema validation")`.
     - `"min_length":-1` → `ValueError("min_length must be >= 0, got -1")`.
     - `"min_length":5,"max_length":2` → `ValueError("min_length (5) must be <= max_length (2)")`.
     - `"severity":"info"` → `ValueError("severity must be 'error' or 'warning'")`.
     - `"severity":null` → `TypeError("severity must be a string")`.
7. `strict` not a bool, e.g. `"strict":1`, `"strict":"yes"` or `"strict":null` → `TypeError("Schema JSON 'strict' must be a boolean.")`.
8. `unique` not a list or null, e.g. `"unique":"a"` → `TypeError("Schema JSON 'unique' must be a list of strings or null.")`.
9. Then `Schema.__post_init__` runs:
   - `"unique":[1]` → `TypeError("Schema 'unique' members must be strings, got int for element 1.")`.
   - `"unique":["a","a"]` → `ValueError("Schema 'unique' must not contain duplicate column names")`.

## 4. Returns and guarantees
- Returns a new `Schema(fields=..., strict=..., unique=...)` with `rules=None`.
- `fields` is a dict that keeps the JSON key order. `to_json` writes keys sorted.
- `unique` stays a **list** (not a tuple). `"unique": []` is accepted and gives `[]`.
- Field values:
  - `allowed` is a `set`.
  - `required_if` is a `tuple`.
  - `_datetime_min` is a `pd.Timestamp`.
  - `_date_min` is a `datetime.date`.
  - `min`/`max` keep their JSON int/float type.

## 5. Side effects
- No mutation, no registry changes, no warnings, no I/O.
- `to_json` warns when rules are present; `from_json` never warns.

## 6. Branches and edge cases
- `'{"fields":{}}'` → `Schema` with `fields == {}`, `strict is False`, `unique is None`, `rules is None`.
- `'{"fields":{"a":{}}}'` → `fields["a"] == Field()`, which has all defaults (`dtype None`, `nullable True`, `severity "error"`).
- `'{"fields":{},"rules_omitted":true}'` → OK and `rules is None`. `"rules_omitted": "anything"` is also accepted.
- `'{"fields":{"a":{"allowed":["x","y"],"required_if":["b",1]}}}'` → `allowed == {"x","y"}` and `required_if == ("b", 1)`.
- `'{"fields":{"t":{"dtype":"datetime","datetime_min":"2024-01-01"}}}'` → `_datetime_min == pd.Timestamp("2024-01-01")`.
- `'{"fields":{"d":{"date_max":"2024-12-31"}}}'` → `_date_max == datetime.date(2024,12,31)`.
- If both a field error and a bad `strict` are present, the field error is raised.

## 7. Rules stated in comments or docstrings
- The marker `rules_omitted` "is accepted and silently ignored — rules cannot be round-tripped through JSON and must be re-attached manually".

## 8. Testing hints
- Round trip: for a schema such as the one below, `ar.Schema.from_json(s.to_json()) == s` holds. Field and Schema are dataclasses with `__eq__`.
  ```python
  s = ar.Schema({"id": ar.Int64(nullable=False, min=1, max=9),
                 "c": ar.String(allowed=["x","y"]),
                 "d": ar.Date(min="2024-01-01")},
                strict=True, unique=["id"])
  ```
  - Equality breaks if the original used `unique=("id",)`, because `from_json` returns a list.
- Build field objects directly when needed: `from arnio.schema import Field`.
- The restored schema should behave the same when used. Use `ar.validate(ar.from_pandas(df), restored)` to check behaviour.
- Use `pytest.raises(..., match=re.escape(msg))`. The messages contain `[ ] ( ) . '`.
- Don't hash `Schema` or `Field` objects that contain a set or dict; they are unhashable. Compare them with `==` only.
- Don't assert the full `JSONDecodeError` text beyond `exc.msg`. For example, `"Invalid schema JSON: Expecting value"` is fine to assert exactly.

### arnio.schema.Schema.to_json

# Notes: `arnio.schema.Schema.to_json`

## 1. Purpose
- Serializes a `Schema` (its fields, `strict` and `unique`) to a deterministic JSON string.
- Cross-field `rules` are not serialized. If any are present, a `UserWarning` is emitted and a `"rules_omitted": true` marker is added.

## 2. Inputs
- `self` only; there are no parameters.
- The `Schema` has already been validated in `__post_init__`:
  - field names are `str`;
  - values are `Field` instances;
  - `strict` is a bool;
  - `unique` is a list/tuple of distinct strings or `None`;
  - `rules` is a list/tuple of callables or `None`.

## 3. Raises
- `TypeError` from `json.dumps` when a field value is not JSON-serializable. The message is the stdlib one: `"Object of type <T> is not JSON serializable"`. Examples:
  - `ar.Schema({"d": ar.String(allowed=[datetime.date(2024,1,1)])}).to_json()` → `Object of type date is not JSON serializable`
  - `ar.Schema({"a": ar.Int64(required_if=("b", np.int64(1)))}).to_json()` → `Object of type int64 is not JSON serializable` (`np.int64` is accepted by `Field` because it passes `pd.api.types.is_scalar`)
- Order of events: the rules warning is emitted first, then the payload is built, then `json.dumps` runs. So a schema with rules and a bad value warns and then raises.

## 4. Returns and guarantees
- Returns a `str` produced by `json.dumps(payload, sort_keys=True)`:
  - no indent;
  - default separators `", "` and `": "`;
  - `allow_nan=True`, so a `NaN` float is written as `NaN`.
- Top-level keys, sorted: `"fields"`, optionally `"rules_omitted"` (only when rules are present, value `true`), `"strict"`, `"unique"`.
- `"unique"` is `list(self.unique)`, keeping the original order. It is not sorted (unlike `diff_schema`). If unset it is `null`. A tuple becomes a JSON array.
- `"fields"` maps each column name to a dict with exactly 18 keys, in this sorted order:
  `allowed, case_sensitive, date_max, date_min, datetime_max, datetime_min, dtype, format, max, max_length, min, min_length, nullable, pattern, required_if, semantic, severity, unique`.
- How each field value is written:
  - **`allowed`**: a set becomes a sorted list. If its types are mixed and not comparable, it is sorted by `(type(x).__name__, repr(x))`, so `{1, "a"}` → `[1, "a"]`. `None` → `null`.
  - **`required_if`**: the tuple becomes a list, e.g. `("x", 1)` → `["x", 1]`.
  - **`datetime_min` / `datetime_max`**: `Timestamp.isoformat()`. `ar.DateTime(min="2024-01-01")` → `"2024-01-01T00:00:00"`.
  - **`date_min` / `date_max`**: `date.isoformat()`. `ar.Date(min="2024-01-01")` → `"2024-01-01"`.
  - **`min` / `max`**: raw values. `ar.Int64(min=1, max=5)` → `1`, `5`. `ar.Float64(min=1)` stays `1`, not `1.0`.
- `semantic` for each field factory:

  | Factory | `semantic` |
  |---|---|
  | `Email()` | `"email"` |
  | `Email(validation="strict")` | `"email:strict"` |
  | `URL()` | `"url"` |
  | `URL(allowed_schemes=["https","http"])` | `"url:http|https"` (sorted, joined by `|`) |
  | `PhoneNumber()` | `"phone"` |
  | `CountryCode()` | `"country_code"` |
  | `UUID()` | `"uuid"` |
  | `IPv4()` | `"ipv4"` |
  | `MACAddress()` | `"mac_address"` |
  | `Date()` | `"date"`; `dtype` is `"string"` |
  | `DateTime()` | `dtype` is `"datetime"`; `semantic` is `null` |
  | `Custom("n")` | `"custom:n"`; `dtype` is `null` |

- Exact output for `ar.Schema({"a": ar.Int64()}).to_json()`:
```
{"fields": {"a": {"allowed": null, "case_sensitive": true, "date_max": null, "date_min": null, "datetime_max": null, "datetime_min": null, "dtype": "int64", "format": null, "max": null, "max_length": null, "min": null, "min_length": null, "nullable": true, "pattern": null, "required_if": null, "semantic": null, "severity": "error", "unique": false}}, "strict": false, "unique": null}
```
- `ar.Schema({}).to_json()` → `'{"fields": {}, "strict": false, "unique": null}'`

## 5. Side effects
- Emits a `UserWarning` only when `bool(self.rules)` is true. The warning uses `stacklevel=2`. Exact message:
  `Schema.to_json(): cross-field rules cannot be serialized and have been omitted. The JSON payload includes "rules_omitted": true. Restore rules manually after from_json().`
- The schema is not mutated (it is a frozen dataclass).

## 6. Branches and edge cases
- `rules=None` or `rules=[]`: no warning and no `rules_omitted` key.
- `rules=[f]` (list or tuple): warns, and the key appears between `fields` and `strict`: `'{"fields": {...}, "rules_omitted": true, "strict": ...}'`.
- `strict=True` → `"strict": true`.
- `unique=("b","a")` → `"unique": ["b", "a"]` (order preserved).
- Multiple fields are emitted in name order, whatever order they were inserted in. For example, `{"b":..., "a":...}` gives `a` first.
- `severity="warning"` → `"severity": "warning"`.
- `ar.String(min_length=2, max_length=5, pattern=r"^x$", case_sensitive=False)` → those values appear as-is (`"pattern": "^x$"`, `"case_sensitive": false`).

## 7. Rules from the docstring
- The output is stable: field names are sorted and `sort_keys=True` is used.
- Rules are silently omitted apart from the warning.
- `from_json()` restores only the field part of the schema. It accepts and ignores the `rules_omitted` key.

## 8. Testing hints
- Prefer `json.loads(s)` and compare dicts. Exact-string checks are also deterministic (see §4), and checking the exact string is a good way to catch separator or ordering regressions.
- Check the key order with `list(json.loads(s)["fields"]["a"].keys())` (dicts preserve the parsed order) or by checking the raw string.
- Warning: `with pytest.warns(UserWarning, match=r"cross-field rules cannot be serialized"):`
- No warning: use
  ```python
  with warnings.catch_warnings():
      warnings.simplefilter("error")
      schema.to_json()
  ```
- Round trip: `ar.Schema.from_json(s.to_json()).fields == s.fields` holds for factory-built fields, including `allowed` sets, `required_if` tuples, and date/datetime bounds.
- Round-trip pitfalls:
  - `Schema` equality fails if the original `unique` was a tuple (it comes back as a list).
  - `Schema` equality fails if the original had rules (they come back as `None`).
- `Custom` needs `ar.register_validator(...)` first. The registry is global and duplicate names raise, so use a unique name or pass `overwrite=True`.
- Do not expect `NaN`/`Infinity` from `Int64`/`Float64`, which reject non-finite bounds. Only a raw `ar.Field(dtype="float64", min=float("nan"))` produces `"min": NaN`.

### arnio.schema.Schema.validate

# Notes: `arnio.schema.Schema.validate(self, frame, max_errors=None)`

## 1. Purpose
- Validates an `ArFrame` against this schema and returns a `ValidationResult`.
- It is a one-line wrapper: `return validate(frame, self, max_errors=max_errors)`. Every behaviour below is the module-level `ar.validate`.

## 2. Inputs
- `frame`: must be an `ArFrame` instance. A `pd.DataFrame` or a chunk iterator is rejected.
- `max_errors`: `None` (no limit) or an `int >= 1`. A `bool` is rejected.
- Internally:
  - `df = to_pandas(frame)` and `dtypes = frame.dtypes`. The dtype strings come from the C++ core; they are expected to be `"int64"`, `"float64"`, `"string"`, `"bool"`, but I cannot see that code.
  - Fields are checked in the dict insertion order of `self.fields`.

## 3. Raises (checked in this order)
1. Non-ArFrame `frame` raises `TypeError`. Message: `"validate() expects a single ArFrame. Chunked validation is not currently supported; validate each chunk explicitly or materialize the data before validation."` Example: `schema.validate(pd.DataFrame())`.
2. `max_errors` is a bool or not an int: `TypeError("max_errors must be an int or None")`. Examples: `max_errors=True`, `max_errors=1.5`.
3. `max_errors <= 0`: `ValueError("max_errors must be >= 1")`. Examples: `0`, `-1`.
4. A rule in `rules` returns a non-list: `TypeError(f"Rule {rule_name!r} must return a list of ValidationIssue, got {type(result).__name__!r}")`.
   - `rule_name` is `fn.__name__`.
   - Example: `def r(df): return None` gives `"Rule 'r' must return a list of ValidationIssue, got 'NoneType'"`.
5. A rule returns a list containing a non-issue: `TypeError(f"Rule {rule_name!r} returned a non-ValidationIssue item: {type(item).__name__!r}")`.
   - Only `KeyError` from a rule is caught. Any other exception propagates.
6. A Custom validator raises: `ArnioError(f"Custom validator {name!r} failed for column {col!r} with value {v!r}: {exc}")`.
7. A Custom validator returns something other than True/False/None/pd.NA: `TypeError`.

## 4. Returns and guarantees
`ValidationResult` fields:
- `row_count = len(df)`.
- `issues`: a list in check order.
- `issue_count == len(issues)`.
- `bad_rows`: sorted, de-duplicated `row_index` values (`None` excluded).

Indexing and status:
- `row_index` is **1-based**: pandas index + 1. Row 0 gives `row_index == 1`.
- `issue.value` is the raw offending cell value. Column-level issues have `value=None` and `row_index=None`.
- `result.passed` is True iff no issue has `severity == "error"`. Warning-only issues still pass.

Issue order:
1. Per field, in field order. A missing column gives `rule="required_column"`, `message=f"Missing required column: {name}"`, `severity=field.severity`.
2. Strict check: `rule="unexpected_column"`, `message=f"Unexpected column: {name}"`, severity `"error"`, in frame column order.
3. Composite `unique`.
4. `rules`.

Within one column, `_validate_column` checks in this order:

| Order | Rule | Fires when / message |
|---|---|---|
| 1 | `dtype` | Fires if `field.dtype` is set and differs from the actual dtype. Skipped when the field is `"datetime"` and the column is `"string"`. Message: `f"Column {name!r} has dtype {actual!r}; expected {exp!r}"`. If the column is string, the field is int64/float64, and the values look convertible, it gains the suffix `". Values appear safely convertible to 'int64'"`. "Convertible" means all non-null values match `^-?\d+$` for int. For id-like names (`id`, `*_id`, `uuid`, `zip*`), leading-zero values like `"007"` block the suffix. |
| 2 | `nullable` | Only if `nullable=False`. Message: `"Column 'x' contains null or empty values"`. For string/object columns, `""` and whitespace-only values also count as null. |
| 3 | `required_if` | If the condition column is absent: `rule="missing_column"`, `message=f"Column {c!r} not found"`, severity `"error"`. Otherwise one issue per row where `df[c] == v` and the cell is null. Message: `f"Column {name!r} is required when {c!r} == {v!r}"`. |
| 4 | `unique` | Duplicates among non-null values, with `keep=False`, so **every** occurrence is reported. Message: `"Column 'x' contains duplicate values"`. |
| 5 | `allowed` | Message: `"Column 'x' contains values outside the allowed set"`. Case folding applies only when `dtype=="string"` and `case_sensitive=False`. |
| 6 | datetime: `format`, `min`, `max` | Bound messages: `"Column 'x' has values below <Timestamp>"` / `"above"`. |
| 6 (else) | numeric: `numeric`, `min`, `max` | Applies when min or max is set. Messages: `"Column 'x' contains non-numeric values"`, `f"Column {name!r} has values below {min}"`, `f"... above {max}"`. Bounds are strict: a value equal to a bound passes. |
| 7 | `pattern` | Uses `fullmatch`. Message: `"Column 'x' has values that do not match the pattern"`. |
| 8 | semantic | The rule name equals the semantic (e.g. `"email"`, `"url"`, `"ipv4"`, `"date"`). Message: `f"Column {name!r} contains invalid {semantic} values"`. Date min/max use rule `min`/`max` with message `"has date values before 2020-01-01"` / `"after"`. Custom validators use `rule="custom"`. |
| 9 | `min_length` / `max_length` | Messages: `"has values shorter than N"` / `"longer than N"`. |

- Null and empty cells are skipped by every check after `nullable`.
- Severity: column checks use `field.severity`. Strict, composite-unique and rule-`KeyError` issues use `"error"`.

## 5. Side effects
- None. The frame and schema are not mutated. Rules receive the `df` produced by `to_pandas`.

## 6. Branches and edge cases
- **`max_errors=N`**: stops as soon as N issues have accumulated and returns exactly N.
  - Example: three bad rows plus `max_errors=1` gives `issue_count==1`.
  - Later columns, strict, unique and rules are then not evaluated.
- **Composite `unique=[]`**: one issue with `column=None`, `rule="composite_unique"`, `message="Composite unique columns cannot be empty"`.
- **Composite unique column absent**: `rule="missing_column"`, `message=f"Column {c!r} not found"`.
- **Composite duplicates**: one issue per duplicated row (all occurrences), with `message=f"Duplicate rows found for columns {list(unique)}"`. Example: `"Duplicate rows found for columns ['a', 'b']"`.
- **Rule raises `KeyError('z')`**: an issue with `column="z"`, `rule="missing_column"`, `message="Cross-field rule referenced a missing column: 'z'"`.
- **Zero-row frame with the column present**: `row_count==0` and no row issues. A dtype issue can still appear.
- **Empty schema `Schema({})`**: passes. With `strict=True`, every frame column is reported.

## 7. Rules stated in docstrings
- "Validation stops once this limit (`max_errors`) is reached."
- `passed` means "zero error-level issues".

## 8. Testing hints
- Build frames with `ar.from_pandas(pd.DataFrame({...}))`.
- For nulls in string columns, use `""` or `"  "`. These count as null regardless of how the C++ core stores None.
- Prefer substring or `rule` assertions for dtype messages, because the actual-dtype string comes from the C++ core.
- `schema.validate(f)` and `ar.validate(f, schema)` return equal results; you can compare them via `.to_dict()`.
- Example of a precise test:
  ```python
  f = ar.from_pandas(pd.DataFrame({"a": [1, 5, 10]}))
  r = ar.Schema({"a": ar.Int64(min=2, max=9)}).validate(f)
  assert [(i.rule, i.row_index) for i in r.issues] == [("min", 1), ("max", 3)]
  assert r.bad_rows == [1, 3] and not r.passed
  ```
- Pitfalls:
  - `row_index` is 1-based. Do not expect 0.
  - `unique` reports all occurrences, not just the second.
  - `Int64(min=5)` with value 5 passes.
  - `issue.value` may be a numpy scalar, so compare with `==`.
  - `severity="warning"` issues still appear in `issues` and `bad_rows`, but `passed` stays True.
  - Do not write tests that rely on `TypeError` raised by rules being caught; it propagates.
