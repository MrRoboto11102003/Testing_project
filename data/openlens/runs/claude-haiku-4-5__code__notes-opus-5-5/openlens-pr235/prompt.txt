# openlens.analysis.plots.draw_system_outline
def draw_system_outline(ax, system: OpticalSystem) -> None:
    """Draw the lens element outlines of ``system`` onto ``ax`` (Z vs Y).

    Canonical convention: ``lens.thickness`` is the CENTER (vertex to
    vertex) thickness, matching the ray tracers, the ABCD matrix, the
    lensmaker equation, and ``LensGeometry``. The rim (edge) thickness is
    derived as ``thickness - sag1 + sag2`` at the clear aperture.
    """
    current_z = 0.0
    for i, element in enumerate(system.elements):
        lens = element.lens
        thickness = lens.thickness

        # Shared outline (same helper as every 2D view), shifted so the
        # front vertex sits at current_z.
        outline = LensGeometry.lens_outline(lens, num_points=50, max_points=2000)
        bad = not outline["feasible"]
        z1 = [current_z + x for x, _ in outline["front"]]
        y_front = [y for _, y in outline["front"]]
        z2 = [current_z + x for x, _ in outline["back"]]
        y_back = [y for _, y in outline["back"]]
        edge_z = [current_z + x for x, _ in outline["edge"]]
        edge_y = [radius for _, radius in outline["edge"]]

        # Filled lens (matches editor's translucent fill) + colored outlines
        # Build closed polygon: front -> rim chain -> back -> mirrored chain.
        try:
            from matplotlib.patches import Polygon

            poly_z = z1 + edge_z[1:] + z2 + list(reversed(edge_z[1:-1]))
            poly_y = y_front + edge_y[1:] + y_back + [-r for r in reversed(edge_y[1:-1])]
            poly = Polygon(
                list(zip(poly_z, poly_y)),
                closed=True,
                facecolor=COLOR_LENS_BAD if bad else COLOR_LENS_FILL,
                edgecolor="none",
                alpha=0.3,
            )
            ax.add_patch(poly)
        except (ImportError, TypeError, ValueError) as e:
            # A degenerate profile (mismatched edge data, non-finite
            # vertices) must not stop the outlines from being drawn, but a
            # silently skipped fill hides the cause.
            logger.debug("Skipping lens fill: %s", e)

        front_color = COLOR_LENS_BAD if bad else COLOR_LENS_R1
        back_color = COLOR_LENS_BAD if bad else COLOR_LENS_R2
        ax.plot(z1, y_front, color=front_color, alpha=0.9, linewidth=2)
        ax.plot(z2, y_back, color=back_color, alpha=0.9, linewidth=2)
        # Rim chains (custom profile or the straight default edge)
        ax.plot(
            edge_z,
            edge_y,
            color=COLOR_LENS_RIM,
            alpha=0.7,
            linewidth=1,
        )
        ax.plot(
            edge_z,
            [-r for r in edge_y],
            color=COLOR_LENS_RIM,
            alpha=0.7,
            linewidth=1,
        )

        if i < len(system.air_gaps):
            current_z += thickness + system.air_gaps[i].thickness
        else:
            current_z += thickness

# openlens.validation.check_physical_feasibility
def check_physical_feasibility(
    radius1: float,
    radius2: float,
    thickness: float,
    diameter: float,
    is_parabolic_1: bool = False,
    parabolic_sag_1: float = 0.0,
    is_parabolic_2: bool = False,
    parabolic_sag_2: float = 0.0,
    clear_aperture_1: Optional[float] = None,
    clear_aperture_2: Optional[float] = None,
    conic_constant_1: float = 0.0,
    conic_constant_2: float = 0.0,
) -> Tuple[bool, Optional[str]]:
    """
    Check if lens parameters are physically feasible.

    Args:
        radius1: Front surface radius (mm)
        radius2: Back surface radius (mm)
        thickness: Center thickness (mm)
        diameter: Lens diameter (mm)
        is_parabolic_1: Whether the front surface is parabolic
        parabolic_sag_1: Front parabolic sag at D/2 (mm)
        is_parabolic_2: Whether the back surface is parabolic
        parabolic_sag_2: Back parabolic sag at D/2 (mm)
        clear_aperture_1: Front clear aperture diameter (mm), or None. Judged
            at the radius the lens is actually polished to rather than at
            diameter/2 - see Lens.get_manufacturing_radius.
        clear_aperture_2: Back clear aperture diameter (mm), or None
        conic_constant_1: Front conic constant K
        conic_constant_2: Back conic constant K

    Returns:
        Tuple[bool, Optional[str]]: (is_feasible, warning_message)
    """
    warnings = []

    # Hard geometric check via the single source of truth: probe the spec
    # through Lens so rim/edge math cannot drift from the tracer or views.
    # Thickness is the CENTER (vertex to vertex) thickness, so the VERTEX-
    # correct edge is thickness - sag1 + sag2 (a plus-sign variant would
    # reject valid concave lenses). Edge <= 0 means the surfaces intersect
    # within the clear aperture; edge None means the aperture overhangs a
    # surface (|R| < D/2, sag undefined).
    hard_error = None
    h = diameter / 2
    try:
        probe = Lens(
            radius_of_curvature_1=radius1,
            radius_of_curvature_2=radius2,
            thickness=thickness,
            diameter=diameter,
            is_parabolic_1=is_parabolic_1,
            parabolic_sag_1=parabolic_sag_1,
            is_parabolic_2=is_parabolic_2,
            parabolic_sag_2=parabolic_sag_2,
            conic_constant_1=conic_constant_1,
            conic_constant_2=conic_constant_2,
            clear_aperture_1=clear_aperture_1,
            clear_aperture_2=clear_aperture_2,
        )
        edge = probe.calculate_edge_thickness()
    except Exception:
        edge = None
    if edge is None:
        for r in (radius1, radius2):
            if r is not None and math.isfinite(r) and abs(r) >= EPSILON and abs(h) > abs(r):
                hard_error = (
                    f"Aperture overhangs a surface (|R| = {abs(r):.1f}mm < "
                    f"D/2 = {h:.1f}mm) - no real surface exists at the rim. "
                    "Reduce the diameter or flatten the radii."
                )
                break
        if hard_error is None:
            hard_error = (
                "Lens geometry is undefined for these parameters. "
                "Reduce the diameter or flatten the radii."
            )
    elif edge <= 0:
        hard_error = (
            f"Surfaces intersect within the clear aperture (edge thickness "
            f"{edge:.2f}mm <= 0). Increase center thickness, reduce diameter, "
            "or flatten the radii."
        )

    # Soft floor: positive yet fragile rim (parabolic-aware via the probe).
    if hard_error is None and edge is not None and edge < MIN_EDGE_THICKNESS:
        warnings.append(
            f"Edge thickness ({edge:.2f}mm) is below {MIN_EDGE_THICKNESS}mm. " "May be fragile."
        )

    min_radius = min(abs(radius1), abs(radius2))

    # Check if thickness is reasonable compared to radii
    if thickness > 0.5 * min_radius:
        warnings.append(
            f"Thickness ({thickness:.1f}mm) is very large compared to "
            f"minimum radius ({min_radius:.1f}mm). Lens may be impractical."
        )

    # Check if diameter is reasonable compared to radii
    if diameter > 0.8 * min_radius:
        warnings.append(
            f"Diameter ({diameter:.1f}mm) is large compared to "
            f"minimum radius ({min_radius:.1f}mm). Edge effects may be significant."
        )

    # Check for very thin lenses relative to diameter
    if thickness < 0.02 * diameter:
        warnings.append(
            f"Lens is very thin ({thickness:.1f}mm) relative to "
            f"diameter ({diameter:.1f}mm). May be fragile."
        )

    if hard_error is not None:
        if warnings:
            return False, hard_error + " " + " ".join(warnings)
        return False, hard_error

    if warnings:
        return False, " ".join(warnings)

    return True, None

# openlens.validation.validate_file_path
def validate_file_path(
    file_path: Union[str, Path],
    must_exist: bool = False,
    create_parent: bool = False,
    param_name: str = "file_path",
) -> Path:
    """
    Validate and sanitize file path.

    Args:
        file_path: Path to validate
        must_exist: If True, file must already exist
        create_parent: If True, create parent directory if it doesn't exist
        param_name: Parameter name for error messages

    Returns:
        Path: Validated and resolved Path object

    Raises:
        ValidationError: If path is invalid
    """
    if not isinstance(file_path, (str, Path)):
        raise ValidationError(f"{param_name} must be a string or Path object")

    try:
        path = Path(file_path).resolve()
    except (ValueError, RuntimeError) as e:
        raise ValidationError(f"Invalid {param_name}: {e}")

    # Check if file must exist
    if must_exist and not path.exists():
        raise ValidationError(f"{param_name} does not exist: {path}")

    # Check parent directory
    parent = path.parent
    if not parent.exists():
        if create_parent:
            try:
                parent.mkdir(parents=True, exist_ok=True)
            except (OSError, PermissionError) as e:
                raise ValidationError(f"Cannot create directory for {param_name}: {e}")
        else:
            raise ValidationError(f"Parent directory does not exist for {param_name}: {parent}")

    # Check if parent is writable (for new files)
    if not must_exist and not os.access(parent, os.W_OK):
        raise ValidationError(f"Parent directory is not writable for {param_name}: {parent}")

    return path

### openlens.analysis.plots.draw_system_outline

# Notes: `openlens.analysis.plots.draw_system_outline(ax, system)`

## 1. Purpose
- Draws every lens element of an `OpticalSystem` onto a matplotlib Axes, in a Z (optical axis) vs Y side view.
- For each lens it draws one translucent filled polygon plus four outline lines (front surface, back surface, upper rim, lower rim).
- Lenses are placed one after another along Z.

## 2. Inputs
- `ax`: any object with `.plot(...)` and `.add_patch(...)`.
  - A real matplotlib Axes or a `unittest.mock.MagicMock()` both work.
  - The function never validates `ax`.
- `system`: an `OpticalSystem`. The function only reads:
  - `system.elements`: a list of `LensElement`. For each it uses `element.lens` and `element.lens.thickness`.
  - `system.air_gaps`: a list of `AirGap`. For each it uses `.thickness`.
  - There is no type check, so any duck-typed object with these attributes works.
- Geometry for each lens comes from `LensGeometry.lens_outline(lens, num_points=50, max_points=2000)`. That function lives in `openlens.geometry` and its code is not shown here. It returns a dict with these keys:
  - `"front"`, `"back"`, `"edge"`: lists of `(x, y)` pairs. For `"edge"` the second value is treated as a radius.
  - `"feasible"`: a bool.

## 3. Raises
- The function raises nothing itself.
- Exceptions of type `ImportError`, `TypeError` or `ValueError` raised while building or adding the fill polygon are swallowed:
  - This covers the `from matplotlib.patches import Polygon` import, the `Polygon(...)` constructor and `ax.add_patch`.
  - The error is logged with `logger.debug("Skipping lens fill: %s", e)` on logger `openlens.analysis.plots`.
  - The four outline lines are still drawn afterwards.
- Any other exception from the fill step propagates. Example: `ax.add_patch` raising `RuntimeError`, or an `ax` object with no `add_patch` attribute (gives `AttributeError`).
- Exceptions from `ax.plot` or from `lens_outline` propagate.

## 4. Returns and guarantees
- Returns `None`.
- `current_z` starts at `0.0`. After element `i` is drawn:
  - if `i < len(system.air_gaps)`: `current_z += lens.thickness + system.air_gaps[i].thickness`
  - otherwise: `current_z += lens.thickness`
- How `add_lens(air_gap_before=...)` fills `air_gaps` is defined in `optical_system.py`, which is not shown. In tests, read `system.air_gaps` directly to compute the expected offsets.
- For each element, with outline `o` and offset `cz`, the values are:
  - `z1 = [cz + x for x, _ in o["front"]]` and `y_front = [y for _, y in o["front"]]`
  - `z2` and `y_back` are built the same way from `o["back"]`.
  - `edge_z = [cz + x ...]` and `edge_y = [r ...]` come from `o["edge"]`.
- For each element, the calls are made in this exact order:
  1. `ax.add_patch(Polygon(verts, closed=True, facecolor=F, edgecolor="none", alpha=0.3))`
     - `verts = zip(z1 + edge_z[1:] + z2 + reversed(edge_z[1:-1]), y_front + edge_y[1:] + y_back + [-r for r in reversed(edge_y[1:-1])])`
  2. `ax.plot(z1, y_front, color=FC, alpha=0.9, linewidth=2)`
  3. `ax.plot(z2, y_back, color=BC, alpha=0.9, linewidth=2)`
  4. `ax.plot(edge_z, edge_y, color=COLOR_LENS_RIM, alpha=0.7, linewidth=1)`
  5. `ax.plot(edge_z, [-r for r in edge_y], color=COLOR_LENS_RIM, alpha=0.7, linewidth=1)`
- Colours depend on `bad = not o["feasible"]`:
  - `F = COLOR_LENS_BAD if bad else COLOR_LENS_FILL`
  - `FC = COLOR_LENS_BAD if bad else COLOR_LENS_R1`
  - `BC = COLOR_LENS_BAD if bad else COLOR_LENS_R2`
  - Rim lines are always `COLOR_LENS_RIM`.
- All colour constants come from `openlens.constants`; their values are not shown. Import them in the test rather than hard-coding strings.
- The function sets no title, labels or grid.

## 5. Side effects
- Draws on `ax` only.
- Does not mutate `system` or the lenses, assuming `lens_outline` is pure (its code is not visible).
- May emit one debug log record per lens whose fill fails.

## 6. Branches and edge cases
- **Empty system** (`OpticalSystem()` with no lenses): no `plot` or `add_patch` calls at all; returns `None`.
- **One lens**: exactly 1 `add_patch` call and 4 `plot` calls. Every z value is offset by `0.0`, so `z1` equals the `x` values of `lens_outline(lens, 50, 2000)["front"]`.
- **Two or more lenses**: each element adds 1 patch and 4 lines. Element `k`'s z values are shifted by the sum of the earlier thicknesses plus the matching `air_gaps[i].thickness`.
- **Fill fails with `ValueError`** (for example a MagicMock `ax` with `ax.add_patch.side_effect = ValueError("x")`): no exception escapes, and the 4 `plot` calls per lens still happen.
- **Infeasible outline** (`feasible` is False): front, back and fill all use `COLOR_LENS_BAD`; the rims stay `COLOR_LENS_RIM`.
  - To force this path without relying on geometry, monkeypatch `LensGeometry.lens_outline` (as imported in `openlens.analysis.plots`) to return a small fake dict.

## 7. Rules from docstring/comments
- `lens.thickness` is the centre (vertex-to-vertex) thickness.
- Rim (edge) thickness is `thickness - sag1 + sag2`; this is computed inside `LensGeometry`, not here.
- A degenerate profile must not stop the outlines being drawn. That is why the fill errors are caught.

## 8. Testing hints
- Use a real Axes on a non-GUI backend:
  ```python
  import matplotlib; matplotlib.use("Agg")
  import matplotlib.pyplot as plt
  fig, ax = plt.subplots()
  ```
  Then check `len(ax.lines) == 4*n` and `len(ax.patches) == n`.
- Read line data with `ax.lines[0].get_xdata()` / `get_ydata()` and compare with `pytest.approx`.
- For exact call and kwarg checks, use a `MagicMock` ax:
  - `ax.plot.call_args_list[k].kwargs` should equal `{"color": ..., "alpha": 0.9, "linewidth": 2}`.
  - `ax.plot.call_count == 4*n`.
- Build systems with the public API:
  ```python
  s = OpticalSystem()
  s.add_lens(Lens())
  s.add_lens(Lens(), air_gap_before=5.0)
  ```
  Compute expected offsets from `s.elements[i].lens.thickness` and `s.air_gaps[i].thickness`, not from assumed values.
- Get the expected geometry by calling `LensGeometry.lens_outline(lens, num_points=50, max_points=2000)` yourself; it uses the same arguments.
- The lower rim's y values are exactly the negated upper rim's y values, and both rims share the same x data. This holds for every lens.
- Pitfalls:
  - Don't hard-code colour strings or outline point counts.
  - Don't expect a return value.
  - Don't expect titles or labels.
  - `plt` figures accumulate across tests, so call `plt.close(fig)`.

### openlens.validation.check_physical_feasibility

# Notes: `openlens.validation.check_physical_feasibility`

## 1. Purpose
- Checks whether a lens spec is physically buildable.
- Returns `(is_feasible, message)`.
- It never raises a `ValidationError`. It probes the geometry through a temporary `Lens`, then adds soft heuristic warnings.

## 2. Inputs
- `radius1`, `radius2`: front and back radius in mm. Negative is concave. `float('inf')` means flat.
  - Not validated here.
  - `abs()` is called on both unconditionally, so `None` or a string raises `TypeError` (see Raises).
- `thickness`, `diameter`: centre thickness and diameter in mm. Used raw in the heuristics. `h = diameter / 2`.
- `is_parabolic_1/2`, `parabolic_sag_1/2`, `clear_aperture_1/2`, `conic_constant_1/2`:
  - These are only forwarded to the probe `Lens(...)`.
  - They affect the result only through `edge`.
- The probe also gets `Lens` defaults for everything else (material `'BK7'` and so on).
- Edge maths is in `openlens/lens.py` (not shown). `MIN_EDGE_THICKNESS` and `EPSILON` come from `openlens.constants` (values not shown).

## 3. Raises
- The probe is wrapped in `try/except Exception`. Any error from `Lens(...)` or `calculate_edge_thickness()` is swallowed and gives `edge = None`.
- `TypeError` from `min(abs(radius1), abs(radius2))` when a radius is not numeric:
  - Example: `check_physical_feasibility(None, 100, 5, 25)` raises `TypeError` (`abs(None)`).
  - If `edge` is None, a string radius may already raise `TypeError` inside `math.isfinite`.
- No `ValidationError` is ever raised.

## 4. Returns and guarantees
- Always returns a 2-tuple:
  - `(True, None)` only when there is no hard error AND no warnings.
  - Otherwise `(False, str)`. **Soft warnings alone also give `False`.**
- Message assembly:
  - hard error and warnings: `hard + " " + " ".join(warnings)`
  - hard error only: `hard`
  - warnings only: `" ".join(warnings)`
- Hard error, decided from `edge = probe.calculate_edge_thickness()`. The comment says edge = t − sag1 + sag2.
  - **`edge is None`**: loop over `(radius1, radius2)` in order. The first `r` that is not None, finite, has `|r| >= EPSILON` and `|h| > |r|` gives:
    `f"Aperture overhangs a surface (|R| = {abs(r):.1f}mm < D/2 = {h:.1f}mm) - no real surface exists at the rim. Reduce the diameter or flatten the radii."`
  - If no radius qualifies: `"Lens geometry is undefined for these parameters. Reduce the diameter or flatten the radii."`
  - **`edge <= 0`**: `f"Surfaces intersect within the clear aperture (edge thickness {edge:.2f}mm <= 0). Increase center thickness, reduce diameter, or flatten the radii."`
- Warnings, appended in this fixed order:
  1. Edge warning, only if there is no hard error and `edge < MIN_EDGE_THICKNESS`:
     `f"Edge thickness ({edge:.2f}mm) is below {MIN_EDGE_THICKNESS}mm. May be fragile."`
  2. If `thickness > 0.5*min_radius`:
     `f"Thickness ({thickness:.1f}mm) is very large compared to minimum radius ({min_radius:.1f}mm). Lens may be impractical."`
  3. If `diameter > 0.8*min_radius`:
     `f"Diameter ({diameter:.1f}mm) is large compared to minimum radius ({min_radius:.1f}mm). Edge effects may be significant."`
  4. If `thickness < 0.02*diameter`:
     `f"Lens is very thin ({thickness:.1f}mm) relative to diameter ({diameter:.1f}mm). May be fragile."`
- `min_radius = min(|R1|, |R2|)`. With both radii inf it is inf, so warnings 2 and 3 can never fire.
- All comparisons are strict (`>`, `<`). Exactly equal to a threshold gives no warning.

## 5. Side effects
- No inputs are mutated and no global state is changed by this module.
- A throw-away `Lens` is constructed. Whether `Lens` touches the material database is in `lens.py` (not visible).

## 6. Branches and edge cases
These examples are computed assuming the usual spherical sag (`R − sqrt(R² − h²)`) inside `Lens`. Values marked "≈" depend on lens.py.

- **Fine biconvex:** `(200, -200, 10, 25)` → `(True, None)`.
  - edge ≈ 9.2; min_radius 200, and no heuristic fires.
- **Flat/flat:** `(inf, inf, 5, 25)` → expected `(True, None)`, with edge ≈ 5.
- **Thickness warning only:** `(float('inf'), -20, 15, 10)`.
  - edge ≈ 14.4; 15 > 10; 10 > 16 is false.
  - Result: `(False, "Thickness (15.0mm) is very large compared to minimum radius (20.0mm). Lens may be impractical.")`
- **Overhang:** `(10, -10, 5, 30)`. h = 15 > 10, so edge is expected None. The diameter warning is also added (30 > 8). Result:
  `(False, "Aperture overhangs a surface (|R| = 10.0mm < D/2 = 15.0mm) - no real surface exists at the rim. Reduce the diameter or flatten the radii. Diameter (30.0mm) is large compared to minimum radius (10.0mm). Edge effects may be significant.")`
- **Overhang picks the first qualifying radius:** `(-8, 100, 5, 20)` → message starts `"Aperture overhangs a surface (|R| = 8.0mm < D/2 = 10.0mm)"`.
- **Intersecting surfaces:** `(50, -50, 1, 40)`, edge ≈ −7.35.
  - Message starts `"Surfaces intersect within the clear aperture (edge thickness "` and contains `"mm <= 0)"`.
  - No heuristics fire: 40 > 40 is false, and 1 < 0.8 is false.
- **Hard error suppresses the edge warning:** when edge <= 0, the "Edge thickness (…) is below" text is never present.
- **Thin-lens warning:** fires only when `thickness < 0.02*diameter`, e.g. t = 0.9, d = 50.
  - The edge-fragile warning may also precede it, depending on `MIN_EDGE_THICKNESS`. Use `in` checks here.

## 7. Rules from comments
- The centre thickness is vertex-to-vertex. The edge is `thickness - sag1 + sag2`; a plus-sign variant would wrongly reject concave lenses.
- `edge <= 0` means the surfaces intersect within the aperture.
- `edge None` means the aperture overhangs a surface (|R| < D/2).
- Clear apertures are judged at the polished radius (`Lens.get_manufacturing_radius`), not at D/2.
- The edge floor is a soft warning, not a hard error.

## 8. Testing hints
- Import with `from openlens.validation import check_physical_feasibility`. Pass positional floats.
- Use exact string equality only for heuristic-only or overhang messages whose numbers come from the inputs (`.1f` formatting, e.g. `30.0mm`).
- For edge-dependent text (`{edge:.2f}`), assert the prefix or substrings, not the full number.
- Pin the ordering and separators: one space between parts, hard error first, and warnings in order edge → thickness → diameter → thin.
- Pin strictness at boundaries:
  - `(inf, 100, 50, 10)` → thickness 50 > 50 is false. No thickness warning; expect `(True, None)` if the edge is OK.
  - `(inf, 100, 50.1, 10)` → the thickness warning appears.
- Pin that soft warnings make the first element `False`, not `True`.
- Pitfalls:
  - Do not assume `MIN_EDGE_THICKNESS` has any specific value. Keep edges large (> 5 mm) when expecting `(True, None)`.
  - Do not expect `ValidationError` for bad geometry; it returns a tuple.
  - `None` radii raise `TypeError`.

### openlens.validation.validate_file_path

# Notes: `openlens.validation.validate_file_path`

## 1. Purpose
- Checks a file path's type, resolves it to an absolute `Path`, and checks that it exists when asked.
- Checks that the parent directory exists, and creates it when asked.
- Checks that the parent directory is writable, but only for a new file.
- Returns the resolved `Path`. It never creates or opens the file itself.

## 2. Inputs
- `file_path`: must be an instance of `str` or `pathlib.Path`. `WindowsPath` and `PosixPath` are accepted because they subclass `Path`. These are rejected: `bytes`, `int`, `None`, `PurePath`/`PurePosixPath` (they are not `Path`), and `os.PathLike` objects that are not `Path`.
- `must_exist: bool = False`: when true, the resolved path must exist. A directory also counts, because only `path.exists()` is checked.
- `create_parent: bool = False`: when true, a missing parent is created with `parent.mkdir(parents=True, exist_ok=True)`.
- `param_name: str = "file_path"`: this string is placed into every error message.
- Normalisation: `path = Path(file_path).resolve()`. This is non-strict, so a nonexistent path is fine. A relative path is resolved against the current working directory, and `""` resolves to the cwd.

## 3. Raises
Every error is `openlens.validation.ValidationError`. It subclasses `Exception` only, not `ValueError`. The checks run in this order:

1. **Type check**: `f"{param_name} must be a string or Path object"`.
   - Trigger: `validate_file_path(123)` gives `"file_path must be a string or Path object"`.
   - Trigger: `validate_file_path(b"x")` raises the same error.
2. **Resolve failure**: `f"Invalid {param_name}: {e}"`. It fires only if `resolve()` raises `ValueError` or `RuntimeError`. Behaviour here is platform-dependent (for example a path with an embedded NUL), so do not test it.
3. **Missing file**: `must_exist and not path.exists()` gives `f"{param_name} does not exist: {path}"`. Here `path` is the *resolved* path.
   - Trigger: `validate_file_path(tmp_path/"nope.txt", must_exist=True)`.
4. **Missing parent, not allowed to create it**: `f"Parent directory does not exist for {param_name}: {parent}"`.
   - Trigger: `validate_file_path(tmp_path/"missing"/"f.txt")`.
5. **Missing parent, creation fails**: `mkdir` raises `OSError`, giving `f"Cannot create directory for {param_name}: {e}"`.
   - Trigger: create a regular file `tmp_path/"blk"`, then call `validate_file_path(tmp_path/"blk"/"sub"/"f.txt", create_parent=True)`.
6. **Parent not writable**: `not must_exist and not os.access(parent, os.W_OK)` gives `f"Parent directory is not writable for {param_name}: {parent}"`.

Ordering facts worth testing:
- `must_exist` is checked before the parent check. For a missing file whose parent is also missing, with `must_exist=True`, you get the "does not exist" message, not the "Parent directory" one.
- The type check comes first. `validate_file_path(None, must_exist=True)` therefore raises the type error.

## 4. Returns and guarantees
- Returns a new `pathlib.Path` equal to `Path(file_path).resolve()`. It is absolute (`.is_absolute()` is True).
- Returns the same value whether the input was a `str` or a `Path`.
- It does not check that the target is a regular file. An existing directory passes, including with `must_exist=True`.
- The extension is not checked; that is done by `validate_json_file_path`.

## 5. Side effects
- The only side effect: with `create_parent=True` and a missing parent, the whole parent chain is created on disk. This happens before the writability check, and the directories stay even if that check later fails.
- The file itself is never created.
- Nothing global changes and there is no logging.

## 6. Branches and edge cases
- **New file, parent exists**: `validate_file_path(tmp_path/"a.txt")` returns `(tmp_path/"a.txt").resolve()`, and the file still does not exist afterwards.
- **Existing file**: `validate_file_path(f, must_exist=True)` returns the resolved path. The writability check is skipped in this branch.
- **Nested missing parent**: `validate_file_path(tmp_path/"x"/"y"/"f.txt", create_parent=True)` creates `tmp_path/x/y` as a directory. The file is not created.
- **Missing parent, no creation**: the same call without `create_parent` raises, and nothing is created (`(tmp_path/"x").exists()` stays False).
- **Custom name**: `param_name="output"` gives messages such as `"output does not exist: ..."` and `"output must be a string or Path object"`.
- **Relative path**: after `monkeypatch.chdir(tmp_path)`, `validate_file_path("rel.txt") == (tmp_path/"rel.txt").resolve()`.

## 7. Rules from docstring
- "Validate and sanitize file path".
- Returns a "Validated and resolved Path object".
- The writability check applies only "for new files", i.e. only when `must_exist` is False.

## 8. Testing hints
- Import with `from openlens.validation import validate_file_path, ValidationError`.
- Use `tmp_path` for every filesystem case.
- Compare results against `(tmp_path/...).resolve()`, not the raw `tmp_path`. On Windows, `tmp_path` may contain 8.3 short names or a different case, and `resolve()` normalises them.
- Messages embed Windows paths containing backslashes. Use `pytest.raises(ValidationError, match=re.escape(expected))`, or assert `str(exc.value) == expected`, with `expected` built from the resolved path.
  - Example: `f"Parent directory does not exist for file_path: {(tmp_path/'missing').resolve()}"`.
- **Do not try to trigger the not-writable error with chmod.** On Windows, `os.access(dir, W_OK)` is effectively always True for directories, and root ignores the bits on POSIX. Instead, monkeypatch it: `monkeypatch.setattr(openlens.validation.os, "access", lambda p, m: False)`.
  - Then `validate_file_path(tmp_path/"a.txt")` raises the "not writable" error.
  - With the same patch, `validate_file_path(existing_file, must_exist=True)` still succeeds, because the check is skipped.
- For the mkdir-failure case, assert only the message prefix. The `{e}` part is an OS-specific error string.
  - Example: `match=r"^Cannot create directory for file_path: "`.
- Do not assert that `ValidationError` is a `ValueError`; it is not.
