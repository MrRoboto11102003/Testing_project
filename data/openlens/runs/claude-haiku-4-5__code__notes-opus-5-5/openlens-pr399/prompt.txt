# openlens.ray.Ray3D.propagate
    def propagate(self, distance: float) -> None:
        """Propagate ray in current direction.

        A zero (or negligible) distance is a no-op rather than a recorded
        step: appending would leave consecutive identical path points, which
        breaks any consumer that derives a direction or a segment length from
        consecutive path points.
        """
        if abs(distance) <= EPSILON:
            return
        self.origin = self.origin + self.direction * distance
        self.path.append(self.origin)
        self.optical_path_length += distance * self.n

# openlens.tracer_3d.LensRayTracer3D.trace_surface
    def trace_surface(
        self, ray: Ray3D, surface_type: str, interaction: str = "refract"
    ) -> RefractionResult:
        """Trace ray interaction with a specific surface."""
        if surface_type == "front":
            center = self.front_center
            is_flat = self.front_is_flat
            is_parabolic = self.front_is_parabolic
            is_conic = self.front_is_conic
            conic_constant = self.conic_constant_1
            vertex = self.front_vertex
            R = self.R1
            default_n1 = REFRACTIVE_INDEX_AIR
            default_n2 = self.n
        elif surface_type == "back":
            center = self.back_center
            is_flat = self.back_is_flat
            is_parabolic = self.back_is_parabolic
            is_conic = self.back_is_conic
            conic_constant = self.conic_constant_2
            vertex = self.back_vertex
            R = self.R2
            default_n1 = self.n
            default_n2 = REFRACTIVE_INDEX_AIR
        else:
            return RefractionResult.MISSED

        fresnel_facets = self.fresnel_facets_1 if surface_type == "front" else self.fresnel_facets_2
        fresnel_hit = None
        if fresnel_facets:
            fresnel_hit = self._intersect_fresnel_surface(ray, surface_type)
            if fresnel_hit is None:
                return RefractionResult.MISSED
            intersection, normal = fresnel_hit
            is_flat = False
            is_parabolic = False
            is_conic = False
        elif is_parabolic:
            # R is the effective vertex radius (may be negative).
            if abs(R) < EPSILON or abs(R) > 1e10:
                normal = -self.optical_axis if surface_type == "front" else self.optical_axis
                intersection = self._intersect_plane(ray, vertex, self.optical_axis)
            else:
                normal = None  # gradient-based: oriented outward below
                intersection = self._intersect_paraboloid(ray, vertex, R, surface_type)
        elif is_conic:
            normal = None  # gradient-based: oriented outward below
            intersection = self._intersect_conic(ray, vertex, R, conic_constant, surface_type)
        elif is_flat:
            normal = self.optical_axis
            intersection = self._intersect_plane(ray, vertex, normal)
        else:
            intersection = self._intersect_sphere(ray, center, abs(R), (R > 0))

        if intersection is None:
            # No forward intersection with this surface (the ray line misses
            # the sphere/paraboloid/plane entirely). This is a miss however
            # the ray got here: report MISSED with state unchanged so the
            # caller terminates the ray, exactly like the 2D tracer. (A ray
            # that has somehow crossed the surface already must NOT be
            # fabricated into a transmitted ray: returning REFRACTED here
            # would leave origin/direction stale while flipping n to air.)
            return RefractionResult.MISSED

        v_to_i = intersection - vertex
        proj = v_to_i.dot(self.optical_axis)
        dist_sq = v_to_i.magnitude_sq() - proj**2

        aperture_sq = self._aperture_sq_front if surface_type == "front" else self._aperture_sq_back
        if dist_sq > aperture_sq:
            return RefractionResult.MISSED

        # Accumulate optical path for the segment just travelled in the
        # current medium (air gap before the front surface, glass between
        # the surfaces). Without this, Ray3D.optical_path_length would only
        # contain the exit-propagation leg and every OPL difference (hence
        # every wavefront map) would be wrong by ~millimetres of glass path.
        segment = intersection - ray.origin
        ray.optical_path_length += segment.magnitude() * ray.n

        # A zero-length hit (t ~ 0) is legitimate and must still refract: in a
        # cemented doublet the two surfaces share a vertex, so the ray has to
        # change index at the point it is already standing on. This is why
        # _intersect_sphere passes min_t=-EPSILON here while the 2D tracer
        # rejects such hits outright.
        #
        # What must not happen is *recording* it. Appending unconditionally
        # left consecutive identical path points separated by zero length, so
        # anything deriving a direction or a segment length from consecutive
        # path points divides by zero. A cemented doublet produced three such
        # repeats. The refraction below is unaffected.
        previous_origin = ray.origin
        ray.origin = intersection
        if not ray.path or (intersection - previous_origin).magnitude() > EPSILON:
            ray.path.append(intersection)

        if fresnel_hit is None and is_parabolic and normal is None:
            # Paraboloid normal: gradient of F = x - (y²+z²)/(2R) - vx = 0,
            # i.e. (1, -y/R, -z/R) in the local frame, oriented outward.
            # (Planar fallback keeps the ∓axis normal assigned above.)
            try:
                local_hit = self._to_local_point(intersection, vertex)
                ly, lz = local_hit.y, local_hit.z
                if abs(R) < EPSILON:
                    normal_local = vec3(1, 0, 0)
                else:
                    normal_local = vec3(1, -ly / R, -lz / R).normalize()
                geometric = self.transform.multiply_vector(normal_local)
            except Exception:
                geometric = None
            if geometric is None:
                normal = -self.optical_axis if surface_type == "front" else self.optical_axis
            else:
                normal = self._outward_normal(geometric, surface_type)
        elif fresnel_hit is None and is_conic and normal is None:
            # Conic gradient in the vertex-local frame:
            # ((1+K)x - R, y, z).
            try:
                local_hit = self._to_local_point(intersection, vertex)
                local_hit = local_hit - self._to_local_point(vertex, vertex)
                normal_local = vec3(
                    (1.0 + conic_constant) * local_hit.x - R,
                    local_hit.y,
                    local_hit.z,
                ).normalize()
                geometric = self.transform.multiply_vector(normal_local)
            except Exception:
                geometric = None
            if geometric is None:
                normal = -self.optical_axis if surface_type == "front" else self.optical_axis
            else:
                normal = self._outward_normal(geometric, surface_type)
        elif fresnel_hit is None and is_flat:
            if surface_type == "front":
                normal = -self.optical_axis
            else:
                normal = self.optical_axis
        elif fresnel_hit is None:
            normal = self._outward_normal(intersection - center, surface_type)

        current_n = ray.n

        if abs(current_n - default_n1) < 1e-3:
            n1, n2 = default_n1, default_n2
        elif abs(current_n - default_n2) < 1e-3:
            n1, n2 = default_n2, default_n1
        else:
            n1, n2 = current_n, default_n2

        if interaction == "reflect":
            ray.reflect(normal, n1, n2)
            return RefractionResult.REFLECTED
        elif interaction == "refract":
            return ray.refract_or_reflect(n1, n2, normal)

        return RefractionResult.MISSED

# openlens.tracer_3d.SystemRayTracer3D.trace_off_axis_rays
    def trace_off_axis_rays(
        self,
        field_angle_deg: float,
        num_rays: int = DEFAULT_NUM_RAYS,
        wavelength_mm: float = WAVELENGTH_GREEN * NM_TO_MM,
    ) -> List[Ray3D]:
        """
        Trace a bundle of rays at a given field angle.
        Rays are distributed across the entrance pupil.
        """
        if not self.system.elements:
            return []

        first_lens = self.system.elements[0].lens
        max_r = first_lens.diameter / 2.0

        ep_x = self.system.elements[0].position
        start_x = ep_x - RAY_START_OFFSET_3D_MM

        angle_rad = math.radians(field_angle_deg)
        direction = vec3(math.cos(angle_rad), math.sin(angle_rad), 0.0)

        rays = []
        for i in range(num_rays):
            r = -max_r + 2.0 * max_r * i / (num_rays - 1) if num_rays > 1 else 0

            p_ep = vec3(ep_x, r, 0)

            t = (ep_x - start_x) / direction.x
            origin = p_ep - direction * t

            ray = Ray3D(origin, direction, wavelength=wavelength_mm)
            self.trace_ray(ray)
            rays.append(ray)

        return rays

### openlens.ray.Ray3D.propagate

# Notes: `openlens.ray.Ray3D.propagate(self, distance: float) -> None`

## 1. Purpose
- Moves the ray's origin forward by `distance` along its current unit `direction`.
- Records the new point in `path` and adds `distance * n` to `optical_path_length`.
- A zero or negligible distance does nothing.

## 2. Inputs
- `distance`: a float or int, in mm. There is no validation or type check.
  - Negative values are accepted. The ray moves backwards and the optical path length decreases.
- The method also reads this ray state:
  - `self.direction`: a `Vector3`. `Ray3D.__init__` stores `direction.normalize()`, so it is a unit vector unless reassigned.
  - `self.n`: the refractive index at call time. The default is `REFRACTIVE_INDEX_AIR`, whose value is in `openlens.constants` and not shown here (presumably 1.0).
- `EPSILON` comes from `openlens.constants`. Its numeric value is not visible here. Do not write tests that depend on its exact size.

## 3. Raises
- No explicit raises.
- A non-numeric `distance` (e.g. `"a"`) fails inside `abs()` with `TypeError`. Do not test this.

## 4. Returns and guarantees
- Returns `None`.
- When `abs(distance) > EPSILON`:
  - `self.origin` is replaced by `self.origin + self.direction * distance`. This is a new `Vector3`, assuming `Vector3` arithmetic returns new objects (that module is not shown).
  - The new origin object is appended to `self.path`, so `ray.path[-1] is ray.origin`.
  - `self.optical_path_length += distance * self.n`.
- These are untouched: `direction`, `n`, `intensity`, `wavelength`, `terminated`, `absorbed`, `polarization_vector`.

## 5. Side effects
- Only mutates the ray's `origin`, `path` and `optical_path_length`.
- The `Vector3` originally passed to the constructor is not modified. It stays as `path[0]`, because `path` starts as `[origin]` with the same object.
- No global state or file I/O.

## 6. Branches and edge cases
- **No-op branch** (`abs(distance) <= EPSILON`): `propagate(0)`, `propagate(0.0)` and `propagate(-0.0)` all return immediately.
  - `origin` is unchanged (same object).
  - `len(path)` is unchanged (still 1 on a fresh ray).
  - `optical_path_length` stays `0.0`.
- **Normal branch**, positive distance. For `r = Ray3D(vec3(0,0,0), vec3(1,0,0))`, calling `r.propagate(10.0)` gives:
  - `r.origin` = (10, 0, 0)
  - `len(r.path) == 2`
  - `r.optical_path_length == 10.0`
- **Direction is unit length.** For `Ray3D(vec3(0,0,0), vec3(2,0,0))`, calling `propagate(5)` moves the origin to x = 5 (not 10).
  - For direction `vec3(3,4,0)`, calling `propagate(5)` gives origin (3, 4, 0).
- **Uses the current n.** For `Ray3D(vec3(0,0,0), vec3(0,0,1), n=1.5)`, calling `propagate(2.0)` gives:
  - origin (0, 0, 2)
  - `optical_path_length == 3.0`
- **Negative distance.** For a fresh x-directed ray, calling `propagate(-4.0)` gives:
  - origin (-4, 0, 0)
  - `optical_path_length == -4.0` (times n)
  - the path gains one point
- **Accumulation.** Calling `propagate(1)` then `propagate(2)` on an x ray gives:
  - origin x = 3
  - `path` x-values `[0, 1, 3]`
  - `optical_path_length == 3 * n`
  - A `propagate(0)` call in between adds no path point.
- **Changing n between calls.** Set `r.n = 2.0` after the first `propagate(1)` (with n = 1), then call `propagate(1)`. The total optical path length is 3.0.

## 7. Rules from the docstring
- "A zero (or negligible) distance is a no-op rather than a recorded step."
- The reason given: consecutive identical path points break consumers that derive a direction or segment length from consecutive points.
- So `path` must never gain a duplicate point from a zero-length move.

## 8. Testing hints
- Build rays like this: `from openlens.ray import Ray3D` and `from openlens.vector3 import vec3, Vector3`. Call it as `Ray3D(vec3(x,y,z), vec3(dx,dy,dz), n=...)`.
- Read coordinates via `.x`, `.y`, `.z` or `.to_tuple()`.
  - Compare with `pytest.approx`, because normalisation introduces float error (e.g. direction (3,4,0)/5).
- Choose distances well clear of `EPSILON`, whose value is unknown.
  - Use `0` / `0.0` for the no-op case.
  - Use values of 1e-3 or more for real moves.
  - Avoid tiny values like 1e-12.
- Assert the exact `len(ray.path)` both after no-op calls and after real calls. This pins down the no-op rule.
- Assert `ray.path[-1]` coordinates equal `ray.origin`, and `ray.path[0]` equals the start point.
- Assert `optical_path_length` is exactly `distance * n` for a fresh ray. Use n values like 1.5 or 2.0, so a dropped or replaced factor of n is detected.
- Assert that `direction`, `n` and `intensity` are unchanged afterwards.
- Do not rely on the default `n` being exactly 1.0. Pass `n=` explicitly.

### openlens.tracer_3d.LensRayTracer3D.trace_surface

# Notes: `LensRayTracer3D.trace_surface(ray, surface_type, interaction="refract")`

## 1. Purpose
- Finds where a `Ray3D` meets one surface ("front" or "back") of a lens.
- Moves the ray to that point and adds the optical path length for the segment.
- Then refracts or reflects the ray in place.
- Returns a `RefractionResult` (`openlens.ray`).

## 2. Inputs
- `self`: a `LensRayTracer3D(lens, transform=None, x_offset=0.0)`.
  - Lens values are **snapshotted at construction**: R1/R2 (`get_effective_radius_*`), thickness `d`, diameter `D`, `n=lens.refractive_index`, parabolic/conic flags, clear apertures, Fresnel facets.
  - Editing the lens afterwards does not affect `trace_surface`.
  - With no transform, the front vertex is at world `(x_offset,0,0)` and the back vertex at `(x_offset+d,0,0)`. The optical axis is +x.
- `ray`: a `Ray3D(origin: Vector3, direction: Vector3, ..., n=REFRACTIVE_INDEX_AIR)`. It is mutated.
- `surface_type`: only `"front"` or `"back"`. Any other string returns `MISSED` immediately.
- `interaction`:
  - `"refract"` (default) or `"reflect"`.
  - Any other value is accepted but returns `MISSED` *after* the ray has already been moved (see §6).

**How the surface kind is chosen** (in `_calculate_geometry`; the order matters):
1. Parabolic: `is_parabolic_k` and `|parabolic_sag_k| > EPSILON`.
2. Flat: `_is_flat(R)` (defined in `openlens.lens`, not shown; presumably an infinite or huge radius).
3. Conic: `|conic_constant_k| > EPSILON`.
4. Otherwise a sphere.
   - Front centre is at local `x=R1`; back centre is at local `x=d+R2`.
   - The cap nearest the vertex is used (`abs(R)`, convex flag `R>0`).

## 3. Raises
- No explicit `raise`.
- Errors could only come from helpers in other modules (`Ray3D.reflect`, `refract_or_reflect`, `OpticalIntersector`). Do not test for exceptions.

## 4. Returns and guarantees
- `RefractionResult.MISSED` in the following cases. In all of them the ray is **unchanged** (origin, direction, n, OPL, path):
  - bad `surface_type`;
  - no intersection;
  - the hit lies outside the aperture;
  - the Fresnel facet intersection is None.
- Aperture test:
  - The radial distance squared, measured perpendicular to the axis from the vertex, must be `<= (CA/2 + 1e-9)**2`.
  - `CA` is `lens.get_clear_aperture_k()`, falling back to `D` when it returns falsy.
  - Hit exactly at `CA/2` passes. A hit at radius `> CA/2` is a miss.
- On a hit, in this order:
  1. `ray.optical_path_length += |hit - old_origin| * ray.n`, using the `n` from *before* refraction.
  2. `ray.origin = hit`.
  3. `hit` is appended to `ray.path` only if the path is empty or the ray moved by more than `EPSILON`. A zero-length hit is not recorded again, but it is still refracted.
- Choice of `(n1, n2)`. Defaults: front is `(air, lens.n)`, back is `(lens.n, air)`.
  - If `|ray.n - default_n1| < 1e-3`, use the defaults.
  - Else if `|ray.n - default_n2| < 1e-3`, swap them: `(default_n2, default_n1)`.
  - Otherwise use `(ray.n, default_n2)`.
- `"reflect"`: calls `ray.reflect(normal, n1, n2)` and always returns `REFLECTED`.
- `"refract"`: returns whatever `ray.refract_or_reflect(n1, n2, normal)` returns. That is in `openlens.ray`; expect `REFRACTED`, or `REFLECTED` on total internal reflection.
- Normals are always oriented outward: the front normal is about `-axis`, the back normal about `+axis`.
- Never sets `ray.terminated`.

## 5. Side effects
- Mutates `ray`: origin, OPL, path, and direction/n through the refraction or reflection call.
- The tracer and the lens are not changed.

## 6. Branches and edge cases
Shared test lens:
`lens = Lens(radius_of_curvature_1=50, radius_of_curvature_2=-50, thickness=5, diameter=25, refractive_index=1.5)`
`t = LensRayTracer3D(lens)`

- **On-axis front hit.**
  - Call: `ray = Ray3D(vec3(-10,0,0), vec3(1,0,0))`, then `t.trace_surface(ray,"front")`.
  - Returns `REFRACTED`; origin is `(0,0,0)`; OPL increases by `10.0` (n=1).
  - Direction stays about `(1,0,0)` (normal incidence).
- **Back hit after that.**
  - Origin `(0,0,0)` going +x hits `(5,0,0)` (back sphere centre `-45`, radius `50`).
  - OPL increases by `5*ray.n`. If `refract_or_reflect` set `n=1.5`, that is `+7.5`.
  - Since `ray.n≈1.5`, `(n1,n2)=(1.5,1.0)`.
- **Off-axis sphere hit.**
  - Ray from `(-10,5,0)` going +x hits the front at `x = 50 - sqrt(2475) ≈ 0.2506281`, `y=5`.
  - OPL increases by about `10.2506281`.
- **Aperture miss.**
  - Ray from `(-10,13,0)` going +x: the sphere is hit (|y|<50), but `169 > 12.5²`, so `MISSED`.
  - Origin stays `(-10,13,0)` and OPL is unchanged.
- **Sphere missed entirely.**
  - Ray at `y=60` going +x returns `MISSED`.
- **Invalid surface type.**
  - `t.trace_surface(ray, "middle")` returns `MISSED` with nothing changed.
- **Unknown interaction.**
  - `t.trace_surface(ray, "front", "absorb")` returns `MISSED`.
  - However, origin is now `(0,0,0)` and OPL has increased by 10. This is a deliberate observable quirk.
- **Reflect.**
  - On-axis front hit with `"reflect"` returns `REFLECTED`.
  - Direction should become about `(-1,0,0)`; this relies on `Ray3D.reflect` being standard mirror reflection.
- **Zero-length hit.**
  - A ray whose origin is already at the vertex, with a non-empty path: the path length does not grow and OPL increases by 0.
  - The ray is still refracted.
- **x_offset.**
  - `LensRayTracer3D(lens, x_offset=100)` puts the front hit at `(100,0,0)`.
- **Flat surface.**
  - The surface is a plane through the vertex.
  - A ray parallel to it, e.g. direction `(0,1,0)` (|denom|<EPSILON), returns `MISSED`.
  - A plane hit behind the ray (`t < -EPSILON`) returns `MISSED`.
- **Parabolic surface with effective R that is ~0 or > 1e10.**
  - Treated as a plane.
- **Fresnel lens (facets non-empty).**
  - Only facet hits count. If no facet is hit, the result is `MISSED`.
  - Facet construction is in `openlens.geometry` (not shown), so avoid exact values.

## 7. Rules from comments
- "A zero-length hit (t ~ 0) is legitimate and must still refract" (cemented doublets), but the point must not be recorded twice in `path`.
- A no-intersection result must be `MISSED` with the state unchanged. The ray must never be faked into a transmitted ray.
- OPL is accumulated per segment using the current medium's `n`.
- Normal convention: the front normal points to −axis and the back normal to +axis, whatever the sign of the curvature.

## 8. Testing hints
- Imports:
  - `from openlens.tracer_3d import LensRayTracer3D`
  - `from openlens.ray import Ray3D, RefractionResult`
  - `from openlens.vector3 import vec3`
- Compare floats with `pytest.approx(abs=1e-6)`. Compare results with `is` / `==` on the enum.
- Record `len(ray.path)` before the call and compare deltas. The initial contents of `Ray3D.path` are defined in `openlens.ray` and are probably `[origin]`.
- Pass `refractive_index=` explicitly. Lens defaults (type, material index) come from code not shown here.
- Safest assertions:
  - return values;
  - `ray.origin` coordinates;
  - the OPL delta on the front surface (n=1);
  - that the state is unchanged on `MISSED`.
- Assertions that depend on the `openlens.ray` implementation, so keep them loose:
  - `ray.n` after refraction;
  - exact refracted directions on oblique rays.
- Do not test backward-travelling rays. The result depends on `rank_cap_roots` (not shown).

### openlens.tracer_3d.SystemRayTracer3D.trace_off_axis_rays

# Notes: `SystemRayTracer3D.trace_off_axis_rays(field_angle_deg, num_rays=DEFAULT_NUM_RAYS, wavelength_mm=WAVELENGTH_GREEN*NM_TO_MM)`

## 1. Purpose
- Builds a 1-D meridional fan of parallel rays at a field angle.
- The fan spans the first element's full diameter in the y direction.
- Each ray is traced through the whole system with `self.trace_ray`.
- Returns the traced `Ray3D` objects in fan order.

## 2. Inputs
- `field_angle_deg` (float, required):
  - Converted with `math.radians`.
  - Every ray gets `direction = vec3(cos(a), sin(a), 0.0)`. At 0° this is exactly `(1.0, 0.0, 0.0)`.
  - There is no validation.
- `num_rays` (int):
  - The default is `openlens.constants.DEFAULT_NUM_RAYS`. Its value is not visible here, so import it rather than hard-coding a number.
  - It is used in `range(num_rays)`, so values of 0 or below give an empty list.
- `wavelength_mm` (float):
  - Passed as `Ray3D(..., wavelength=wavelength_mm)`. It is in **mm**, not nm.
  - The default is `WAVELENGTH_GREEN * NM_TO_MM`, from `openlens.constants`.
  - There is no validation.
- Geometry is read from `self.system.elements[0]`:
  - `max_r = elements[0].lens.diameter / 2.0`. Only the **first** lens's diameter matters.
  - `ep_x = elements[0].position`.
- Two more constants come from `openlens.constants`, with values not visible here: `RAY_START_OFFSET_3D_MM` and `RAY_EXIT_PROPAGATION_3D_MM` (used inside `trace_ray`).

## 3. Raises
- Nothing is raised explicitly.
- `field_angle_deg=90` does not raise. `cos` is about 6e-17, so `t` becomes huge but is finite.
- `num_rays` that is not an int (e.g. `2.5`) raises `TypeError` from `range`.

## 4. Returns and guarantees
- Return type is `List[Ray3D]`. Every ray is a new object.
- Length is `max(num_rays, 0)`.
- Fan heights for ray `i`:
  - When `num_rays > 1`: `r_i = -max_r + 2*max_r*i/(num_rays-1)`. This runs from `-max_r` up to `+max_r`, ascending, and the endpoints are included exactly.
  - When `num_rays == 1`: `r = 0`.
- Initial origin, set before tracing:
  - The ray is aimed so its straight line passes through `(ep_x, r_i, 0)`.
  - `t = RAY_START_OFFSET_3D_MM / cos(a)`.
  - `origin = (ep_x, r_i, 0) - direction*t`.
  - That gives `x = ep_x - OFFSET` and `y = r_i - tan(a)*OFFSET` (up to float rounding), with `z = 0`.
- Each ray has already been traced by `self.trace_ray(ray)`, which mutates the ray:
  - `ray.origin` and `ray.direction` are the **final** values, not the start values.
  - Points are appended to `ray.path`.
  - `optical_path_length` is accumulated.
- How a ray ends:
  - A ray that is not terminated gets a final `propagate(RAY_EXIT_PROPAGATION_3D_MM)`.
  - A ray that misses a surface gets `terminated=True` and stays at its last point.
  - In both cases no extra propagation is added after a termination.
- Whether `Ray3D.__init__` seeds `path` with the origin is in `openlens/ray.py`, which is not visible. Don't assume `path[0]`.

## 5. Side effects
- The system is not mutated.
- `self._batch` (the cached scene) may be built or rebuilt on the first trace.
- No files or global state are touched.

## 6. Branches and edge cases
- **Empty system:** `OpticalSystem()` with no lenses gives `[]`. This happens before any trace, and `self._batch` stays `None`.
- **`num_rays=1`:** one ray at `r=0`. On axis at 0° it stays on the axis: final `y == 0`, `z == 0`, direction about `(1,0,0)`.
- **`num_rays=0` or negative:** `[]`. Note this is not the empty-system path; `ep_x` and `max_r` are still read first.
- **Field 0°:**
  - All rays start at the same x.
  - Every ray's z stays exactly `0.0`, because the direction's z and all normal z-components are 0.
  - For a symmetric lens and odd `n`, ray `i` and ray `n-1-i` are mirror images: `y_i ≈ -y_{n-1-i}`, and the middle ray has y about 0.
- **Nonzero angle:**
  - Directions have `y = sin(a)`.
  - Edge rays may exceed the clear aperture because surface sag shifts the hit point, so they can come back `terminated`.
  - Don't assert pass/fail for edge rays at an angle.
- **Edge rays at 0°:** `r = ±max_r` hit the front surface at radius `max_r`. The aperture test is `(ca/2 + 1e-9)^2`, so they pass the front surface, assuming the clear aperture defaults to the diameter (`Lens.get_clear_aperture_1` is not shown).
- **Inside `trace_ray`:**
  - A casing can absorb rays (`ray.absorbed = True`).
  - An aperture stop can vignette rays.
  - Both are optional; without a casing or stop, neither happens.

## 7. Rules from comments / docstring
- "Rays are distributed across the entrance pupil."
- In practice this means the plane `x = elements[0].position` spanning ±(first diameter)/2.
- Stop and pupil size are **not** used for the distribution.

## 8. Testing hints
Build the system:
```python
from openlens.lens import Lens
from openlens.optical_system import OpticalSystem
from openlens.tracer_3d import SystemRayTracer3D
from openlens.ray import Ray3D
from openlens.vector3 import vec3
from openlens import constants as C
lens = Lens(radius_of_curvature_1=50.0, radius_of_curvature_2=-50.0, thickness=5.0,
            diameter=25.0, refractive_index=1.5168)
sys_ = OpticalSystem(); sys_.add_lens(lens)
tr = SystemRayTracer3D(sys_)
```

Strong checks:
- **Count:**
  - `len(tr.trace_off_axis_rays(0.0, num_rays=7)) == 7`.
  - The default gives `len == C.DEFAULT_NUM_RAYS`.
  - `num_rays=0` gives `[]`.
  - An empty system gives `[]`.
- **Exact equivalence (most robust):** rebuild each ray by hand and trace it with a second tracer. Then compare `origin`, `direction` and `optical_path_length` with `pytest.approx`.
  ```python
  ep = sys_.elements[0].position
  ray = Ray3D(vec3(ep, r, 0) - d * (C.RAY_START_OFFSET_3D_MM / d.x), d, wavelength=w)
  SystemRayTracer3D(sys_).trace_ray(ray)
  ```
- **Symmetry at 0° with odd n:**
  - The middle ray's final y is about 0 (`abs < 1e-9`).
  - Outer rays mirror each other.
  - Every final `origin.z == 0.0` and `direction.z == 0.0`.
- **Focusing:** a biconvex lens with positive R1 and negative R2 makes the top edge ray's final `direction.y < 0` and the bottom one's `> 0` at 0°.
- **Fan order:** final y or direction changes monotonically with index. The first ray starts at `-max_r`.
- **Wavelength:** `ray.wavelength == wavelength_mm`, if `Ray3D` stores it under that attribute (check `openlens/ray.py`, which is not visible).
- **First diameter only:** adding a second, larger lens doesn't change the start heights. Check this with the equivalence method above.

Pitfalls:
- Read `sys_.elements[0].position` instead of assuming it is 0; how `add_lens` sets the position is in `optical_system.py`.
- Don't hard-code `DEFAULT_NUM_RAYS` or the offset and exit-propagation constants.
- The returned `ray.origin` is the end point after tracing, not the launch point.
- Use `approx` for all floats.
