    def propagate_to_image(
        self,
        wavelength_mm: float = WAVELENGTH_GREEN * NM_TO_MM,
        grid_size: int = 32,
        image_plane_x: Optional[float] = None,
        detector_size: float = 0.1,
        detector_pixels: int = 64,
    ) -> Tuple[np.ndarray, np.ndarray, np.ndarray]:
        """
        Decompose pupil into Gaussian beamlets, propagate them, and sum the
        coherent field at the image plane.

        Returns:
            Y, Z grids (detector) and Intensity map.
        """
        if not NUMPY_AVAILABLE:
            raise ImportError("numpy is required for BSP calculation.")

        # 1. Define Pupil Grid and Beamlets
        if not self.system.elements:
            return np.array([]), np.array([]), np.array([])

        ep_diam = self.system.elements[0].lens.diameter
        max_r = ep_diam / 2.0

        # Beamlet spacing. The grid is np.linspace(-max_r, max_r, grid_size),
        # which spans 2*max_r in grid_size-1 intervals, so the spacing is
        # ep_diam/(grid_size-1). Dividing by grid_size overstated the spacing
        # by grid_size/(grid_size-1) - 3% at grid_size=32 - and that error went
        # straight into the beamlet waist below.
        delta = (2.0 * max_r) / (grid_size - 1) if grid_size > 1 else 2.0 * max_r
        # Beamlet waist w0. Overlap factor ~1.5
        w0 = 1.5 * delta

        # Initial q (planar wavefront at pupil)
        # 1/q = 1/R - i * lambda / (pi * w0^2)
        # R = inf -> 1/q = -i * lambda / (pi * w0^2)
        # q = i * (pi * w0^2 / lambda) = i * zR
        zR = math.pi * w0**2 / wavelength_mm
        q0 = complex(0, zR)

        beamlets = []

        # Grid loop (using simple loops for beamlet creation)
        start_x = self.system.elements[0].position - 20
        direction = vec3(1, 0, 0)

        y_vals = np.linspace(-max_r, max_r, grid_size)
        z_vals = np.linspace(-max_r, max_r, grid_size)

        pupil_x = self.system.elements[0].position
        dist = pupil_x - start_x
        element_tracers = self._element_tracers()

        for py in y_vals:
            for pz in z_vals:
                if py**2 + pz**2 > max_r**2:
                    continue

                # Create ray
                origin = vec3(pupil_x, py, pz) - direction * dist
                ray = Ray3D(origin, direction, wavelength=wavelength_mm)

                # Create beamlet
                beam = GaussianBeam(wavelength_mm, q0, q0, ray)

                # Manual trace loop:
                self._trace_beamlet(beam, element_tracers)

                if not beam.ray.terminated:
                    beamlets.append(beam)

        # 2. Define Detector Grid
        if image_plane_x is None:
            image_plane_x = self._default_image_plane(wavelength_mm)

        det_y = np.linspace(-detector_size / 2, detector_size / 2, detector_pixels)
        det_z = np.linspace(-detector_size / 2, detector_size / 2, detector_pixels)
        DY, DZ = np.meshgrid(det_y, det_z)

        E_field = np.zeros_like(DY, dtype=complex)

        # 3. Sum Beamlets
        for beam in beamlets:
            # Intersect ray with image plane
            if abs(beam.ray.direction.x) < 1e-6:
                continue

            t = (image_plane_x - beam.ray.origin.x) / beam.ray.direction.x
            # Propagate beam to image plane
            beam.propagate(t)

            center_y = beam.ray.origin.y
            center_z = beam.ray.origin.z

            # Beam width at image plane
            wx = beam.w_x
            wy = beam.w_y

            # Radius of curvature
            Rx = beam.R_x
            Ry = beam.R_y

            # Local coordinates on detector
            local_y = DY - center_y
            local_z = DZ - center_z

            # Simplified Gaussian field
            amp = (w0 / wx) * np.exp(-(local_y**2) / (wx**2) - (local_z**2) / (wy**2))

            # Phase
            k = 2 * np.pi / wavelength_mm
            phase_curv = k * (local_y**2 / (2 * Rx) + local_z**2 / (2 * Ry))
            phase_opl = k * beam.ray.optical_path_length

            E_beam = amp * np.exp(-1j * (phase_curv + phase_opl))
            E_field += E_beam

        Intensity = np.abs(E_field) ** 2
        return DY, DZ, Intensity

# openlens.analysis.psf_mtf.ImageQualityAnalyzer.calculate_mtf
    def calculate_mtf(
        self,
        field_angle_deg: float = 0.0,
        wavelength_nm: float = WAVELENGTH_GREEN,
        focus_shift_mm: float = 0.0,
        max_freq: float = 100.0,
        use_diffraction: bool = False,
        **kwargs,
    ) -> Dict[str, Any]:
        """
        Calculate the Modulation Transfer Function (MTF).

        Args:
            field_angle_deg: Field angle in degrees
            wavelength_nm: Wavelength in nm
            focus_shift_mm: Defocus in mm
            max_freq: Maximum spatial frequency to return (lp/mm)
            use_diffraction: If True, uses FFT of Pupil Function.

        Returns:
            Dictionary containing:
                - freq: Array of spatial frequencies (lp/mm)
                - mtf_tan: Tangential MTF values
                - mtf_sag: Sagittal MTF values
        """
        # Handle legacy aliases
        if "field_angle" in kwargs:
            field_angle_deg = kwargs.pop("field_angle")
        if "max_freq" in kwargs:
            max_freq = kwargs.pop("max_freq")
        if kwargs:
            raise TypeError(f"Unexpected kwargs: {list(kwargs.keys())}")
        if not self.system.elements:
            logger.warning("MTF: system has no elements; returning an empty MTF")
            empty = np.array([])
            return {
                "freq": empty,
                "mtf_tan": empty.copy(),
                "mtf_sag": empty.copy(),
            }
        if use_diffraction:
            # For diffraction MTF, we calculate the autocorrelation of the pupil function
            # OR take the FFT of the Diffraction PSF.
            # DiffractionPSFCalculator has a helper for this.

            # 1. Get Wavefront
            pupil_grid_size = 64
            wf = self.wavefront_sensor.get_pupil_wavefront(
                field_angle_deg=field_angle_deg,
                wavelength_nm=wavelength_nm,
                grid_size=pupil_grid_size,
            )
            if self._wavefront_is_degenerate(wf):
                logger.warning(
                    "MTF: empty pupil wavefront for %s; returning an empty MTF",
                    self.system.name,
                )
                empty = np.array([])
                return {
                    "freq": empty,
                    "mtf_tan": empty.copy(),
                    "mtf_sag": empty.copy(),
                }

            # 1b. Apply the defocus. This branch had its own copy of the
            # wavefront->PSF chain and dropped focus_shift_mm on the floor, so
            # the diffraction MTF was the same at every defocus.
            _, _, na = self._pupil_geometry()
            self._apply_defocus(wf, focus_shift_mm, wavelength_nm, na)

            # 2. Calculate PSF (high res)
            pad_factor = 4
            psf_raw = DiffractionPSFCalculator.calculate_psf(wf, pad_factor=pad_factor)

            # 3. Calculate MTF from PSF
            mtf_2d = DiffractionPSFCalculator.calculate_mtf(psf_raw)

            # 4. Extract Profiles
            center_y, center_x = mtf_2d.shape[0] // 2, mtf_2d.shape[1] // 2

            # Tangential (along Y axis in frequency domain) -> Corresponds to spatial Y modulation?
            # Wait. MTF(fy, fz).
            # Tangential MTF is usually defined for modulation along the tangential direction (Y).
            # This corresponds to frequency fy? Yes.
            # Sagittal MTF -> fz.

            # Extract slices from 2D MTF
            # Positive frequencies only
            n_half = mtf_2d.shape[0] // 2

            # Slice along axes
            mtf_tan_full = mtf_2d[:, center_x]
            mtf_sag_full = mtf_2d[center_y, :]

            mtf_tan = mtf_tan_full[center_y:]  # From DC to max freq
            mtf_sag = mtf_sag_full[center_x:]

            # 5. Frequency Scale
            # Max frequency (cutoff) = 1 / (lambda * F/#)
            # The FFT frequency scale is determined by the spatial sampling.
            # dx (spatial sampling of PSF) = (lambda * f) / (N_pupil * dx_pupil) * (something)
            # Actually, simpler:
            # Max Frequency in MTF corresponds to the cutoff frequency of the system.
            # Cutoff = D / (lambda * f).
            # The MTF array spans from -Cutoff to +Cutoff?
            # No. The MTF array spans -Fs/2 to Fs/2 where Fs = 1/dx_psf.
            # dx_psf = (lambda * f) / L_pupil_grid.
            # So Fs = L_pupil_grid / (lambda * f).
            # This makes sense. The max frequency we can resolve is limited by the pupil size.
            # The MTF array has size M (padded size).
            # So frequency step df = Fs / M = L_pupil_grid / (lambda * f * M).
            # The cutoff frequency is D / (lambda * f).
            # D is approx L_pupil_grid.
            # So the cutoff is at index M * (D/L) ?
            # If L_pupil_grid is exactly D (which we try to set), then Fs = Cutoff * (something).

            # Let's use physical parameters.
            ep_diam = self.system.elements[0].lens.diameter if self.system.elements else 1.0
            efl = self.system.get_system_focal_length() or 100.0

            # Pupil Grid physical size L = ep_diam (approx, based on get_pupil_wavefront range)
            L_grid = ep_diam  # We used +/- max_r = ep_diam/2

            # Sampling frequency in MTF domain
            # Fs = L_grid / (wavelength_nm * 1e-6 * efl)
            # This assumes the PSF calculation uses the standard FFT scaling.
            # Yes, if we pad to M, the spatial extent of PSF is lambda*f/dx_pupil.
            # dx_pupil = L_grid / N.
            # So Extent = lambda*f*N/L_grid.
            # Fs = 1 / (Extent/M) = M / Extent = M * L_grid / (lambda*f*N).
            # Wait, frequency resolution df = 1/Extent.
            # df = L_grid / (lambda * f * N) ? No. 1/Extent = (dx_pupil) / (lambda * f).

            # Let's re-verify:
            # df = 1 / (Total Spatial Range).
            # Total Spatial Range (FOV of PSF) = M * dx_psf.
            # dx_psf = (lambda * f) / (M * dx_pupil).
            # FOV = (lambda * f) / dx_pupil.
            # So df = 1 / ( (lambda*f)/dx_pupil ) = dx_pupil / (lambda * f).
            # This is consistent.

            grid_size = wf.W.shape[0]  # N
            # The pupil grid is np.linspace(-max_r, max_r, N): N samples
            # spanning 2*max_r in N-1 intervals, so the sample spacing is
            # L_grid/(N-1). Dividing by N understated it by N/(N-1) and
            # reported every frequency low by that factor.
            dx_pupil = L_grid / (grid_size - 1) if grid_size > 1 else L_grid
            df = dx_pupil / (wavelength_nm * 1e-6 * efl)

            freqs = np.arange(len(mtf_tan)) * df

            # Filter
            mask = freqs <= max_freq

            return {
                "freq": freqs[mask],
                "mtf_tan": mtf_tan[mask],
                "mtf_sag": mtf_sag[mask],
            }

        # 1. Calculate PSF with sufficient resolution.
        # The window used to be a fixed 0.256 mm with a fixed 128 pixels, so
        # a spot wider than about 0.1 mm radius had most of its light
        # discarded by the histogram - and the MTF was then computed from the
        # clipped core, describing a better lens than the one being tested.
        #
        # The window is now sized from the spot the tracer produces (the same
        # 2.2 RMS radii calculate_psf uses), and the pixel count is derived
        # from it so the sampling stays at ~2 microns. Fixing the window while
        # leaving the pixel count alone would have kept the clipping away but
        # coarsened the sampling, trading one wrong answer for another.
        target_pixel_mm = 0.002  # 2 microns
        window_floor_mm = 0.256  # ~128 pixels at 2 microns

        spot_res = self.spot_analyzer.trace_spot(
            field_angle_y_deg=field_angle_deg,
            wavelength_nm=wavelength_nm,
            focus_shift_mm=focus_shift_mm,
            num_rings=25,
        )
        spot_rms = spot_res.get("rms_radius")
        if spot_rms is not None and math.isfinite(spot_rms) and spot_rms > 0:
            window_size_mm = max(
                window_floor_mm,
                2.0 * GEOMETRIC_WINDOW_SPOT_FACTOR * spot_rms,
            )
        else:
            window_size_mm = window_floor_mm

        num_pixels = max(128, int(math.ceil(window_size_mm / target_pixel_mm)))
        num_pixels = min(num_pixels, GEOMETRIC_MTF_MAX_PIXELS)

        psf_data = self.calculate_psf(
            field_angle_deg=field_angle_deg,
            wavelength_nm=wavelength_nm,
            focus_shift_mm=focus_shift_mm,
            sensor_size_mm=window_size_mm,
            pixels=num_pixels,
        )

        psf = psf_data["image"]  # shape (128, 128)

        # 2. Compute LSF
        # Y is axis 0 (Tangential), Z is axis 1 (Sagittal)

        # Tangential MTF: modulation along Y.
        # LSF_tan(y) = sum(PSF(y, z), axis=z)
        lsf_tan = np.sum(psf, axis=1)  # Sum over Z (cols) -> profile along Y

        # Sagittal MTF: modulation along Z.
        # LSF_sag(z) = sum(PSF(y, z), axis=y)
        lsf_sag = np.sum(psf, axis=0)  # Sum over Y (rows) -> profile along Z

        # 3. Compute MTF (Magnitude of FFT of LSF)
        # Use rfft for efficiency as LSF is real
        mtf_tan_raw = np.abs(np.fft.fft(lsf_tan))
        mtf_sag_raw = np.abs(np.fft.fft(lsf_sag))

        # Normalize DC to 1.0
        if mtf_tan_raw[0] > 0:
            mtf_tan_raw /= mtf_tan_raw[0]
        if mtf_sag_raw[0] > 0:
            mtf_sag_raw /= mtf_sag_raw[0]

        # 4. Frequency Scaling
        # Sampling freq Fs = 1 / pixel_size
        # Freq resolution df = Fs / N
        # Freqs = k * df

        freqs = np.fft.fftfreq(num_pixels, d=psf_data["step_size"])

        # Take only positive frequencies up to max_freq
        # fftfreq returns [0, 1, ..., n/2-1, -n/2, ..., -1]
        # We only want the first half

        n_half = num_pixels // 2
        pos_freqs = freqs[:n_half]
        pos_mtf_tan = mtf_tan_raw[:n_half]
        pos_mtf_sag = mtf_sag_raw[:n_half]

        # Filter to max_freq
        mask = (pos_freqs <= max_freq) & (pos_freqs >= 0)

        return {
            "freq": pos_freqs[mask],
            "mtf_tan": pos_mtf_tan[mask],
            "mtf_sag": pos_mtf_sag[mask],
        }

    def simulate_image(
        self,
        image_array: np.ndarray,
        pixel_size_mm: float = 0.005,
        wavelengths_nm: Tuple[float, float, float] = (
            WAVELENGTH_C_LINE,
            WAVELENGTH_D_LINE,
            WAVELENGTH_F_LINE,
        ),
        field_angle_deg: float = 0.0,
        focus_shift_mm: float = 0.0,
        **kwargs,
    ) -> np.ndarray:
        """
        Simulate image degradation by convolving with PSF.

        Args:
            image_array: Input image, either (H, W) grayscale or (H, W, C)
                multi-channel, with C matching ``wavelengths_nm``. Float input
                is taken as normalized 0-1; integer input is scaled by its
                dtype maximum first.
            pixel_size_mm: Sensor pixel size in mm.
            wavelengths_nm: Tuple of (R, G, B) wavelengths in nm.
            field_angle_deg: Field angle for PSF calculation.
            focus_shift_mm: Defocus in mm.

        Returns:
            Simulated image in the same shape as the input, normalized 0-1
            and always float64.

        Raises:
            ValueError: If the array is neither 2-D nor 3-D, or its channel
                count does not match ``wavelengths_nm``.
        """
        # Handle legacy alias pixel_size
        if "pixel_size" in kwargs:
            pixel_size_mm = kwargs.pop("pixel_size")
        if "wavelengths" in kwargs:
            wavelengths_nm = kwargs.pop("wavelengths")
        if kwargs:
            raise TypeError(f"Unexpected kwargs: {list(kwargs.keys())}")
        try:
            from scipy.signal import fftconvolve

            # Helper for when scipy is available
            def _convolve(img, kern):
                return fftconvolve(img, kern, mode="same")

        except ImportError:
            # Fallback: use numpy-based fft convolution
            def _convolve(img, kern):
                return self._numpy_fftconvolve(img, kern)

        output_image = np.zeros(image_array.shape, dtype=np.float64)

        # Accept 2-D grayscale or 3-D RGB, and integer or float input. The
        # old np.zeros_like(image_array) inherited the input dtype, so a uint8
        # image truncated every float result to 0 or 1 before the clip, and
        # image_array[:, :, i] raised IndexError on a 2-D input.
        image_array = np.asarray(image_array)
        if image_array.ndim not in (2, 3):
            raise ValueError(
                f"image_array must be 2-D (H, W) or 3-D (H, W, C); got shape "
                f"{image_array.shape}"
            )
        if image_array.ndim == 3 and image_array.shape[2] != len(wavelengths_nm):
            raise ValueError(
                f"image_array has {image_array.shape[2]} channels but "
                f"{len(wavelengths_nm)} wavelengths were given"
            )

        if np.issubdtype(image_array.dtype, np.integer):
            # Integer input carries 0..max, not the documented 0..1. Scale it
            # so the convolution sees real intensities; without this the
            # np.clip(0, 1) below saturates the whole image to 1.
            scale = float(np.iinfo(image_array.dtype).max)
            work = image_array.astype(np.float64) / scale
        else:
            work = image_array.astype(np.float64)

        is_grayscale = work.ndim == 2

        # PSF physical size (FOV)
        # Should be large enough to contain the spot.
        # Say 0.1 mm or 0.2 mm.
        # Let's use 0.2 mm or 64x64 pixels minimum
        psf_size_mm = max(0.2, pixel_size_mm * 64)
        psf_pixels = int(psf_size_mm / pixel_size_mm)
        # Ensure odd size for centering
        if psf_pixels % 2 == 0:
            psf_pixels += 1

        # Process each channel
        channel_results = []
        for i, wl in enumerate(wavelengths_nm):
            # Calculate PSF for this channel
            # Use Diffraction PSF for realistic results if possible?
            # Or Geometric? Let's default to Geometric for speed unless specifically asked.
            # Actually, Geometric PSF is often sparse (dots). Convolving with
            # sparse dots looks like multiple images.
            # Diffraction PSF is smoother. Let's use Diffraction.

            psf_data = self.calculate_psf(
                field_angle_deg=field_angle_deg,
                wavelength_nm=wl,
                focus_shift_mm=focus_shift_mm,
                sensor_size_mm=psf_pixels * pixel_size_mm,
                pixels=psf_pixels,
                use_diffraction=True,
            )

            kernel = psf_data["image"]

            # Normalize kernel
            kernel_sum = np.sum(kernel)
            if kernel_sum > 0:
                kernel /= kernel_sum

            # Convolve
            channel = work if is_grayscale else work[:, :, i]
            # mode='same' keeps the output size same as input
            convolved = np.asarray(_convolve(channel, kernel), dtype=np.float64)

            channel_results.append(convolved)

        if is_grayscale:
            # One plane in, one plane out: combine the per-wavelength results
            # rather than inventing a third axis the caller did not ask for.
            output_image[:, :] = np.mean(channel_results, axis=0)
        else:
            for i, convolved in enumerate(channel_results):
                output_image[:, :, i] = convolved

        return np.clip(output_image, 0, 1)

### openlens.analysis.beam_synthesis.BeamSynthesisPropagator.propagate_to_image

# Notes: `BeamSynthesisPropagator.propagate_to_image`

## 1. Purpose
- Splits the entrance pupil (the first element's full diameter) into a square grid of Gaussian beamlets.
- Traces each beamlet through every element and sums their complex fields on a square detector plane perpendicular to x.
- Returns the detector grids `(DY, DZ)` and the intensity `|E|²`.

## 2. Inputs
- `self`: built with `BeamSynthesisPropagator(system)`. The constructor stores `self.system` and creates `self.tracer = SystemRayTracer3D(system)`.
- `wavelength_mm` (default `WAVELENGTH_GREEN * NM_TO_MM`, from `openlens.constants`):
  - This is the vacuum wavelength in mm. It is not validated.
  - It is used for the beamlet Rayleigh range `zR = π·w0²/λ`, for the phase `k = 2π/λ`, and for the paraxial-focus search, which receives `wavelength_mm / NM_TO_MM` nm.
- `grid_size` (int, default 32):
  - Pupil samples per axis: `np.linspace(-r, r, grid_size)`, where `r = elements[0].lens.diameter/2`.
  - `delta = 2r/(grid_size-1)` when `grid_size > 1`, otherwise `2r`. The beamlet waist is `w0 = 1.5*delta`.
  - Only points with `py²+pz² <= r²` become beamlets; the boundary is included.
- `image_plane_x` (float or None, default None):
  - It is not validated.
  - If None, the plane comes from `_default_image_plane` (see §6).
- `detector_size` (float, default 0.1 mm) and `detector_pixels` (int, default 64):
  - Grid: `det = np.linspace(-size/2, size/2, pixels)` on both axes.
  - Neither value is validated.

## 3. Raises
- `ImportError("numpy is required for BSP calculation.")`: only when numpy is missing (`NUMPY_AVAILABLE` False). This is checked first and is not practically testable.
- A negative `grid_size` or `detector_pixels` raises numpy's `ValueError` from `linspace` ("Number of samples, -1, must be non-negative."). This is only reached when the system has elements.
- No other explicit raises. Tracing errors from `tracer_3d` or `paraxial` would propagate; that code is not visible here.

## 4. Returns and guarantees
- Empty system (`not system.elements`): returns the tuple `(np.array([]), np.array([]), np.array([]))`.
  - Each array has shape `(0,)`.
  - This path does no tracing and no image-plane lookup.
- Otherwise it returns the tuple `(DY, DZ, Intensity)`, built from `DY, DZ = np.meshgrid(det_y, det_z)`:
  - All three arrays have shape `(detector_pixels, detector_pixels)`.
  - `DY[i, j] == det_y[j]`: values vary along axis 1.
  - `DZ[i, j] == det_z[i]`: values vary along axis 0.
  - `DY[0,0] == -detector_size/2` and `DY[0,-1] == detector_size/2`.
  - `Intensity = np.abs(E_field)**2` is a real float64 array with values `>= 0`.
- With no surviving beamlets, `Intensity` is exactly all `0.0`.
- Per-beamlet details:
  - Each beamlet starts at `x = elements[0].position - 20`, at `(py, pz)`, travelling along `+x`.
  - It starts with `q0 = 1j*zR`.
  - Beamlets whose ray is terminated (front or back surface not `REFRACTED`) are dropped.
  - Beamlets with `|direction.x| < 1e-6` are skipped.
  - Each remaining beamlet is propagated by `t = (image_plane_x - origin.x)/direction.x`; `t` may be negative.
  - Its field is `(w0/wx)·exp(-ly²/wx² - lz²/wy²)·exp(-1j·k·(ly²/(2Rx)+lz²/(2Ry)+OPL))`.
- New arrays are created on every call, and the call is deterministic: the same inputs give an identical result.

## 5. Side effects
- New `LensRayTracer3D` objects are built per call, from the global node transforms when `system.root` has one flat element node per element, otherwise from `x_offset=element.position`.
- Beamlet rays are local objects, so the system is not mutated as far as this module shows.
- Logs at WARNING on logger `openlens.analysis.beam_synthesis`:
  - `"Beam synthesis: ignoring non-finite image plane %r"` when the configured plane is unusable.
  - `"Beam synthesis: no paraxial focus (afocal or untraceable system); placing the detector %.1f mm past the last surface at x=%.3f"` when the focus search fails.

## 6. Branches and edge cases
- `OpticalSystem()` with no lenses returns three empty arrays (§4).
- `grid_size=1`: `linspace` returns `[-r]`, and the point `(-r,-r)` lies outside the circle, so there are no beamlets and `Intensity` is all zeros. The image plane is still resolved.
- `grid_size=0` also gives all zeros.
- `detector_pixels=0` gives empty `(0,0)` arrays.
- `detector_pixels=1` gives `DY == [[-size/2]]`.
- Image-plane resolution when `image_plane_x is None`, in order:
  1. `getattr(system, "image_plane_x_mm", None)`. It is not visible here whether `OpticalSystem` defines this attribute. If the value is not None, it is converted with `float()`. If that succeeds and the result is finite, it is used.
  2. If the value is non-convertible (e.g. `"abc"`) or non-finite (`inf`, `nan`), the warning is logged and resolution falls through to step 3.
  3. `find_paraxial_focus(system, wavelength_mm/NM_TO_MM, tracer=self.tracer)`. A finite result is used.
  4. If the focus is None or non-finite, the plane is `system.get_total_length() + 50.0` (`DEFAULT_IMAGE_PLANE_PAST_LAST_SURFACE`), and the warning is logged.
- An explicit `image_plane_x` skips all of the above, including the warnings.

## 7. Rules from comments and docstrings
- Beamlet spacing is `diameter/(grid_size-1)`, deliberately not `/grid_size`.
- The default detector sits at the paraxial focus, the same plane `DiffractionPSF` uses. It is not "50 mm past the last vertex" unless the system is afocal or untraceable.
- `q` is advanced by `distance/n` inside glass.
- Refraction of a beamlet's `q`:
  - Front surface: from the beam's current `n` (air) into `lens.refractive_index`, using `R1`.
  - Back surface: from `lens.refractive_index` into 1.0, using `R2`.
  - After each refraction the beam's wavelength is rescaled by `n1/n2`.

## 8. Testing hints
Build a system:
```python

from openlens.analysis.beam_synthesis import BeamSynthesisPropagator

s.add_lens(Lens(diameter=25.0))
p = BeamSynthesisPropagator(s)
DY, DZ, I = p.propagate_to_image(grid_size=8, detector_pixels=16)
```
- Keep `grid_size` small (6–10) so tests run fast.
- Check `I.shape == (16,16)`, `np.isrealobj(I)`, `(I >= 0).all()`, and `np.isfinite(I).all()`.
- Check the grids with `np.allclose(DY[0], np.linspace(-0.05, 0.05, 16))` and `np.allclose(DZ[:,0], same)`.
- Check `grid_size=1` gives `I.sum() == 0` (exact), and that an empty system gives all `.size == 0`.
- Check the default plane equals the paraxial focus:
  - Compare `propagate_to_image(grid_size=6)` against an explicit `image_plane_x=find_paraxial_focus(s)` (from `openlens.analysis.paraxial`) using `np.allclose`.
  - Use `allclose`, not exact equality: `WAVELENGTH_GREEN*NM_TO_MM/NM_TO_MM` may differ in the last bit.
- Check the fallback plane with `monkeypatch.setattr(bs, "find_paraxial_focus", lambda *a, **k: None)`:
  - The default call should equal `image_plane_x=s.get_total_length()+50.0`, using `np.array_equal` or `allclose`.
  - `caplog` should contain `"no paraxial focus"`.
  - Repeat with the patch returning `float("nan")`.
- Check the configured plane by setting `s.image_plane_x_mm = <focus value>`:
  - The default call should equal the explicit call with the same x.
  - With `s.image_plane_x_mm = "abc"` or `float("inf")`, the call should fall back, and `caplog` should contain `"ignoring non-finite image plane"`.
  - Whether attribute assignment works on `OpticalSystem` is not visible here; prefer the monkeypatch tests.
- Determinism: two identical calls should give `np.array_equal` intensities.
- Changing `image_plane_x` far from focus (e.g. focus ± 20 mm) should change the intensity: assert `not np.allclose(...)`.
- Pitfalls:
  - Do not assert exact peak locations or values.
  - Do not assert exact y/z symmetry; use a loose tolerance if you check it.
  - Do not pass `wavelength_mm` in nm.

### openlens.analysis.psf_mtf.ImageQualityAnalyzer.calculate_mtf

# Notes: `ImageQualityAnalyzer.calculate_mtf` (openlens/analysis/psf_mtf.py)

## 1. Purpose
- Returns the tangential and sagittal MTF as 1-D arrays against spatial frequency (lp/mm).
- The default is geometric: a ray-binned PSF, then the line spread function (LSF), then an FFT.
- With `use_diffraction=True` it takes the FFT of the pupil-function PSF instead.

## 2. Inputs
- `field_angle_deg` (float, 0.0): field angle in Y. It is passed to `SpotDiagram.trace_spot(field_angle_y_deg=...)`, or to the wavefront sensor.
- `wavelength_nm` (float, `WAVELENGTH_GREEN`).
- `focus_shift_mm` (float, 0.0): defocus.
- `max_freq` (float, 100.0): upper frequency cut, inclusive (`<=`).
- `use_diffraction` (bool, False).
- `**kwargs`: legacy aliases only.
  - `field_angle=` is popped and **overrides** `field_angle_deg`.
  - The `"max_freq"` alias branch is dead code: `max_freq=` always binds to the named parameter.
  - Any other key raises `TypeError`.
- No type or range validation is done on any argument.

## 3. Raises
- `TypeError(f"Unexpected kwargs: {list(kwargs.keys())}")`.
  - Example: `an.calculate_mtf(foo=1)` raises `TypeError: Unexpected kwargs: ['foo']`.
  - Order of checks: aliases are popped first, then this check runs, and only then the empty-system check.
  - So the error is raised even on an empty `OpticalSystem()`.
  - `calculate_mtf(field_angle=1, foo=2)` gives the message `Unexpected kwargs: ['foo']`.
- Anything else would come from other modules: the tracer, `WavefrontSensor`, `DiffractionPSFCalculator`.

## 4. Returns
- Always a new `dict` with exactly the keys `{"freq", "mtf_tan", "mtf_sag"}`.
- All three values are numpy arrays of the same length.
- **Empty system** (`not system.elements`): three empty float64 arrays, `shape == (0,)`.
  - Logs a warning: `"MTF: system has no elements; returning an empty MTF"`.
  - This holds for both values of `use_diffraction`.
- **Geometric branch** (default):
  1. It traces the spot with `num_rings=25` and reads `rms_radius`.
  2. Window size:
     - `window = max(0.256, 2*2.2*rms)` = `max(0.256, 4.4*rms)` when `rms` is finite and > 0.
     - Otherwise `window = 0.256`.
  3. Pixel count: `num_pixels = min(max(128, ceil(window/0.002)), 1024)`.
  4. It calls `self.calculate_psf(..., sensor_size_mm=window, pixels=num_pixels)`.
     - That function caps its own window at 8.0 mm.
     - So `step_size = min(window, 8.0)/num_pixels`.
  5. LSFs: `lsf_tan = psf.sum(axis=1)` (profile along Y) and `lsf_sag = psf.sum(axis=0)` (profile along Z).
  6. `mtf = |fft(lsf)|`, divided by `mtf[0]` only if `mtf[0] > 0`. An all-zero PSF (no rays) gives an all-zero MTF, not NaN.
  7. Frequencies: `freqs = fftfreq(num_pixels, d=step_size)`. The first `num_pixels//2` entries are kept, then masked to `0 <= f <= max_freq`.
- **Geometric guarantees**:
  - `freq[0] == 0.0`.
  - `mtf_tan[0] == mtf_sag[0] == 1.0` when any rays land.
  - Frequencies are strictly increasing with uniform step `df = 1/(num_pixels*step_size)`. For a small spot (rms ≤ 0.256/4.4 ≈ 0.0582 mm) that is ≈ 1/0.256 = 3.90625 lp/mm.
  - With the default `max_freq=100` and a small spot, the length is 26 (k = 0..25, last value ≈ 97.656).
  - All MTF values lie in [0, 1] (up to float tolerance), because the LSF is non-negative.
  - Frequency is capped at index `num_pixels//2 - 1`. For 128 px that is 63 × 3.90625 ≈ 246 lp/mm, so `max_freq=1000` still returns only ~64 entries.
- **Diffraction branch**:
  - Gets the wavefront with `grid_size=64`. If it is degenerate (`W` is None, not 2-D, or a side < 2), it returns three empty arrays and logs a warning.
  - Otherwise it adds defocus to the wavefront in place. This happens only if `focus_shift_mm` is truthy, with `W += dz*NA²*p²/(2*λ_mm)` and `NA = D1/(2|efl|)`.
  - It then builds the PSF with `pad_factor=4`, then the 2-D MTF (both from another module, not visible).
  - It slices from the centre outwards: `mtf_tan = mtf2d[cy:, cx]` and `mtf_sag = mtf2d[cy, cx:]`.
  - Frequency step: `df = (D1/(N-1)) / (wavelength_nm*1e-6*efl)`.
    - `D1` is the first element's lens diameter; `N = W.shape[0]`.
    - `efl = system.get_system_focal_length() or 100.0`.
  - `freq = arange(len)*df`, masked by `freq <= max_freq` only. There is no `>= 0` check here, so a negative EFL gives negative frequencies, all of which are kept.
  - `freq[0] == 0`.

## 5. Side effects
- The system and its lenses are not mutated.
- Only internal temporaries (the wavefront object) are modified.
- Logging only; no files are written.

## 6. Branches and edge cases
- `OpticalSystem()` with no lenses returns empty arrays (`len == 0`).
- `max_freq=-1` (geometric, real lens) returns three empty arrays, since 0 > -1.
- `max_freq=0` returns `freq == [0.]` and `mtf_tan == mtf_sag == [1.]`.
- Large `focus_shift_mm` gives a larger rms. The window becomes > 0.256, so `df` gets smaller and there are more entries below `max_freq`. MTF at non-zero frequencies drops.
- `field_angle=x` (alias) gives the same result as `field_angle_deg=x`.

## 7. Rules from comments
- The geometric window must scale with the spot (2.2 × RMS radius each side) so that light is not clipped. The floor is 0.256 mm.
- Sampling is ~2 µm per pixel, 128 to 1024 pixels.
- The diffraction branch must honour `focus_shift_mm`. A bug was fixed here: the MTF used to be identical at every defocus.
- Pupil spacing uses `N-1` intervals, not `N`.

## 8. Testing hints
- Setup:
  ```python
  from openlens.lens import Lens
  from openlens.optical_system import OpticalSystem
  from openlens.analysis.psf_mtf import ImageQualityAnalyzer
  s = OpticalSystem(); s.add_lens(Lens())   # default biconvex BK7
  an = ImageQualityAnalyzer(s)
  r = an.calculate_mtf()
  ```
- Safe geometric assertions:
  - Keys equal the set above.
  - Equal lengths.
  - `r["freq"][0] == 0`, and the array is non-decreasing.
  - `max(freq) <= max_freq`.
  - `np.isclose(r["mtf_tan"][0], 1.0)`.
  - `0 <= mtf <= 1 + 1e-9`.
  - Uniform spacing via `np.allclose(np.diff(freq), freq[1])`.
- To check `df` exactly, recompute it rather than hard-coding a value:
  - `p = an.calculate_psf(sensor_size_mm=W, pixels=N)` with the same `W` and `N` formula.
  - Then `np.allclose(r["freq"], np.fft.fftfreq(N, p["step_size"])[:N//2][mask])`.
  - `ceil(0.256/0.002)` may be 128 or 129 depending on float rounding; do not assume which.
- Pick `max_freq` small (e.g. 10 or 20) so the result does not depend on the cap.
- Compare runs with `np.allclose`; ray tracing is deterministic for the same inputs.
- Defocus test (geometric): `an.calculate_mtf(focus_shift_mm=2.0)["mtf_tan"][1:]` is generally lower than in focus. Prefer comparing a mean over the first few frequencies, and do not assert exact values.
- Diffraction branch: assert only structure — keys, equal lengths, `freq[0] == 0`, `freq <= max_freq`. Its values depend on unseen modules.
- Empty system: assert `r["freq"].size == 0` for both `use_diffraction` values.
- Warnings can be checked with `caplog`; the logger name is `openlens.analysis.psf_mtf`.

### openlens.analysis.psf_mtf.ImageQualityAnalyzer.simulate_image

# Notes: `ImageQualityAnalyzer.simulate_image` (openlens/analysis/psf_mtf.py)

## 1. Purpose
- Blurs an image by convolving each channel with the system's **diffraction** PSF at that channel's wavelength.
- Returns a float64 array with the same shape as the input, clipped to [0, 1].

## 2. Inputs
- `image_array`: must have a `.shape` attribute **before** conversion, because `np.zeros(image_array.shape, dtype=float64)` runs before `np.asarray`.
  - After `np.asarray`, it must be 2-D `(H, W)` or 3-D `(H, W, C)`.
  - Integer dtypes are divided by `np.iinfo(dtype).max`, so uint8 255 becomes 1.0.
  - Every other dtype (float, bool) is cast to float64 unchanged.
- `pixel_size_mm=0.005`: used to size the PSF kernel.
  - `psf_size_mm = max(0.2, pixel_size_mm*64)`
  - `psf_pixels = int(psf_size_mm / pixel_size_mm)`; if even, `+1`, so it is always odd.
- `wavelengths_nm=(WAVELENGTH_C_LINE, WAVELENGTH_D_LINE, WAVELENGTH_F_LINE)`: any sequence. Import the constants from `openlens.constants`; do not hard-code the values.
- `field_angle_deg=0.0` and `focus_shift_mm=0.0`: passed straight through to `calculate_psf`.
- `**kwargs`, legacy aliases:
  - `pixel_size` overrides `pixel_size_mm`.
  - `wavelengths` overrides `wavelengths_nm`.
  - Any other key raises `TypeError`.

## 3. Raises (in this order)
1. `TypeError(f"Unexpected kwargs: {list(kwargs.keys())}")`.
   - `sim(img, foo=1)` → `"Unexpected kwargs: ['foo']"`.
   - Aliases are popped first, so `sim(img, pixel_size=0.01, foo=1)` → `"Unexpected kwargs: ['foo']"`.
2. `AttributeError` for a plain list such as `sim([[0.1]])`, because the list has no `.shape`. This is not a ValueError.
3. `ValueError(f"image_array must be 2-D (H, W) or 3-D (H, W, C); got shape {image_array.shape}")`.
   - `np.zeros(5)` → `"... got shape (5,)"`.
   - `np.array(1.0)` → `"... got shape ()"`.
   - `np.zeros((2,2,2,2))` → `"... got shape (2, 2, 2, 2)"`.
4. `ValueError(f"image_array has {C} channels but {len(wavelengths_nm)} wavelengths were given")`.
   - `np.zeros((4,4,2))` with the default 3 wavelengths → `"image_array has 2 channels but 3 wavelengths were given"`.
   - `np.zeros((4,4,3))` with `wavelengths=(550.0,)` → `"image_array has 3 channels but 1 wavelengths were given"`.
- All four checks happen before any PSF is computed, so they fire even with an empty `OpticalSystem()`.
- If a mocked `calculate_psf` returns an **integer** kernel with a positive sum, `kernel /= kernel_sum` raises numpy's casting error. Mocks should return float arrays.

## 4. Returns and guarantees
- Returns `np.clip(output_image, 0, 1)`: a new float64 ndarray with `shape == input.shape`.
- For each wavelength `wl` at index `i`, it calls:
  ```
  self.calculate_psf(field_angle_deg=..., wavelength_nm=wl, focus_shift_mm=...,
                     sensor_size_mm=psf_pixels*pixel_size_mm, pixels=psf_pixels,
                     use_diffraction=True)
  ```
  - There is exactly one call per wavelength, in order, including for grayscale input.
- Kernel handling: `kernel = psf_data["image"]`. If `sum > 0` it is normalised in place to sum 1; if `sum == 0` it is left unchanged, so convolution gives zeros.
- Convolution is `scipy.signal.fftconvolve(channel, kernel, mode="same")`. If scipy is missing, a numpy FFT fallback with the same "same" cropping is used.
- Grayscale (2-D) output is the **mean** of the per-wavelength convolutions.
- 3-D output: channel `i` is the input channel `i` convolved with kernel `i`.
- Clipping: values above 1 become 1 and values below 0 become 0 (e.g. float input 2.0 → 1.0, -0.5 → 0.0).

## 5. Side effects
- The input array is not mutated (`astype` makes a copy).
- The PSF kernel object returned by `calculate_psf` **is** divided in place. A mock that returns a shared array will see it modified.
- Logging only; no files written.

## 6. Branches and edge cases
- **Empty system:** `ImageQualityAnalyzer(OpticalSystem())`. `calculate_psf` returns all-zero images (its empty-system branch runs before the diffraction branch). Every kernel is zero and not normalised, so the result is `np.zeros(shape)`.
  - Example: `sim(np.ones((8,8)))` → all 0.0, float64, shape (8,8).
  - The same holds for uint8 and for 3-D input.
- **Integer vs float input:** with an identity kernel, `np.array([[0,255],[51,102]], uint8)` → `[[0,1],[0.2,0.4]]`.
- **Grayscale with mixed kernels:** mock an identity kernel for one wavelength and zero kernels for the other two. The output is `input/3`.
- **3-D:** mock an identity kernel for wavelength 0 and zero kernels otherwise. Output channel 0 equals the input; channels 1 and 2 are zero.
- **Non-normalised kernel:** `[[0,0,0],[0,4.0,0],[0,0,0]]` is scaled to sum 1, so it acts as identity (output is not 4× the input).
- **Alias check:** `wavelengths=(500.0, 600.0)` with a 2-channel image is accepted. Check the call list `[500.0, 600.0]` via the mock.

## 7. Rules from comments and docstring
- Output is always float64 and normalised 0–1. The old `zeros_like` dtype bug truncated uint8 results.
- 2-D input must give 2-D output, with no invented channel axis.
- Integer input is scaled by its dtype max so the clip does not saturate the image.
- The diffraction PSF is used deliberately, because geometric PSFs are sparse.

## 8. Testing hints
- **Recommended approach:** build `iqa = ImageQualityAnalyzer(OpticalSystem())`, then monkeypatch the instance:
  ```python
  calls = []
  def fake(**kw): calls.append(kw); k = np.zeros((3,3)); k[1,1] = 1.0; return {"image": k}
  monkeypatch.setattr(iqa, "calculate_psf", fake)
  ```
  - Return a **fresh float array** on every call.
  - Assert `use_diffraction is True` in each recorded call.
  - Assert `kw["pixels"] % 2 == 1` and `kw["sensor_size_mm"] == kw["pixels"] * pixel_size`.
- Do **not** pin the exact `pixels` value. `0.32/0.005` may floor to 63 or 64, giving 63 or 65.
- Compare convolution results with `np.allclose(..., atol=1e-9)`, because FFT noise means outputs are not exact.
- Check `result.dtype == np.float64`, `result.shape == input.shape`, and that `result is not input`.
- **Real lens:** the full diffraction pipeline (WavefrontSensor, DiffractionPSFCalculator) depends on other modules not shown here. Only assert shape, dtype, `0 <= result <= 1`, and finiteness; keep images small, e.g. 16×16.
- **Exceptions:** use `pytest.raises(ValueError, match=re.escape(msg))` with the exact text above.

### openlens.vector3.Vector3.normalize

# Notes: `openlens.vector3.Vector3.normalize`

## 1. Purpose
- Returns a **new** unit-length `Vector3` pointing the same way as `self`.
- If the magnitude is below `1e-15`, it returns the zero vector instead of dividing.

## 2. Inputs
- `self` only. Components `x`, `y`, `z` are stored exactly as passed to `Vector3(x, y, z)`. The constructor does no validation or conversion, so ints, floats and numpy scalars are all kept unchanged.
- How the magnitude is computed:
  - `magnitude()` is `math.sqrt(self.x**2 + self.y**2 + self.z**2)`.
  - That is, `math.sqrt(magnitude_sq())`.

## 3. Raises
None of these are raised explicitly; they come from the arithmetic.
- **`TypeError`** for non-numeric components, raised by `**`.
  - Example: `Vector3("a", 0, 0).normalize()`.
- **`OverflowError`** (`(34, 'Numerical result out of range')`) for huge Python float components, raised by float `**`.
  - Example: `Vector3(1e200, 0, 0).normalize()`.
  - It does not return inf or 0. This is CPython float-pow behaviour.

## 4. Returns and guarantees
The code is exactly:
```python
mag = self.magnitude()
if mag < 1e-15:
    return Vector3(0, 0, 0)      # int zeros!
return self / mag                # Vector3(x/mag, y/mag, z/mag)
```
- The result is always a new `Vector3` object, never `self`. This holds even for the zero case and for vectors that are already unit length.
- **Normal path:** each component is `component / mag`, using Python true division, so the results are floats.
  - The direction and the signs of the components are preserved.
  - `magnitude()` of the result is ≈ 1.0. Use `pytest.approx` for this check.
- **Degenerate path:** returns `Vector3(0, 0, 0)` with **int** zeros.
  - `to_tuple()` gives `(0, 0, 0)`.
  - `type(result.x) is int`.

## 5. Side effects
- `self` is not mutated. Its `x`, `y`, `z` are unchanged after the call.
- There is no global state.

## 6. Branches and edge cases
| Input | Result `to_tuple()` |
|---|---|
| `Vector3(3, 4, 0)` | `(0.6, 0.8, 0.0)`. Exact, because `sqrt(25) == 5.0` |
| `Vector3(0, -2, 0)` | `(0.0, -1.0, 0.0)` |
| `Vector3(0, 0, 5.0)` | `(0.0, 0.0, 1.0)` |
| `Vector3(1, 1, 1)` | each component ≈ `1/sqrt(3)` ≈ 0.5773502691896258 |
| `Vector3(1.0, 0, 0)` (already unit) | `(1.0, 0.0, 0.0)`, as a new object |
| `Vector3()` / `Vector3(0, 0, 0)` | `(0, 0, 0)`, ints |
| `Vector3(1e-16, 0, 0)` | mag 1e-16 < 1e-15, so `(0, 0, 0)` |
| `Vector3(1e-16, 1e-16, 1e-16)` | mag ≈ 1.73e-16, so `(0, 0, 0)` |
| `Vector3(1e-200, 0, 0)` | the square underflows to 0.0, so `(0, 0, 0)` |
| `Vector3(1e-14, 0, 0)` | mag 1e-14 ≥ threshold, so x ≈ 1.0 (approx). Small but nonzero vectors are normalized |
| `Vector3(1e-10, 2e-10, 0)` | ≈ `(1/√5, 2/√5, 0.0)` |
| `Vector3(float('nan'), 0, 0)` | `nan < 1e-15` is False, so the vector is divided and all components become nan |
| `Vector3(1e200, 0, 0)` | `OverflowError` |

- The threshold uses strict `<`. Do not test at exactly `1e-15`, because the rounding of `sqrt` makes that boundary unreliable. Use `1e-16` for the zero branch and `1e-14` for the normalize branch.

## 7. Rules from comments/docstrings
- There are none beyond the class docstring "3D Vector class with basic operations".

## 8. Testing hints
- **No `__eq__` exists.** `v.normalize() == Vector3(1, 0, 0)` is False because it compares identity.
  - Compare with `v.to_tuple()` or with `.x`, `.y`, `.z`.
  - Use `pytest.approx` for non-exact values, e.g. `assert r.to_tuple() == pytest.approx((0.6, 0.8, 0.0))`.
- Exact equality is safe for the Pythagorean-style cases above (3-4-5, single-axis).
- **Check the "new object" guarantee:** `r is not v`. Also confirm `v.to_tuple()` is unchanged after the call, e.g. it stays `(3, 4, 0)`.
- **The zero branch returns ints.** `r.to_tuple() == (0, 0, 0)` passes, and `0 == 0.0` is also True. To pin the exact branch, assert `type(r.x) is int`.
  - The normal path gives floats even for int input. `Vector3(0, 0, 5).normalize().x` is `0.0` and its type is float.
- **Pin the threshold** with a pair of tests on either side, so a regression in it (e.g. to `1e-12` or `<=`) is caught:
  - `Vector3(1e-14, 0, 0).normalize().x == pytest.approx(1.0)`.
  - `Vector3(1e-16, 0, 0).normalize().to_tuple() == (0, 0, 0)`.
  - A threshold change to `1e-12` would make the `1e-14` case return `(0, 0, 0)`.
- **Scale invariance:** `Vector3(6, 8, 0)` and `Vector3(0.3, 0.4, 0)` both give ≈ `(0.6, 0.8, 0)`.
- **Sign preservation:** `Vector3(-3, 0, -4)` gives `(-0.6, 0.0, -0.8)`. The y component is `0/5.0 == 0.0`.
- **Idempotence:** `v.normalize().normalize()` gives approximately the same components.
- `Vector3` uses `__slots__`, so you cannot set extra attributes on instances.
- `vec3(x, y, z)` is an equivalent constructor.
