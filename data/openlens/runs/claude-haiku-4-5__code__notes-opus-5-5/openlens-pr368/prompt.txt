# openlens.analysis.spot_diagram.SpotDiagram.trace_spot
    def trace_spot(
        self,
        field_angle_x_deg: float = 0.0,
        field_angle_y_deg: float = 0.0,
        wavelength_nm: float = WAVELENGTH_GREEN,
        image_plane_x_mm: Optional[float] = None,
        num_rings: int = 6,
        focus_shift_mm: float = 0.0,
    ) -> Dict[str, Any]:
        """
        Trace rays to generate a spot diagram.

        Args:
            field_angle_x_deg: Field angle in degrees (X-Z plane tilt)
            field_angle_y_deg: Angle in degrees relative to optical axis in XY plane.
            wavelength_nm: Wavelength in nm.
            image_plane_x_mm: Absolute X position of the image plane. If None,
                uses the system's configured plane, else the paraxial focus.
            num_rings: Sampling density.
            focus_shift_mm: Offset from the nominal image plane (defocus).

        Returns:
            Dictionary containing spot data and statistics.
        """
        # Convert wavelength to mm
        wl_mm = wavelength_nm * NM_TO_MM

        # Store original states and update lenses for this wavelength
        original_states = []
        for element in self.system.elements:
            lens = element.lens
            original_states.append((lens, lens.wavelength, lens.refractive_index))
            lens.update_refractive_index(wavelength_nm=wavelength_nm)

        target_x = 0.0
        # Hoisted above the try: the statistics block below reads them, and
        # leaving the initialisation inside the try made any failure before
        # that point surface as UnboundLocalError instead of its real cause.
        spot_points = []
        valid_rays = 0

        try:
            # Determine image plane position. A stored detector plane wins;
            # otherwise the shared paraxial-focus search (#460) places it at
            # the focus, or well behind a collimated system.
            if image_plane_x_mm is None:
                image_plane_x_mm = resolve_image_plane_x(
                    self.system, wavelength_nm, tracer=self.tracer
                )
                if image_plane_x_mm is None:
                    image_plane_x_mm = self.system.get_total_length() + 100

            target_x = image_plane_x_mm + focus_shift_mm

            # Generate pupil points
            pupil_points = self.generate_hexapolar_grid(rings=num_rings)

            # Calculate ray direction based on field angles
            # Tan(theta)
            tan_y = math.tan(math.radians(field_angle_y_deg))
            tan_z = math.tan(math.radians(field_angle_x_deg))

            # Direction vector (normalized later by Ray3D)
            # dx=1, dy=tan_y, dz=tan_z
            direction = vec3(1.0, tan_y, tan_z).normalize()

            # Start rays before first element
            start_x = self.system.elements[0].position - 50.0

            first_lens_x = self.system.elements[0].position

            for py, pz in pupil_points:
                # Launch ray from pupil coordinate projected back to start_x
                # Back-propagate from aperture to start position
                dist = first_lens_x - start_x

                t = dist / direction.x
                origin = vec3(first_lens_x, py, pz) - direction * t

                ray = Ray3D(origin, direction, wavelength=wl_mm)
                self.tracer.trace_ray(ray)

                if not ray.terminated:
                    # Find intersection with image plane X = target_x
                    if abs(ray.direction.x) > 1e-6:
                        t = (target_x - ray.origin.x) / ray.direction.x

                        spot_y = ray.origin.y + t * ray.direction.y
                        spot_z = ray.origin.z + t * ray.direction.z

                        spot_points.append((spot_y, spot_z))
                        valid_rays += 1

        finally:
            # Restore original states
            for lens, wl, n in original_states:
                lens.wavelength = wl
                lens.refractive_index = n

        # Reached only when the try body completed without raising. This check
        # must NOT live in finally: a return there discards any in-flight
        # exception, so a tracer failure that happened before the first spot
        # point landed was reported as a clean rms_radius of 0.0 - a tracer bug
        # masquerading as an optical result - instead of propagating.
        if valid_rays < 2:
            # Fewer than two rays cannot define a spot: the RMS radius would be
            # the distance of a single ray from its own centroid, i.e. exactly
            # 0.0. Reporting that as a perfect spot lets a fully vignetted
            # system outscore every physically real design in the optimizer
            # and score 100% production yield in Monte Carlo. None forces
            # callers to treat it as a failure instead of a great number.
            logger.warning(
                "Only %d ray(s) reached the image plane (blocked or TIR); "
                "spot statistics are undefined",
                valid_rays,
            )
            return {
                "rms_radius": None,
                "geo_radius": None,
                "centroid": (0.0, 0.0),
                "points": spot_points,
                "valid_rays": valid_rays,
                "image_plane_x": target_x,
                "error": "Fewer than 2 rays reached the image plane",
            }

        # Centroid
        sum_y = sum(p[0] for p in spot_points)
        sum_z = sum(p[1] for p in spot_points)
        cent_y = sum_y / valid_rays
        cent_z = sum_z / valid_rays

        # RMS Radius (relative to centroid)
        sum_sq_dist = 0.0
        max_dist_sq = 0.0

        for y, z in spot_points:
            dy = y - cent_y
            dz = z - cent_z
            dist_sq = dy * dy + dz * dz
            sum_sq_dist += dist_sq
            if dist_sq > max_dist_sq:
                max_dist_sq = dist_sq

        rms_radius = math.sqrt(sum_sq_dist / valid_rays)
        geo_radius = math.sqrt(max_dist_sq)

        return {
            "rms_radius": rms_radius,  # mm
            "geo_radius": geo_radius,  # mm
            "centroid": (cent_y, cent_z),  # mm
            "points": spot_points,
            "valid_rays": valid_rays,
            "image_plane_x": target_x,
        }

# openlens.tolerancing.InverseSensitivityAnalyzer.calculate_sensitivities
    def calculate_sensitivities(self, criterion: str = "rms_spot_radius") -> List[Dict[str, Any]]:
        """
        Calculate sensitivity of each operand.

        Args:
            criterion: Performance metric. Only ``"rms_spot_radius"`` is
                implemented, the same set MonteCarloAnalyzer accepts.

        Returns:
            List of dicts with 'operand', 'sensitivity', 'change'.

        Raises:
            ValueError: If *criterion* is unsupported. The argument used to be
                ignored, so a caller asking for another metric silently got
                RMS-spot sensitivities and budgeted tolerances against the
                wrong performance measure.
        """
        if criterion != self.SUPPORTED_CRITERIA:
            raise ValueError(
                f"Unsupported criterion {criterion!r}; " f"supported: {self.SUPPORTED_CRITERIA!r}"
            )

        results = []

        # Nominal performance
        spot_nom = SpotDiagram(self.system)
        res_nom = spot_nom.trace_spot()
        # A vignetted nominal reports None; inf keeps the sensitivity
        # arithmetic below well-defined instead of raising on None.
        nominal_val = res_nom["rms_radius"]
        if nominal_val is None:
            nominal_val = float("inf")

        for i, tol in enumerate(self.tolerances):
            # Test at max value
            # Apply single tolerance at max value
            delta = tol.max_val

            # Save state
            original_state = self._get_system_state(self.system)

            # The perturbation is applied in place, so the restore has to
            # happen even if the apply or the trace raises - otherwise the
            # design is left perturbed and the next save writes that to the
            # database.
            try:
                self._apply_single_tolerance(self.system, tol, delta)

                # Analyze
                spot = SpotDiagram(self.system)
                res = spot.trace_spot()
                val = res["rms_radius"]
                if val is None:
                    val = float("inf")
            except Exception as e:
                logger.warning("Inverse sensitivity for tolerance %d failed: %s", i, e)
                val = float("inf")
            finally:
                # Restore state
                self._set_system_state(self.system, original_state)

            change = val - nominal_val
            # Sensitivity = change / tolerance_value. When either end is a
            # vignetted inf this is inf - inf, i.e. undefined: the operand
            # never moved the design between two unmanufacturable states, so
            # there is no sensitivity to report. NaN says "undefined" without
            # pretending the operand is infinitely sensitive.
            sensitivity = change / delta if delta != 0 else 0
            if not (math.isfinite(change) and math.isfinite(sensitivity)):
                change = float("nan")
                sensitivity = float("nan")

            results.append(
                {
                    "operand_index": i,
                    "type": tol.param_type.name,
                    "element": tol.element_index,
                    "test_value": delta,
                    "result_value": val,
                    "change": change,
                    "sensitivity": sensitivity,
                }
            )

        return results

### openlens.analysis.spot_diagram.SpotDiagram.trace_spot

# Notes: `SpotDiagram.trace_spot` (openlens/analysis/spot_diagram.py)

## 1. Purpose
- Traces a hexapolar bundle of parallel rays through `self.system` at a given field angle and wavelength.
- Intersects each surviving ray with the plane `X = target_x`.
- Returns the spot points plus centroid, RMS radius and geometric (max) radius.

## 2. Inputs
- `field_angle_x_deg=0.0`: sets the **z** component of the direction (`tan_z = tan(radians(field_angle_x_deg))`). The name is misleading.
- `field_angle_y_deg=0.0`: sets the **y** component (`tan_y`).
- The direction is `vec3(1.0, tan_y, tan_z).normalize()`.
- `wavelength_nm=WAVELENGTH_GREEN`:
  - Converted with `wl_mm = wavelength_nm * NM_TO_MM` and used as the ray wavelength.
  - Every lens gets `lens.update_refractive_index(wavelength_nm=wavelength_nm)` for the duration of the call.
- `image_plane_x_mm=None`:
  - If `None`, calls `resolve_image_plane_x(system, wavelength_nm, tracer=self.tracer)` from `analysis.paraxial` (not shown). It returns the configured detector plane, else the paraxial focus.
  - If that returns `None`, falls back to `system.get_total_length() + 100`.
  - An explicit value is used as-is, with no validation.
- `num_rings=6`: number of hexapolar rings, passed to `generate_hexapolar_grid(rings=num_rings)`.
  - The grid has `1 + 3*n*(n+1)` points: 127 for 6, 7 for 1, 1 for 0.
  - Ring `i` has `6*i` points at radius `(D/2)*(i/n)`.
  - `D` is `system.elements[0].lens.diameter`, the full diameter, not the clear aperture.
- `focus_shift_mm=0.0`: `target_x = image_plane_x_mm + focus_shift_mm`.

## 3. Raises
- **Empty system** raises `IndexError` (from `self.system.elements[0]`, message `list index out of range`).
  - Reliable trigger: `SpotDiagram(OpticalSystem()).trace_spot(image_plane_x_mm=100.0)`.
  - With `image_plane_x_mm=None`, `resolve_image_plane_x` runs first. Its behaviour on an empty system is not visible, so pass an explicit plane.
- Any exception from the tracer or the lens propagates; it is not swallowed. Lens state is still restored (see §5).
- There is no validation of the arguments.

## 4. Returns and guarantees
The result is always a new `dict`.

**Success** (`valid_rays >= 2`) has exactly 6 keys:
- `"rms_radius"`: `sqrt(sum(dy²+dz²)/valid_rays)`, measured from the centroid.
- `"geo_radius"`: `sqrt(max(dy²+dz²))`. Always `geo_radius >= rms_radius >= 0`.
- `"centroid"`: tuple `(mean_y, mean_z)`.
- `"points"`: list of `(spot_y, spot_z)` tuples, in pupil-grid order (centre first, then ring by ring, angle increasing).
- `"valid_rays"`: an int, `== len(points)`.
- `"image_plane_x"`: `target_x`.
- There is no `"error"` key.

Each point is computed as `t = (target_x - ray.origin.x)/ray.direction.x`, then `(origin.y + t*dir.y, origin.z + t*dir.z)`.

**Failure** (`valid_rays < 2`) has 7 keys:
```python
{"rms_radius": None, "geo_radius": None, "centroid": (0.0, 0.0),
 "points": spot_points, "valid_rays": valid_rays,
 "image_plane_x": target_x, "error": "Fewer than 2 rays reached the image plane"}
```
- It also logs a WARNING on logger `openlens.analysis.spot_diagram`.
- The message is `"Only %d ray(s) reached the image plane (blocked or TIR); spot statistics are undefined"`, for example `Only 1 ray(s) ...`.

## 5. Side effects
- During the call, each lens's refractive index is updated for `wavelength_nm`.
- In `finally`, each lens gets `lens.wavelength` and `lens.refractive_index` set back to their original values. This happens even on an exception.
- After the call, `lens.refractive_index` and `lens.wavelength` equal their pre-call values, even when `wavelength_nm=486.1`.
- No files are written and no global state changes.
- `self.tracer` is built once in `__init__`. Build the system completely before constructing `SpotDiagram`.

## 6. Branches and edge cases
- **Ray launch:**
  - Each pupil point `(py, pz)` becomes `origin = (x0, py, pz) - direction*(50/direction.x)`, where `x0 = elements[0].position`.
  - So the origin X is `x0 - 50` and, on axis, `(y, z) = (py, pz)`.
- **Rays not counted:**
  - Rays with `ray.terminated` after `tracer.trace_ray` (vignetted or TIR).
  - Rays with `|direction.x| <= 1e-6`.
  - Rim rays at exactly `D/2` may be clipped by the tracer (not visible). Do not assert `valid_rays == 127`; assert `<= 127` and `== len(points)`.
- **`num_rings=0`:** only the centre ray, so the failure dict is returned. `valid_rays` is ≤ 1 (1 for a normal lens, with `points ≈ [(0,0)]`). `"image_plane_x"` is still `target_x`.
- **Explicit plane:** `image_plane_x_mm=120.0, focus_shift_mm=-2.5` gives `result["image_plane_x"] == 117.5`, on both the success and failure paths.
- **On-axis symmetric lens:** centroid ≈ `(0, 0)`; use `abs < 1e-9`, not `==`.
- **Off-axis `field_angle_y_deg=5` with z field 0:**
  - `centroid[1] ≈ 0`.
  - `centroid[0] != 0`; it is > 0 for a positive lens imaged at or near its focus.
- **Off-axis `field_angle_x_deg`:** moves `centroid[1]`, not `centroid[0]`.

## 7. Rules from comments
- `spot_points` and `valid_rays` are initialised before the `try`, so earlier failures surface as their real cause, not `UnboundLocalError`.
- The `< 2` check sits after the `finally`, never inside it. Tracer exceptions must propagate and must not become `rms_radius == 0.0`.
- Fewer than two rays must report `None` rather than 0.0, so optimisers and Monte Carlo treat it as a failure.

## 8. Testing hints
Build the system:

Checks worth asserting:

  - `paraxial` is another module, and the element position assigned by `add_lens` is not shown.
  - Read the element position as `sys_.elements[0].position`.
- Recompute the centroid, RMS and geo radius from `result["points"]` with the formulas in §4 and compare with `pytest.approx`.
- Check the key sets exactly: 6 keys on success, 7 on failure, with the exact `"error"` string.

- Failure-path checks: `rms_radius is None`, `geo_radius is None`, `centroid == (0.0, 0.0)` (exact).

### openlens.tolerancing.InverseSensitivityAnalyzer.calculate_sensitivities

# Notes: `InverseSensitivityAnalyzer.calculate_sensitivities(criterion="rms_spot_radius")`

## 1. Purpose
- Applies each operand in `self.tolerances` on its own, at its `max_val`, to `self.system`.
- Measures the RMS spot radius for each one and reports how far it moved from the nominal value, along with a sensitivity (change per unit of perturbation).

## 2. Inputs
- `criterion: str`, default `"rms_spot_radius"`.
  - Compared by plain string equality with the class constant `SUPPORTED_CRITERIA == "rms_spot_radius"`. Note this constant is a str, not a set.
- From `__init__(system, tolerances)`:
  - `self.system` is an `OpticalSystem`.
  - `self.tolerances` is a list of `ToleranceOperand`. Nothing in it is validated or copied.
- Only `tol.max_val`, `tol.param_type`, `tol.element_index` and `tol.surface` are used.
  - `min_val`, `distribution` and `std_dev` are **ignored**.

## 3. Raises
- `ValueError` when `criterion != "rms_spot_radius"`. This is checked **first**, before any tracing.
  - Template: `f"Unsupported criterion {criterion!r}; supported: 'rms_spot_radius'"`
  - Example: `an.calculate_sensitivities("mtf")` gives the message `Unsupported criterion 'mtf'; supported: 'rms_spot_radius'`.
  - Use `pytest.raises(ValueError, match="Unsupported criterion 'mtf'")`.
- The nominal trace `SpotDiagram(self.system).trace_spot()` is **not** guarded. Any exception it raises propagates out unchanged.
- Exceptions raised while applying or tracing a single operand are caught. They produce `val = inf` and log the warning `"Inverse sensitivity for tolerance %d failed: %s"`.

## 4. Returns and guarantees
The result is a new `list` with one dict per tolerance, in the same order as `self.tolerances`. Each dict has exactly these 7 keys:

| Key | Value |
|---|---|
| `"operand_index"` | `i` (0-based position in the tolerances list) |
| `"type"` | `tol.param_type.name`, e.g. `"RADIUS_1"`, `"FOCUS"` (the enum *name*, not `.value`) |
| `"element"` | `tol.element_index` |
| `"test_value"` | `tol.max_val` |
| `"result_value"` | `val` (rms_radius after the perturbation; `None`→`inf`; exception→`inf`) |
| `"change"` | `val - nominal` (nominal `None`→`inf`) |
| `"sensitivity"` | `change / max_val`, or the **int `0`** if `max_val == 0` |

- If `change` or `sensitivity` is not finite, **both** are set to `float("nan")`.
  - `result_value` keeps its inf in that case.
- The docstring says the keys are 'operand', 'sensitivity', 'change'. The real keys are the 7 listed above.
- An empty `tolerances` list returns `[]`, but the nominal trace still runs once.

## 5. Side effects
- Each operand mutates `self.system` in place, through the module helper `_apply_value`.
- After each operand the system is restored in a `finally` block via `_restore_state`. The restore covers:
  - every node's position and rotation;
  - each lens's r1, r2, thickness, `model_glass_mode`, `model_nd`, `model_vd` and `refractive_index`;
  - all air-gap thicknesses;
  - then a call to `system._update_positions()`.
- So after the call, for example, `lens.radius_of_curvature_1` equals its original value.
- `REFRACTIVE_INDEX` and `ABBE_NUMBER` operands call `openlens.lens.ensure_model_glass`, which is not visible here. Only the fields listed above are guaranteed to be restored.
- No files are written and no global state is changed.

## 6. Branches and edge cases
What `_apply_value` does with `value = max_val` (its return value is ignored here):

| `param_type` | Effect |
|---|---|
| `RADIUS_1` / `RADIUS_2` | `+= value`; skipped if that radius is infinite (flat) |
| `THICKNESS` | `+= value` |
| `REFRACTIVE_INDEX` | `model_nd += value` (after switching to model glass) |
| `ABBE_NUMBER` | `model_vd += value` (after switching to model glass) |
| `DECENTER_Y` / `DECENTER_Z` | `node.position.y/z += value` |
| `TILT_X` / `TILT_Y` | `node.rotation.x/y += value` |
| `WEDGE` | `rotation.x += value/120` |
| `DECENTER_X` | `air_gaps[idx-1] += value`; skipped when `idx <= 0` |
| `AIR_GAP` | `air_gaps[idx] += value`; skipped when `idx >= len(air_gaps)` |
| `IRREGULARITY` | Radius solved for the fringe error; skipped if the geometry is unrepresentable |
| `FOCUS` | Always skipped |
| Out-of-range `element_index` | Always skipped |

- When a skip happens, the system is unchanged, so `result_value == nominal`, `change == 0.0` and `sensitivity == 0.0`, assuming the trace is deterministic.
  - Example: `ToleranceOperand(0, ToleranceType.FOCUS, -0.1, 0.1)`.
- `max_val == 0`: `sensitivity` is the int `0` and `change` stays finite (≈0).
- Vignetted nominal (`None`): every entry gets `change` and `sensitivity` as nan.
- Vignetted perturbed trace (`None`), or an exception during it: `result_value == inf`, and `change` and `sensitivity` are nan.

## 7. Rules stated in comments
- A non-finite change or sensitivity means "undefined". It is reported as NaN, never as an infinite sensitivity.
- The restore must happen even if the apply or the trace raises, so a design is never left perturbed.
