    def calculate_back_focal_length(self, n_overrides=None) -> Optional[float]:
        """Calculate Back Focal Length (BFL) of the system."""
        matrix = self._calculate_system_matrix(n_overrides=n_overrides)
        if not matrix:
            return None
        A, B, C, D = matrix
        if abs(C) < 1e-10:
            return None
        return -A / C
